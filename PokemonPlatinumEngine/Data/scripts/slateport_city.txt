# Slateport City (plan 21 · H3). docs/scripts.md describes the language; the lines are our own, on Emerald's beats
# (data/maps/SlateportCity/scripts.inc).
#
# Every arrival sets the museum's counter to ask for its fee again (VAR_SLATEPORT_MUSEUM_1F_STATE 0, as Emerald's own
# arrival does). Once Team Aqua has been driven out of the museum (VAR_SLATEPORT_OUTSIDE_MUSEUM_STATE 1), Scott, a man
# who watches trainers, meets the player outside its door and swaps numbers for the PokéNav (2). The Energy Guru
# sells vitamins at the market (marts.json's slateport_energy_guru).


script OnEnter
  if not flag FLAG_HOENN_CHAPTER_EIGHT_B call HoennChapterEightB
  setflag FLAG_VISITED_SLATEPORT_CITY
  setvar VAR_SLATEPORT_MUSEUM_1F_STATE 0
  if var VAR_SLATEPORT_OUTSIDE_MUSEUM_STATE != 1 end
  scene "Scott outside the museum"
  place slateport_scott 225 288 right
  show slateport_scott
  walk slateport_scott right 3 up 1
  face player slateport_scott
  speaker "Scott"
  say "So you're the one who drove Team Aqua out of the museum! The whole city's talking about it."
  say "I'm Scott. I travel all over looking for trainers with real promise. And you might just be one!"
  if not flag FLAG_SYS_POKENAV_GET goto Leave
  say "Here, let me register you in my PokéNav. I'd like to keep an eye on how you do."
  addcontact scott
  text "Registered Scott in the PokéNav."
label Leave
  say "Keep at it, and I'm sure we'll meet again. Take care!"
  walk slateport_scott down 1 left 6
  hide slateport_scott
  setvar VAR_SLATEPORT_OUTSIDE_MUSEUM_STATE 2
  endscene
  end


# Captain Stern before the harbour's door (plan 21 · H7, part b; Emerald's SlateportCity_EventScript_CaptStern). Once
# Groudon has woken in Team Magma's hideout (VAR_SLATEPORT_CITY_STATE 1) he is back from the sea floor's survey and
# Gabby and Ty have him in front of their camera. Spoken to, he finishes the interview and the pair leave; he tells the
# player of what his team found down there, and a shout comes from the harbour: Team Aqua is taking his submarine. He
# runs in with the player behind him (VAR_SLATEPORT_CITY_STATE 2; the harbour's Stern, Archie and grunt shown), and
# the harbour's own triggers play the rest (slateport_city_harbor.txt).
script CaptStern
  if var VAR_SLATEPORT_CITY_STATE != 1 end
  scene "Captain Stern's interview"
  speaker "Captain Stern"
  say "...So, yes, we'll go on with the survey of the sea floor, a little deeper every time."
  speaker "Gabby"
  say "Wonderful! Thank you so much for talking to us, Captain. Come on, Ty, we've got our story!"
  face slateport_gabby right
  face slateport_ty down
  walk slateport_gabby down 2 right 3
  hide slateport_gabby
  walk slateport_ty down 3 right 2
  hide slateport_ty
  setflag FLAG_HIDE_SLATEPORT_CITY_GABBY_AND_TY
  face slateport_capt_stern player
  speaker "Captain Stern"
  say "Phew... My first interview. I was so nervous my knees were shaking."
  say "Oh, {player}! Listen to this. On the sea floor we found a cave, and inside it a Pokémon that may be the one in the legends!"
  music "hoenn/eye_galactic"
  speaker none
  text "Out of the way! Team Aqua is taking charge of this submarine!"
  face slateport_capt_stern down
  speaker "Captain Stern"
  say "What was that just now?"
  emote slateport_capt_stern exclaim
  face slateport_capt_stern up
  say "That came from the harbour! Somebody's trying to take my submarine!"
  say "{player}, please, come with me!"
  walk slateport_capt_stern up 1
  hide slateport_capt_stern
  setflag FLAG_HIDE_SLATEPORT_CITY_CAPTAIN_STERN
  clearflag FLAG_HIDE_SLATEPORT_CITY_HARBOR_CAPTAIN_STERN
  clearflag FLAG_HIDE_SLATEPORT_CITY_HARBOR_SUBMARINE_SHADOW
  clearflag FLAG_HIDE_SLATEPORT_CITY_HARBOR_AQUA_GRUNT
  clearflag FLAG_HIDE_SLATEPORT_CITY_HARBOR_ARCHIE
  setvar VAR_SLATEPORT_CITY_STATE 2
  warp "SlateportCity_Harbor" 11 13 up
  endscene
  end


# Gabby and Ty in front of Captain Stern, busy with their interview
script Interviewer
  speaker "Gabby"
  say "Shh! We're rolling! The captain's telling us all about the sea floor."
  end


# The Energy Guru of the market, who sells the vitamins
script EnergyGuru
  face self player
  speaker "Energy Guru"
  say "Welcome! I'm the Energy Guru! I sell whatever your Pokémon needs to grow strong!"
  shop "slateport_energy_guru"
  say "Come back whenever your Pokémon need a boost!"
  end


# The market's two stalls of decorations for a secret base (SlateportCity_EventScript_DecorClerk and _DollClerk, plan 21
# · H10: Emerald's pokemartdecoration lists, at Emerald's prices, decorations.json): what is bought goes into the
# decoration PC (`openbase buy`: RESULT 1 bought, 0 short of money, 2 no room for its kind). A question takes six
# answers, so a long list is asked a page at a time.
script DecorClerk
  speaker "Clerk"
  say "Welcome! Bricks, balloons and mats to brighten up a secret base."
label Page1
  choose "Which would you like?" "RED BRICK  500" "BLUE BRICK  500" "YELLOW BRICK  500" "RED BALLOON  500" "MORE" "NO, THANKS"
  if result == 0 goto RedBrick
  if result == 1 goto BlueBrick
  if result == 2 goto YellowBrick
  if result == 3 goto RedBalloon
  if result == 4 goto Page2
  goto Bye
label Page2
  choose "Which would you like?" "BLUE BALLOON  500" "YELLOW BALLOON  500" "C LOW NOTE MAT  500" "D NOTE MAT  500" "MORE" "NO, THANKS"
  if result == 0 goto BlueBalloon
  if result == 1 goto YellowBalloon
  if result == 2 goto CLowMat
  if result == 3 goto DMat
  if result == 4 goto Page3
  goto Bye
label Page3
  choose "Which would you like?" "E NOTE MAT  500" "F NOTE MAT  500" "G NOTE MAT  500" "A NOTE MAT  500" "MORE" "NO, THANKS"
  if result == 0 goto EMat
  if result == 1 goto FMat
  if result == 2 goto GMat
  if result == 3 goto AMat
  if result == 4 goto Page4
  goto Bye
label Page4
  choose "Which would you like?" "B NOTE MAT  500" "C HIGH NOTE MAT  500" "BACK TO THE START" "NO, THANKS"
  if result == 0 goto BMat
  if result == 1 goto CHighMat
  if result == 2 goto Page1
  goto Bye
label RedBrick
  openbase buy "Red Brick"
  goto Sold
label BlueBrick
  openbase buy "Blue Brick"
  goto Sold
label YellowBrick
  openbase buy "Yellow Brick"
  goto Sold
label RedBalloon
  openbase buy "Red Balloon"
  goto Sold
label BlueBalloon
  openbase buy "Blue Balloon"
  goto Sold
label YellowBalloon
  openbase buy "Yellow Balloon"
  goto Sold
label CLowMat
  openbase buy "C Low Note Mat"
  goto Sold
label DMat
  openbase buy "D Note Mat"
  goto Sold
label EMat
  openbase buy "E Note Mat"
  goto Sold
label FMat
  openbase buy "F Note Mat"
  goto Sold
label GMat
  openbase buy "G Note Mat"
  goto Sold
label AMat
  openbase buy "A Note Mat"
  goto Sold
label BMat
  openbase buy "B Note Mat"
  goto Sold
label CHighMat
  openbase buy "C High Note Mat"
label Sold
  if result == 0 goto Poor
  if result == 2 goto Full
  say "Thank you! The {thing} has been sent to your decoration PC."
  end
label Poor
  say "Oh, I'm sorry. You haven't quite enough money for that."
  end
label Full
  say "Your decoration PC has no more room for things of that kind."
  end
label Bye
  say "Please come again!"
  end
