# The Survival Area (plan 01 · M10, part a; its story plan 02 · S15, part a). docs/scripts.md describes the language;
# the lines are our own, on the original's beats (scripts_survival_area.s).
#
# The Battleground's door, read where it stands (the original's script 7). It stays shut until Charon has been caught
# at Stark Mountain: the original moves the door's warp away until FLAG_ARRESTED_CHARON_STARK_MOUNTAIN, and the overlay
# opens it by the same flag (the room is Battleground.txt's: plan 02 · S15, part c).
#
# Buck and the rival come to its step only then, too (the original hides both from the start, and the last scene of
# Stark Mountain clears FLAG_HIDE_SURVIVAL_AREA_BUCK and FLAG_HIDE_SURVIVAL_AREA_RIVAL: plan 02 · S15, part b). Buck
# says this is his place and goes in. The rival trains before the door from then on: on Saturdays and Sundays he
# battles, once a day (FLAG_DAILY_DEFEATED_SURVIVAL_AREA_RIVAL), with the stronger of his two teams once the player
# has entered the Hall of Fame twenty times (the original's GetLeagueVictories); on other days he has one of three
# things to say. The original draws which as the town is come to; here he is asked as he is spoken to.

script BattlegroundDoor
  if flag FLAG_ARRESTED_CHARON_STARK_MOUNTAIN goto Open
  text "The door is locked."
  text "A plate beside it reads: \"Members only.\""
  end
label Open
  text "A plate beside the door reads: \"The Battleground. Members only.\""
  end


# SurvivalArea_Cameraman: after something for his camera, and sure he has found it once Heatran has shown itself
# (FLAG_BATTLED_STARK_MOUNTAIN_ROOM_3_HEATRAN, stark_mountain_room_3.txt: plan 02 · S15, part b)
script Cameraman
  face self player
  if flag FLAG_BATTLED_STARK_MOUNTAIN_ROOM_3_HEATRAN goto Heatran
  sayown
  end
label Heatran
  say "You saw it? The Pokémon in the volcano? You actually battled it?!"
  say "That's it, then. That's my next film! I've got the title, I've got the star... All I need now is the nerve to go up there."
  end


# SurvivalArea_Buck
script Buck
  face self player
  speaker "Buck"
  say "{player}! You made it! Well, this is it: my place. The Battleground."
  say "Only the toughest Trainers I know get asked inside. After what we went through up the mountain, that means you. Come in whenever you like!"
  face self up
  sound "door_open"
  wait 0.3
  hide self
  setflag FLAG_HIDE_SURVIVAL_AREA_BUCK
  sound "door_close"
  end


# SurvivalArea_Rival
script Rival
  face self player
  speaker "{rival}"
  if weekday saturday goto Weekend
  if weekday sunday goto Weekend
  random 3
  if result == 0 goto PilingUp
  if result == 1 goto Everyone
  say "You know what I've worked out, {player}? The best Trainers are the ones who are crazy about their Pokémon. That's all there is to it!"
  end
label PilingUp
  say "{player}! I'm piling up the wins at the Battle Frontier! My dad had better watch out. I'm coming for him!"
  end
label Everyone
  say "Have you battled everybody in the Battleground yet, {player}? I have. Twice! ...All right, I lost a few."
  end
label Weekend
  if flag FLAG_DAILY_DEFEATED_SURVIVAL_AREA_RIVAL goto After
  say "{player}! Perfect timing! It's the weekend, and that means it's time I proved which of us is tougher!"
  if var LEAGUE_VICTORIES >= 20 goto Stronger
  if starter "Chimchar" goto Chimchar
  if starter "Piplup" goto Piplup
  battle self as "rival_survival_area_1_turtwig"
  goto Won
label Chimchar
  battle self as "rival_survival_area_1_chimchar"
  goto Won
label Piplup
  battle self as "rival_survival_area_1_piplup"
  goto Won
label Stronger
  if starter "Chimchar" goto Chimchar2
  if starter "Piplup" goto Piplup2
  battle self as "rival_survival_area_2_turtwig"
  goto Won
label Chimchar2
  battle self as "rival_survival_area_2_chimchar"
  goto Won
label Piplup2
  battle self as "rival_survival_area_2_piplup"
label Won
  setflag FLAG_DAILY_DEFEATED_SURVIVAL_AREA_RIVAL
label After
  say "Argh! You just keep getting tougher, {player}! Fine! Then so will I. Same time next week!"
  end


# Plan 44 · SD3: Looker gives the Reclaimer, made by his bureau from a part seized at the quay, and tells of the spring
script LookerReclaimer
  face self player
  speaker self
  if flag FLAG_GOT_RECLAIMER goto Again
  say "Ah, it is you! Looker is here on a matter of shipments. Crates that go out on the ferries and never quite arrive."
  say "A syndicate called Lockwork. They put a machine to a Pokémon's heart and lock it shut. It obeys, but it feels nothing for its trainer."
  say "My bureau took a part of that machine from a crate at the Fight Area's quay. From it the technicians made this."
  heart give
  fanfare keyitem
  text "{player} received the {item}!"
  say "Wear it on your throwing arm. With it a ball can reach a locked Pokémon a trainer sends out. Take them back!"
  say "A locked heart opens in time beside a trainer who cares for it. At Sendoff Spring, the water itself helps."
  say "Their cell here hides behind the quay's containers. Every port the ferries reach has one. Be careful, my friend."
  end
label Again
  say "A locked heart opens with battles, with walking beside you, with the camp's care. And at Sendoff Spring, once a day."
  say "Every port the ferries reach has a cell. Their head office, I am told, is a tower in Castelia City."
  end

# The delve (plan 44 · SD8, SD9; decisions 5 to 7, the user's answers 48a and 62a): the camp's keeper, Brigid Stoat,
# sends two of the team down one of the four hollows behind the camp once Sinnoh's Hall of Fame is entered. The first
# two of the team who may go down go; each hollow cleared raises the camp's rank, which opens the next. Her board holds
# the day's three requests, and she hands over what they paid.
script DelveKeeper
  face self player
  speaker self
  say "Brigid Stoat, keeper of the delve camp. Behind me are four hollows, and Pokémon live on every floor of them."
  say "Your first two Pokémon go down together while you wait here. Reach the bottom and a local may come home with you."
  choose "What can I do for you?" "GO DOWN" "THE BOARD" "PRIZES" "NOTHING"
  if result == 1 goto Board
  if result == 2 goto Prizes
  if result == 3 goto Later
  delve able
  if result == 0 goto NotTwo
  choose "Which hollow?" "BRINECAVE" "THE ROOTWOOD" "HEARTPEAK" "THE OLD TOWER" "NOT TODAY"
  if result == 0 goto Brine
  if result == 1 goto Root
  if result == 2 goto Peak
  if result == 3 goto Tower
  goto Later
label Brine
  delve go "brinecave"
  goto Back
label Root
  delve open "rootwood"
  if result == 0 goto Shut
  delve go "rootwood"
  goto Back
label Peak
  delve open "heartpeak"
  if result == 0 goto Shut
  delve go "heartpeak"
  goto Back
label Tower
  delve open "old_tower"
  if result == 0 goto Shut
  delve go "old_tower"
  goto Back
label Back
  if result == 1 goto Cleared
  if result == 2 goto Fell
  if result == 3 goto Rope
  goto Later
label Cleared
  say "All the way down and back! Well done, the pair of you. Anyone who came home with you is settling in."
  goto Prizes
label Fell
  say "Carried out by the camp's rope team, I'm afraid. Whatever you found stayed below. Rest up and try again!"
  end
label Rope
  say "Out by the rope. Sensible! You kept what you found."
  goto Prizes
label Board
  delve board
  say "The board changes every morning. Do what it asks on a run and I'll have something for you."
  end
label Prizes
  delve reward
  if result == 1 goto Prizes
  say "That's everything waiting for you for now."
  end
label NotTwo
  say "I need two Pokémon who can go: no Egg and no locked heart. Put two at the head of your team."
  end
label Shut
  say "That hollow is too deep for a camp of this rank. Clear the ones before it first."
  end
label Later
  say "The hollows aren't going anywhere."
  end
