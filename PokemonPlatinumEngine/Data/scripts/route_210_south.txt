# Route 210, south of the fog (plan 06 · R14a). docs/scripts.md describes the language.
#
# The original's arrival script: in the morning the joggers battle, and at any other time their doubles on the same
# tiles only jog past. The swap is made as the player comes, so it holds until they come again.


script OnEnter
  if time morning goto Battle
  clearflag FLAG_HIDE_ROUTE_210_SOUTH_JOGGER_WYATT_NO_BATTLE
  setflag FLAG_HIDE_ROUTE_210_SOUTH_JOGGER_WYATT
  end
label Battle
  clearflag FLAG_HIDE_ROUTE_210_SOUTH_JOGGER_WYATT
  setflag FLAG_HIDE_ROUTE_210_SOUTH_JOGGER_WYATT_NO_BATTLE
  end
