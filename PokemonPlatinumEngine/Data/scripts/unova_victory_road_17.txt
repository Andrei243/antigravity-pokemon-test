# overlay's holes, a wall until their flag, and the pits on the holes' tiles, whose script sets the flag as the boulder comes to
# rest. The boulder is gone with the hole it filled.


script FillEast
  call BoulderFallsIn
  setflag FLAG_UNOVA_VICTORY_ROAD_RUINS_HOLE_EAST
  hide self
  end


script FillWest

  hide self
  end
