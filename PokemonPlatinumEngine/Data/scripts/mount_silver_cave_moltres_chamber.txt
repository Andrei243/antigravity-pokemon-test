# Moltres's chamber in Mt. Silver Cave (plan 20 · J15). docs/scripts.md describes the language; the lines are our own,
# on HeartGold's beats (scr_seq_D41R0105).
#
# Moltres waits in the chamber until it has been fought (FLAG_JOHTO_FOUGHT_MOLTRES, ours), as on Mt. Ember in FireRed's
# Kanto: beaten or caught it is gone for good, run from it flies off and is back on the next visit.
#
# Its meeting is plan 29's beat (style guide, "Legendary scenes"): the bars, the camera to Moltres, its fire bursting
# and the heat wavering, the chamber shaken with its cry; the scene ends before the battle, so the hold never skips it.


script OnEnter
  if not flag FLAG_JOHTO_FOUGHT_MOLTRES clearflag FLAG_HIDE_MT_SILVER_CAVE_MOLTRES
  end

script Moltres
  scene "Moltres in Mt Silver"
  music stop
  face player self
  letterbox on 0.4
  camera pan 10 9 1.2
  camera zoom 1.4 0.8
  effect burst 10 9 wait
  effect ripple 10 9
  cry "Moltres"
  camera shake 0.6
  emote player exclaim
  speaker none
  text "The chamber blazed as Moltres spread its wings of fire!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Moltres" 50
  if result == 2 goto Fled
  setflag FLAG_JOHTO_FOUGHT_MOLTRES
  setflag FLAG_HIDE_MT_SILVER_CAVE_MOLTRES
  hide self
  end
label Fled
  setflag FLAG_HIDE_MT_SILVER_CAVE_MOLTRES
  hide self
  text "Moltres flew off into the dark of the cave!"
  end
