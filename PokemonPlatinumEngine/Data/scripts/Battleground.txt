# The Battleground, Buck's family's place in the Survival Area (plan 02 · S15, part c), on the original's plan.
# docs/scripts.md describes the language; the lines are our own, on the original's beats (scripts_battleground.s).
#
# The door opens once Charon has been caught on Stark Mountain (the Survival Area's overlay). The first time in, the
# old man behind the bar, Buck's grandfather, has the player come over and welcomes them: a private club for Sinnoh's
# Gym Leaders and whoever they vouch for (VAR_BATTLEGROUND_STATE 1). Buck stands by the bar until he is spoken to: his
# place is the player's too, and he is off to the Battle Tower (FLAG_TALKED_TO_BATTLEGROUND_BUCK, which lets Heatran
# wake in Stark Mountain's heart; FLAG_BUCK_LEFT_BATTLEGROUND, which lets him be met in the Battle Salon and drawn as
# one of this room's own guests).
#
# Who is in today is drawn as the room is first come into each day (FLAG_DAILY_SET_BATTLEGROUND_TRAINERS), by the
# original's GetRandomBattlegroundTrainers:
#
#   the first place is one of the eight Gym Leaders, each as likely as another;
#   the second and the third are another Leader each, or nobody: a die of nine, where a nine is nobody, thrown again
#   while it shows a Leader already drawn, and nobody at the eighth throw;
#   the fourth is one of the five who have battled beside the player, or nobody (a die of six), and they come only once
#   the story has had the player travel with them: Cheryl through Eterna Forest, Riley on Iron Island (his Egg taken),
#   Marley through Victory Road, Mira out of Wayward Cave, and Buck once he has left for the Tower.
#
# VAR_BATTLEGROUND_TRAINER_1 to _4 keep the draw, as the original's do: a Leader is 0 to 7 in its list's order (Roark,
# Gardenia, Crasher Wake, Maylene, Fantina, Candice, Byron, Volkner) and nobody 8; of the fourth, 0 to 4 are Cheryl,
# Riley, Marley, Buck and Mira and 5 nobody. FLAG_HIDE_BATTLEGROUND_TRAINER_4 is the original's too: set at the draw
# for a guest the story hasn't brought yet, who then stays away for the day. The original gives four objects the looks
# of whoever was drawn; here each of the thirteen is someone of the room, shown by a flag of our own
# (FLAG_BATTLEGROUND_HAS_<WHO>) and put on their place as the room is come into.
#
# Each asks for a battle (one of two sets of words, by a coin thrown at the draw: VAR_RANDOM_BATTLEGROUND_MESSAGE), and
# asks again in other words once refused, until the room is left. Agreed to, the battle is their rematch team of
# Platinum's data, and counts (VAR_TIMES_BATTLED_AT_BATTLEGROUND, our name for the original's game record, which the
# Villa's visitors read: plan 06 · R16b). Beaten, they are beaten for the day (FLAG_DAILY_DEFEATED_BATTLEGROUND_<WHO>)
# and have something else to say; Roark and Byron have other words while both are in.
#
# Stood in for: the Leaders', Buck's and the guests' looks where plan 11 has none yet (Candice, Byron, Volkner, Riley,
# Marley, Buck: the looks the story's other scenes give them), and the Survival Area's song.


# Battleground_OnTransition, and the frame table's welcome
script OnEnter
  setflag FLAG_VISITED_BATTLEGROUND
  if not flag FLAG_DAILY_SET_BATTLEGROUND_TRAINERS call Draw
  call ClearOfThePlaces
  call Stand
  if var VAR_BATTLEGROUND_STATE == 0 call Welcome
  end


# Battleground_SetTrainers
script Draw
  setflag FLAG_DAILY_SET_BATTLEGROUND_TRAINERS
  # The first: a Leader
  random 8
  setvar VAR_BATTLEGROUND_TRAINER_1 RESULT
  # The second: another Leader, or nobody (GetRandomBattlegroundGymLeaderID)
  setvar VAR_BATTLEGROUND_TRAINER_2 8
  setvar VAR_MAP_LOCAL_0x02 0
label Second
  random 9
  if result == 8 goto Third
  addvar VAR_MAP_LOCAL_0x02 1
  if var VAR_MAP_LOCAL_0x02 >= 8 goto Third
  if var RESULT == VAR_BATTLEGROUND_TRAINER_1 goto Second
  setvar VAR_BATTLEGROUND_TRAINER_2 RESULT
label Third
  # The third: a Leader who is neither of those, or nobody
  setvar VAR_BATTLEGROUND_TRAINER_3 8
  setvar VAR_MAP_LOCAL_0x02 0
label ThirdAgain
  random 9
  if result == 8 goto Fourth
  addvar VAR_MAP_LOCAL_0x02 1
  if var VAR_MAP_LOCAL_0x02 >= 8 goto Fourth
  if var RESULT == VAR_BATTLEGROUND_TRAINER_1 goto ThirdAgain
  if var RESULT == VAR_BATTLEGROUND_TRAINER_2 goto ThirdAgain
  setvar VAR_BATTLEGROUND_TRAINER_3 RESULT
label Fourth
  # The fourth: one of the five who have battled beside the player, or nobody (GetRandomBattlegroundStatTrainerID),
  # and only if the story has brought them (Battleground_TryHideCheryl and the rest)
  random 6
  setvar VAR_BATTLEGROUND_TRAINER_4 RESULT
  clearflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
  if var VAR_BATTLEGROUND_TRAINER_4 == 5 setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
  if var VAR_BATTLEGROUND_TRAINER_4 != 0 goto NotCheryl
  if not flag FLAG_TRAVELED_WITH_CHERYL setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
label NotCheryl
  if var VAR_BATTLEGROUND_TRAINER_4 != 1 goto NotRiley
  if not flag FLAG_RECEIVED_RIOLU_EGG_FROM_RILEY setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
label NotRiley
  if var VAR_BATTLEGROUND_TRAINER_4 != 2 goto NotMarley
  if not flag FLAG_TRAVELED_WITH_MARLEY setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
label NotMarley
  if var VAR_BATTLEGROUND_TRAINER_4 != 3 goto NotBuck
  if not flag FLAG_BUCK_LEFT_BATTLEGROUND setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
label NotBuck
  if var VAR_BATTLEGROUND_TRAINER_4 != 4 goto Words
  if not flag FLAG_TRAVELED_WITH_MIRA setflag FLAG_HIDE_BATTLEGROUND_TRAINER_4
label Words
  # Which of each guest's two sets of words the day has
  random 2
  setvar VAR_RANDOM_BATTLEGROUND_MESSAGE RESULT
  # (who was beaten yesterday is forgotten with the day's other flags: FLAG_DAILY_DEFEATED_BATTLEGROUND_<WHO>)
  return


# A game saved on one of the four places and taken up on another day would find a guest put down where the player
# stands: whoever is on one of them as the room is come to is moved to the mat
script ClearOfThePlaces
  if var PLAYER_X == 2 goto West
  if var PLAYER_X == 6 goto Middle
  if var PLAYER_X == 9 goto Table
  if var PLAYER_X == 11 goto Screen
  return
label West
  if var PLAYER_Y == 9 place player 7 11 up
  return
label Middle
  if var PLAYER_Y == 8 place player 7 11 up
  return
label Table
  if var PLAYER_Y == 7 place player 7 11 up
  return
label Screen
  if var PLAYER_Y == 5 place player 7 11 up
  return


# Battleground_SetTrainersGraphics: whoever the day drew, on their places

  if var VAR_BATTLEGROUND_TRAINER_1 == 5 place candice 2 9 right
  if var VAR_BATTLEGROUND_TRAINER_1 == 6 place byron 2 9 right
  if var VAR_BATTLEGROUND_TRAINER_1 == 7 place volkner 2 9 right
  # The second, in the middle of the floor
  setvar VAR_MAP_LOCAL_0x01 VAR_BATTLEGROUND_TRAINER_2
  call Leader
  if var VAR_BATTLEGROUND_TRAINER_2 == 0 place roark 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 1 place gardenia 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 2 place wake 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 3 place maylene 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 4 place fantina 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 5 place candice 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 6 place byron 6 8 down
  if var VAR_BATTLEGROUND_TRAINER_2 == 7 place volkner 6 8 down
  # The third, at the big table
  setvar VAR_MAP_LOCAL_0x01 VAR_BATTLEGROUND_TRAINER_3
  call Leader
  if var VAR_BATTLEGROUND_TRAINER_3 == 0 place roark 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 1 place gardenia 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 2 place wake 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 3 place maylene 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 4 place fantina 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 5 place candice 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 6 place byron 9 7 left
  if var VAR_BATTLEGROUND_TRAINER_3 == 7 place volkner 9 7 left

  if var VAR_BATTLEGROUND_TRAINER_4 == 1 setflag FLAG_BATTLEGROUND_HAS_RILEY
  if var VAR_BATTLEGROUND_TRAINER_4 == 2 setflag FLAG_BATTLEGROUND_HAS_MARLEY
  if var VAR_BATTLEGROUND_TRAINER_4 == 3 setflag FLAG_BATTLEGROUND_HAS_BUCK
  if var VAR_BATTLEGROUND_TRAINER_4 == 4 setflag FLAG_BATTLEGROUND_HAS_MIRA
  return


# The Leader VAR_MAP_LOCAL_0x01 names is in today
script Leader
  if var VAR_MAP_LOCAL_0x01 == 0 setflag FLAG_BATTLEGROUND_HAS_ROARK
  if var VAR_MAP_LOCAL_0x01 == 1 setflag FLAG_BATTLEGROUND_HAS_GARDENIA
  if var VAR_MAP_LOCAL_0x01 == 2 setflag FLAG_BATTLEGROUND_HAS_WAKE
  if var VAR_MAP_LOCAL_0x01 == 3 setflag FLAG_BATTLEGROUND_HAS_MAYLENE
  if var VAR_MAP_LOCAL_0x01 == 4 setflag FLAG_BATTLEGROUND_HAS_FANTINA
  if var VAR_MAP_LOCAL_0x01 == 5 setflag FLAG_BATTLEGROUND_HAS_CANDICE
  if var VAR_MAP_LOCAL_0x01 == 6 setflag FLAG_BATTLEGROUND_HAS_BYRON
  if var VAR_MAP_LOCAL_0x01 == 7 setflag FLAG_BATTLEGROUND_HAS_VOLKNER
  return


# Battleground_OnFrame_Welcome: the old man sees who has come in, and the player goes over to the bar. The original
# walks the player four tiles east, to the tile under Buck's own (11, 9), where the player's head would hide him: here
# the walk stops a tile short, clear of him
script Welcome
  setvar VAR_BATTLEGROUND_STATE 1
  emote expert_m exclaim
  # (whoever is not on the mat, a game saved in the room, is spoken to where they stand)
  if var PLAYER_X != 7 goto Speak
  if var PLAYER_Y != 11 goto Speak
  walk player up 2 right 3
  face player right
label Speak
  speaker "Old Man"
  say "Oh? Then you must be...! Yes, yes: Buck's friend! I've had the whole story twice over, once from Buck and once from Wake."
  say "Well, my young friend, from today this door is open to you whenever you like."
  say "This is a private club. Sinnoh's Gym Leaders come here, and whoever a Gym Leader will vouch for. Nobody else."
  say "The best there are, come to get better still. Welcome to the Battleground!"
  return


# Battleground_ExpertM: Buck's grandfather, behind his bar
script OldMan
  face self player
  speaker "Old Man"
  if flag FLAG_BATTLED_STARK_MOUNTAIN_ROOM_3_HEATRAN goto AtHome
  say "Now, what shall I tell you about today? ...I know. Stark Mountain."
  say "That volcano is as old as Sinnoh, from when time and space first ran together and made the land."
  say "They say the lava that welled up and pooled there became a Pokémon."
  say "And that the Magma Stone was set on it to hold it down, so the Pokémon wouldn't rage and the mountain wouldn't burst."
  end
label AtHome
  say "Welcome to the Battleground! Make yourself at home, the way you always do."
  end


# Battleground_Buck: he goes out by the door, round whoever is in his way
script Buck
  face self player
  speaker "Buck"
  setflag FLAG_TALKED_TO_BATTLEGROUND_BUCK
  say "My place is for the toughest Trainers and nobody else. I mean it!"
  say "You're meant to be brought in by a Gym Leader, but you? You're a special case."
  say "Right, I'm off to the Battle Frontier. I want to see how far up that Tower I can get!"
  if var PLAYER_Y == 8 goto FromTheWest
  walk self left 1 down 1 left 3 down 2
  goto Gone
label FromTheWest
  walk self down 1 left 4 down 2
label Gone
  face player left
  sound "door_open"
  hide self
  setflag FLAG_BUCK_LEFT_BATTLEGROUND
  setflag FLAG_HIDE_BATTLEGROUND_BUCK
  end


script Roark
  face self player
  speaker "Roark"
  if flag FLAG_DAILY_DEFEATED_BATTLEGROUND_ROARK goto After
  if var VAR_RANDOM_BATTLEGROUND_MESSAGE == 1 goto Second
  if flag FLAG_MAP_LOCAL_BATTLEGROUND_DECLINED_ROARK goto Again
  ask "I've been down in the Underground every single day, digging up Fossil after Fossil. My Pokémon came up tougher each time. Could I show you how tough, in a battle?"
  goto Answer
label Again
  ask "Hold on, could I have a moment? Running into you here feels like luck. Will you take my challenge after all?"
label Answer
  if no goto Declined
  say "Here goes! These are my rock-solid Pokémon, my pride and joy!"
  goto Battle
label Declined
  setflag FLAG_MAP_LOCAL_BATTLEGROUND_DECLINED_ROARK
  say "Something else, then. You know the Spheres you dig up down there? Why do they grow when you leave them buried?"
  say "It doesn't matter much. But isn't it odd, when you think about it?"
  end
label Second
  if flag FLAG_MAP_LOCAL_BATTLEGROUND_DECLINED_ROARK goto Again2
  ask "Would you like a battle? You'd meet a different Roark from the one you beat at my Gym."
  goto Answer2
label Again2
  ask "Oh? You do want to battle?"
label Answer2
  if no goto Declined2
  say "I'm sending out my Tyranitar. It's tougher than my dad's Aggron, whatever he says!"
label Battle
  addvar VAR_TIMES_BATTLED_AT_BATTLEGROUND 1
  battle self as "leader_roark_rematch"
  setflag FLAG_DAILY_DEFEATED_BATTLEGROUND_ROARK
  speaker "Roark"
label After
  if flag FLAG_BATTLEGROUND_HAS_BYRON goto WithDad
  if var VAR_RANDOM_BATTLEGROUND_MESSAGE == 1 goto After2
  say "...We got carried away there. Next time, how about a race instead? Whoever digs up a Fossil first, down in the Underground."
  end
label After2
  say "Pokémon from ages past... Fossils nobody has found yet... If I dug the whole of Sinnoh over, would I turn up a new one?"
  end
label Declined2
  setflag FLAG_MAP_LOCAL_BATTLEGROUND_DECLINED_ROARK
  say "Hmm... Now that's an answer to break a heart."
  end
label WithDad
  say "It took becoming a Gym Leader myself to see how good my dad really is."
  say "Not that I'd ever say so where he could hear me. Keep it to yourself!"
  end
