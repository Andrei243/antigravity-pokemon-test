# Route 40 (plan 20 · J13). docs/scripts.md describes the language; the lines are our own, on HeartGold's beats
# (scr_seq_W40).
#
# Monica of Monday, the first of the week's seven siblings, is on the beach on Mondays alone (FLAG_UNK_206, HeartGold's
# W40_000) and gives a Sharp Beak once (FLAG_GOT_SHARP_BEAK_FROM_MONICA), counting herself among the siblings met
# (VAR_NUM_MET_WEEKDAY_SIBLINGS); once all seven are met she gives the ribbon of the day (common.WeekSiblingRibbon).


#
# Warden Baoba (plan 20, part q: HeartGold's map holds no object of his here, so he is the overlay's own) waits on the
# beach by Olivine City until his Safari Zone opens: the route's arrival shows him while his variable is 0
# (VAR_UNK_4057, safari_zone_entrance_interior.txt) and takes him away once his call has set it (olivine.txt). Spoken
# to, he tells of the zone he is building beyond Cianwood City and says he will ring (FLAG_JOHTO_MET_BAOBA, ours), which
# his call then answers ("Sorry to keep you waiting").


# W40_000, the route's own script as it loads
script OnEnter
  setflag FLAG_UNK_206
  if weekday monday clearflag FLAG_UNK_206
  setflag FLAG_JOHTO_HIDE_ROUTE_40_BAOBA
  if var VAR_UNK_4057 == 0 clearflag FLAG_JOHTO_HIDE_ROUTE_40_BAOBA
  end


# Warden Baoba, before his Safari Zone opens (ours)
script Baoba
  face route_40_baoba player
  speaker "Baoba"
  if flag FLAG_JOHTO_MET_BAOBA goto Again
  say "Hello there, young Trainer! The name's Baoba. I'm the warden of a Safari Zone, or I will be soon."
  say "I'm building it past the cliffs on the far side of Cianwood City. Wild Pokémon from all over, free for the catching!"
  say "It isn't quite ready yet. When it is, I'll give you a ring, so don't you go changing your number!"
  setflag FLAG_JOHTO_MET_BAOBA
  end
label Again
  say "The workmen are still busy at my Safari Zone. I'll ring you the moment its gate opens!"
  end


# W40_001: Monica of Monday
script Monica
  face w40_gswoman1 player
  speaker "Monica"
  if var VAR_NUM_MET_WEEKDAY_SIBLINGS >= 7 goto All
  if flag FLAG_GOT_SHARP_BEAK_FROM_MONICA goto Again
  if not weekday monday goto NotToday
  say "Good morning! I'm Monica of Monday. Monday is the start of the week, and the start of my brothers' and sisters' days!"
  give "Sharp Beak"
  setflag FLAG_GOT_SHARP_BEAK_FROM_MONICA
  addvar VAR_NUM_MET_WEEKDAY_SIBLINGS 1
  say "A Pokémon that holds a Sharp Beak hits harder with Flying-type moves. Have a happy Monday!"
  end
label Again
  say "There are seven of us, one for each day. Have you met the others? They're all over Johto."
  end
label NotToday
  say "Today isn't Monday? Oh dear, I must have lost track of the week. Monday is my day!"
  end
label All
  if not weekday monday goto NotToday
  call WeekSiblingRibbon
  end
