# Hokulani Observatory, Mount Hokulani, Ula'ula Island: Captain Sophocles' trial (plan 23 · Z7). The lines are our own,
# on Sun and Moon's beats as they are publicly described (docs/mechanics/rulings.md, "Ula'ula, Poni, Aether Paradise
# and the League"). Three Charjabug have crept into the observatory's wiring; the totem, a Togedemaru, is fought as
# a totem (plan 23 · Z6: its aura, its first call answered by a Charjabug, no ball, no running; wave 91). VAR_ALOLA_STORY 42 is the
# way in, 43 the trial passed.
#
# The people: sophocles, the three Charjabug and the totem are shown by FLAG_ALOLA_HOKULANI_TRIAL while the trial is
# under way.


script OnEnter
  if var VAR_ALOLA_STORY != 42 end
  if flag FLAG_ALOLA_HOKULANI_TRIAL end
  if flag FLAG_ALOLA_HOKULANI_TRIAL_DONE end
  scene "Sophocles' trial begins"
  setflag FLAG_ALOLA_HOKULANI_TRIAL
  speaker "Sophocles"
  say "Oh! A visitor! ...Oh no, you're a trial-goer. I'm Sophocles, the captain. I get nervous with company."
  say "My trial is simple. Three Charjabug got in through the wiring, and they've been charging the whole observatory. Would you find them?"
  say "They're in the dome, around the room. And then the big one. The totem. I've kept it fed. Please be nice."
  endscene
  end


script SophoclesInside
  speaker "Sophocles"
  say "The Charjabug are in the room, and the totem is waiting behind them. I'll hold the door. Metaphorically."
  end


script Den1
  text "A Charjabug bolted out of the wiring!"
  wildbattle "Charjabug" 22 nofleeing
  setflag FLAG_ALOLA_HOKULANI_DEN_1
  hide self
  end

script Den2
  text "A Charjabug bolted out of the wiring!"
  wildbattle "Charjabug" 22 nofleeing
  setflag FLAG_ALOLA_HOKULANI_DEN_2
  hide self
  end

script Den3
  text "A Charjabug bolted out of the wiring!"
  wildbattle "Charjabug" 23 nofleeing
  setflag FLAG_ALOLA_HOKULANI_DEN_3
  hide self
  end


script Totem
  speaker none
  if not flag FLAG_ALOLA_HOKULANI_DEN_1 goto NotYet
  if not flag FLAG_ALOLA_HOKULANI_DEN_2 goto NotYet
  if not flag FLAG_ALOLA_HOKULANI_DEN_3 goto NotYet
  scene "The totem Togedemaru"
  text "The Togedemaru's spines crackled, and electricity ran up every wire in the room!"
  cry "Togedemaru"
  # A totem (plan 23 · Z6): its aura raises its Defense, it calls a Charjabug at once, and it can't be caught
  wildbattle "Togedemaru" 24 totem defense ally "Charjabug"
  setflag FLAG_ALOLA_HOKULANI_TRIAL_DONE
  hide self
  hide sophocles
  speaker "Sophocles"
  say "...You did it! You passed my trial! I've never seen the whole observatory glow like that."
  say "This is the Electrium Z. With it, an Electric-type move becomes a Z-Move. Please keep it safe."
  give "Electrium Z"
  say "Acerola's trial is next, in Tapu Village. Go east along Route 12, and... and don't tell her I sent you."
  setvar VAR_ALOLA_STORY 43
  objective alolaTapu
  endscene
  end
label NotYet
  text "The Togedemaru hasn't come out yet. Perhaps the Charjabug in the room come first."
  end


# After the Hall of Fame (plan 23, the wave after the loose ends): a second totem has come to Hokulani Observatory, the one another
# of Sun and Moon's versions keeps here. Fought as a totem (its aura, its first call answered) at 53, gone for good
# once beaten (FLAG_ALOLA_TOTEM_VIKAVOLT_DONE)
script SecondTotem

  end


# Magearna (plan 23, Alola finished; plan 29 · LS11's rule for Alola, docs/plans/decisions/LS11-proposals-event-legendaries.md,
# "Alola", built to its recommended answers, which the user may still change). Sun and Moon give it by an event alone;
# ours waits in the observatory once Alola's Hall of Fame is entered and the second totem has been beaten here: the
# Vikavolt had been feeding on the old machine's power, and with it gone the machine wakes (the room shows it by
# FLAG_ALOLA_TOTEM_VIKAVOLT_DONE). Level 60, as Alola's other legends after the League. Beaten or caught it is gone for
# good (FLAG_ALOLA_MAGEARNA_GONE); run from, it stays.
script Magearna
  scene "Magearna in Hokulani Observatory"
  music stop
  letterbox on 0.4
  # (one tile south of it, the player would hide it behind their head: they step back first)
  if var PLAYER_X != 12 goto Look
  if var PLAYER_Y != 5 goto Look
  walk player down 2
  face player up
label Look
  camera pan 12 4 1.0
  camera zoom 1.4 0.8
  effect sparkles 12 4 2.0
  shake self 6
  effect burst 12 4
  cry "Magearna"
  speaker none
  text "Gears turned inside the old machine, and a little figure of pink and steel lifted its head. Its heart glowed!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Magearna" 60
  if result == 2 goto Stays
  setflag FLAG_ALOLA_MAGEARNA_GONE
  hide self
  music area
  end
label Stays
  music area
  text "The little machine sat back down among the wires, still ticking softly."
  end
