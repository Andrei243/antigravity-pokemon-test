# The Game Corner's prize exchange in Celadon City (plan 19 · K5, part b). docs/scripts.md describes the language;
# the lines are our own, on FireRed's beats (data/maps/CeladonCity_GameCorner_PrizeRoom/scripts.inc).
#
# FireRed's three counters give Pokémon, TMs and items for coins. The exchange here is plan 06 · R16b's list
# (gamecorner prize, gamecorner buy) at the second and third; the first gives FireRed's five Pokémon at FireRed's
# prices and levels (plan 40 · GE3: PokemonPrizes, our words on FireRed's CeladonCity_GameCorner_PrizeRoom_EventScript_
# PrizeClerk1). A prize Pokémon goes onto the team, or to a box when the team is full.


script PokemonPrizes
  speaker "Clerk"
  if not item "Coin Case" goto NoCase
  gamecorner coins
  say "Welcome! This counter trades coins for Pokémon. You have {coins} coins."
  choose "Which Pokémon would you like?" "Abra: 180" "Clefairy: 500" "Dratini: 2,800" "Scyther: 5,500" "Porygon: 9,999" "No, thanks"
  if result == 0 goto Abra
  if result == 1 goto Clefairy
  if result == 2 goto Dratini
  if result == 3 goto Scyther
  if result == 4 goto Porygon
  goto Bye
label Abra
  gamecorner take 180
  if result == 0 goto Poor
  givepokemon "Abra" 9
  goto Given
label Clefairy
  gamecorner take 500
  if result == 0 goto Poor
  givepokemon "Clefairy" 8
  goto Given
label Dratini
  gamecorner take 2800
  if result == 0 goto Poor
  givepokemon "Dratini" 18
  goto Given
label Scyther
  gamecorner take 5500
  if result == 0 goto Poor
  givepokemon "Scyther" 25
  goto Given
label Porygon
  gamecorner take 9999
  if result == 0 goto Poor
  givepokemon "Porygon" 26
label Given
  if result == 2 goto Boxed
  fanfare pokemon
  say "Here you are! Look after it well."
  end
label Boxed
  fanfare pokemon
  say "Your team is full, so we sent it to your PC's boxes. Look after it well."
  end
label Bye
  say "Please come again!"
  end
label NoCase
  say "Prizes are exchanged for coins. You'll need a Coin Case first."
  end
label Poor
  say "I'm sorry, you don't have enough coins."
  end


script Prizes
  speaker "Clerk"
  if not item "Coin Case" goto NoCase
  gamecorner coins
  say "Welcome! You have {coins} coins. Which prize would you like?"
  gamecorner prize
  if result == 0 goto Bye
  ask "The {item} for {price} coins?"
  if no goto Bye
  gamecorner buy
  if result == 0 goto Poor
  text "{player} received the {item}!"
label Bye
  say "Please come again!"
  end
label NoCase
  say "Prizes are exchanged for coins. You'll need a Coin Case first."
  end
label Poor
  say "I'm sorry, you don't have enough coins."
  end
