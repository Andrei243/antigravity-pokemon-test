# Snowpoint City (plan 02 · S11, part a). docs/scripts.md describes the language; the lines are our own, on the
# original's beats (scripts_snowpoint_city.s).
#
# The town's part in the story is its Gym (SnowpointGym.txt, plan 01 · M9: Candice and the Icicle Badge, which lets
# Rock Climb take the player up Acuity Lakefront's rock face) and Maylene, warming up in its Pokémon Center
# (SnowpointPokemonCenter.txt). Snowpoint Temple stands shut at the top of town: its guard stands before the east
# half of its door, and whoever steps up to the west half (the original's trigger 0, one tile at 366, 198, while
# VAR_SNOWPOINT_CITY_STATE is 0) is turned back.
#
# After the Hall of Fame (plan 02 · S15, part a) the sailor on the quay takes a Champion to the Battle Zone: the
# original asks FLAG_GAME_COMPLETED and nothing else, and lands the player on the Fight Area's pier at 623, 434, facing
# east (TakeShipFromSnowpoint). The first time, Cynthia comes down to the quay to see the player off
# (FLAG_HIDE_SNOWPOINT_CITY_CYNTHIA, set from the start by common.ChapterEight; FLAG_SAILED_TO_BATTLE_ZONE once she
# has).
#
# The temple opens for a Champion with the National Pokédex (plan 02 · S15, part b; the original asks for both, the
# Pokédex first). The guard begins as she always has, and then Candice comes up the street from her Gym
# (FLAG_HIDE_SNOWPOINT_CITY_CANDICE, set from the start by common.ChapterEight) and tells her to let this one in:
# Cynthia has told her of the player's Pokédex. VAR_SNOWPOINT_CITY_STATE is 1 from then on: the trigger before the
# door is dead, the guard tells of the temple's age, the man who longs to see inside has another thing to say, and
# Candice is gone the next time the town is come to. The temple's west door is the one walked in by: the guard stands
# before the east one and never leaves it, in the original either.


# SnowpointCity_OnTransition: Candice has gone back to her Gym once the temple is open
script OnEnter
  if var VAR_SNOWPOINT_CITY_STATE < 1 end
  setflag FLAG_HIDE_SNOWPOINT_CITY_CANDICE
  hide candice
  end


# The temple's guard turns the player back, or is told to let them by (the original's coordinate event, script 3).
script TempleGuard
  face temple_guard left
  speaker "Temple Guard"
  say "Only those the temple has chosen may set foot in Snowpoint Temple."
  say "I'm sorry, but I must ask you to leave."
  # SnowpointCity_CheckAllowEnterTemple
  if not nationaldex goto Back
  if not flag FLAG_GAME_COMPLETED goto Back
  call CandiceLetsIn
  end
label Back
  face temple_guard down
  walk player down 1
  end


# The guard, spoken to (the original's script 2): from the south, the one side she can be spoken to from that is no
# trigger
script Guard
  face self player
  speaker "Temple Guard"
  if not nationaldex goto No
  if not flag FLAG_GAME_COMPLETED goto No
  if var VAR_SNOWPOINT_CITY_STATE == 0 goto Chosen
  say "They say the temple was raised in an age nobody remembers, in honour of the Pokémon of rock, of ice and of steel that gave the land its shape."
  say "So the story goes. I have guarded its door for years, and I have never once been inside."
  end
label Chosen
  say "Only those the temple has chosen may set foot in Snowpoint Temple."
  say "I'm sorry, but I must ask you to leave."
  call CandiceLetsIn
  end
label No
  sayown
  end


# SnowpointCity_AllowEnterTemple: Candice comes up the street from her Gym, to the foot of the temple's steps (a tile
# nearer than the original's seven steps bring her, where the text box would cover her as she speaks)
script CandiceLetsIn
  wait 1.0
  clearflag FLAG_HIDE_SNOWPOINT_CITY_CANDICE
  place candice 366 208 up
  show candice
  walk candice up 8
  speaker "Candice"
  say "It's all right! You can let this one in!"
  face player down
  face temple_guard down
  setvar VAR_SNOWPOINT_CITY_STATE 1
  say "Cynthia's told me all about you, {player}. You're filling in a Pokédex, aren't you?"
  say "There are Pokémon in the temple you won't find anywhere else. Have fun exploring! And wrap up warm: it's colder in there than out here."
  return


# SnowpointCity_Candice
script Candice
  face self player
  speaker "Candice"
  say "There are Pokémon in the temple you won't find anywhere else. Have fun exploring!"
  end


# SnowpointCity_SnowpointNPCM: the man who longs to see inside
script Man
  face self player
  if var VAR_SNOWPOINT_CITY_STATE >= 1 goto Dragged
  sayown
  end
label Dragged
  say "They say Pokémon dragged the whole of Snowpoint City up here, long ago, and that the temple is where they lay down to rest afterwards."
  say "You've been let in? Then tell me what it's like in there! ...No, don't. I want to see it for myself one day."
  end


# The sailor of the ship to the Battle Zone (the original's script 12, SnowpointCity_SailorSSSpiral)
script Ship
  face self player
  if not flag FLAG_GAME_COMPLETED goto NotYet
  ask "Ahoy, Champion! This ship is bound for the Battle Zone. Shall we cast off?"
  if no goto Stay
  if not flag FLAG_SAILED_TO_BATTLE_ZONE call CynthiaSeesOff
  speaker self
  say "All aboard, then! Next stop, the Fight Area!"
  sound "stairs"
  warp "Sinnoh" 623 434 right
  end
label NotYet
  sayown
  end
label Stay
  say "Suit yourself. But mind: there's no surfing across that sea. The only way to the Battle Zone is by this ship."
  end


# SnowpointCity_Cynthia: she comes down from the town to the head of the gangway, behind the player (a tile short of
# the original's place, which is the tile at the player's back, where the player's own head would hide her)
script CynthiaSeesOff
  clearflag FLAG_HIDE_SNOWPOINT_CITY_CYNTHIA
  place cynthia 366 241 left
  show cynthia
  walk cynthia left 4 down 2 left 6 down 1
  speaker "Cynthia"
  say "{player}! I'm so glad I waited. I had a feeling you would come this way."
  emote player exclaim
  face player up
  walk cynthia right 1
  wait 0.25
  say "Do you know, I think it is a fine thing that our world keeps changing."
  say "New places, new Pokémon, new people to meet. However far you go, there is always more of it."
  walk cynthia left 1
  face cynthia down
  say "So go and see the far-off places, {player}. Meet everything they have to show you."
  say "And when you come back, tell me all about it. Safe travels, Champion."
  walk cynthia up 1 right 3 up 2 right 7
  hide cynthia
  setflag FLAG_HIDE_SNOWPOINT_CITY_CYNTHIA
  face player down
  setflag FLAG_SAILED_TO_BATTLE_ZONE
  return
