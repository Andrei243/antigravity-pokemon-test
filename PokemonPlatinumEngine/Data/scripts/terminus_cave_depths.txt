# Terminus Cave's depths (plan 23 · Z4b). The lines are our own, on X and Y's beats as they are publicly described
# (docs/mechanics/rulings.md, "Kalos after its Hall of Fame"): in the last basement a Pokémon that watches over the land
# lies sleeping, Zygarde, at level 70 (met at the level of a region crossed into, as every script's wild battle is). Beaten
# or caught it is gone for good (FLAG_KALOS_ZYGARDE_GONE); run from, it stays.


script Zygarde
  speaker none
  scene "Zygarde in the depths"
  letterbox on 0.4
  camera pan 31 8 1.2
  camera zoom 1.4 0.8
  effect shadow 31 8 hold
  text "A great serpent of black and green lies coiled in the dark, its eyes half closed..."
  cry "Zygarde"
  camera shake 0.5
  text "It raised its head! The whole cave trembled!"
  effect stop 31 8
  camera release 0.4
  letterbox off 0.3
  wildbattle "Zygarde" 70
  if result == 2 goto Stays
  setflag FLAG_KALOS_ZYGARDE_GONE
  hide self
  if result == 3 goto Caught
  text "Zygarde sank back into the rock, and the depths were still again."
  goto Done
label Caught
  text "The depths grew still. Whatever Zygarde watched over, it watches beside {player} now."
  goto Done
label Stays
  text "Zygarde settled its coils again, still watching."
label Done
  endscene
  end
