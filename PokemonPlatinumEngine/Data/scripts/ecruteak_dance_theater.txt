# Ecruteak City's Dance Theater (plan 20 · J7). docs/scripts.md describes the language; the lines are our own, on
# HeartGold's beats (scr_seq_T27R0501).
#
# VAR_UNK_410C: 0 before the player has come in, 1 while the grunt bothers the Kimono Girl, 2 once he is beaten, 3 once
# the gentleman has given HM03, Surf (FLAG_GOT_HM03), 4 once the five dancers are on the stage (ecruteak.txt). The
# red-haired boy's scene outside waits for both the Fog Badge and Surf (VAR_UNK_4079 3: ecruteak_gym.txt, and here).


# T27R0501_013 and _015: the grunt on the stage, the first time the player comes in
script OnEnter
  if var VAR_UNK_410C != 0 end
  scene "Team Rocket at the Dance Theater"
  face t27r0501_rocketm t27r0501_dancer_3
  speaker "Grunt"
  say "Come on, give us a dance! What, you won't dance for Team Rocket? Then I'll take that Eevee off your hands instead!"
  speaker "Kimono Girl"
  say "Please, stop it! Somebody, help!"
  setvar VAR_UNK_410C 1
  endscene
  end


# T27R0501_003: the grunt
script Grunt
  face t27r0501_rocketm player
  speaker "Grunt"
  if var VAR_UNK_410C != 1 end
  scene "The grunt at the Dance Theater"
  music "johto/eye_mystery"
  say "Who asked you? Team Rocket goes where it likes and takes what it wants. Get lost, or get hurt!"
  battle self as "johto_team_rocket_grunt_26"
  say "Tch! Whatever. Dancing's boring anyway. I'm out of here!"
  fade out 0.4
  hide t27r0501_rocketm
  fade in
  setflag FLAG_UNK_23A
  setvar VAR_UNK_410C 2
  music area
  endscene
  end


# T27R0501_004: the girl Team Rocket bothered
script Bothered
  face t27r0501_dancer_3 player
  speaker "Kimono Girl"
  if var VAR_UNK_410C == 1 goto Help
  say "Thank you for saving me. My Eevee and I would have been so frightened without you."
  end
label Help
  say "That man won't leave me alone. Please, can't somebody make him go away?"
  end


# T27R0501_011 and _012: the gentleman comes over once the grunt has gone, and gives HM03
script Surf
  if var VAR_UNK_410C != 2 end
  scene "The gentleman's thanks"
  face t27r0501_gsgentleman player
  call GiveSurf
  endscene
  end


# The five Kimono Girls on the stage (plan 20 · J11, part d; HeartGold's battles at the theater once the bell's holder
# comes). Before the Rainbow Wing (FLAG_UNK_093, goldenrod_radio_tower_observation_deck.txt) each says her own line.
# With it, whichever is spoken to calls the others, and the five battle one after another with no heal between them:
# Zuki, Naoko, Miki, Sayo and Kuni. A battle lost ends the test there, and spoken to again they begin again from the
# first. All five beaten, they go ahead to the Bell Tower (FLAG_JOHTO_KIMONO_GIRLS_BEATEN, ours: the gate's door to the
# tower opens and the five stand on its roof; VAR_UNK_40FA 1, the roof's trigger) and leave the stage
# (FLAG_HIDE_DANCE_STUDIO_KIMONO_GIRLS set again).
script Dancers
  if not flag FLAG_UNK_093 goto OwnLine
  if flag FLAG_JOHTO_KIMONO_GIRLS_BEATEN goto OwnLine
  scene "The Kimono Girls' test"
  face self player
  speaker "Kimono Girl"
  say "That feather... a Rainbow Wing! Then you may be the one the bird of the Bell Tower is waiting for."
  say "We are the tower's dancers. Before we let anyone climb it, the five of us test them, one after another."
  say "There will be no rest between us. Are you ready?"
  speaker "Zuki"
  say "I'm first. Don't hold back!"
  battle t27r0501_dancer as "johto_kimono_girl_zuki" canlose
  if lost goto Lost
  speaker "Naoko"
  say "Zuki's fallen? Then it's my turn to dance."
  battle t27r0501_dancer_2 as "johto_kimono_girl_naoko" canlose
  if lost goto Lost
  speaker "Miki"
  say "You helped me once. Now let me see how strong you really are."
  battle t27r0501_dancer_6 as "johto_kimono_girl_miki" canlose
  if lost goto Lost
  speaker "Sayo"
  say "Three of us, and you're still standing. Then try me."
  battle t27r0501_dancer_4 as "johto_kimono_girl_sayo" canlose
  if lost goto Lost
  speaker "Kuni"
  say "I'm the last. If you pass me, the tower is yours to climb."
  battle t27r0501_dancer_5 as "johto_kimono_girl_kuni" canlose
  if lost goto Lost
  speaker "Kimono Girl"
  say "Five battles, and not one rest. You and your Pokémon are one heart. We are sure of it now."
  say "We will go ahead to the Bell Tower and tell the Sages at its gate to let you pass. Meet us at the very top."
  fade out 0.6
  setflag FLAG_JOHTO_KIMONO_GIRLS_BEATEN
  setflag FLAG_HIDE_DANCE_STUDIO_KIMONO_GIRLS
  setvar VAR_UNK_40FA 1
  hide t27r0501_dancer
  hide t27r0501_dancer_2
  hide t27r0501_dancer_4
  hide t27r0501_dancer_5
  hide t27r0501_dancer_6
  fade in 0.6
  endscene
  end
label Lost
  speaker "Kimono Girl"
  say "Not yet, it seems. Rest your Pokémon, and come back to us. We'll test you again from the beginning."
  endscene
  end
label OwnLine
  face self player
  sayown
  end


# T27R0501_000: the gentleman of the audience
script Gentleman
  face t27r0501_gsgentleman player
  speaker "Gentleman"
  if var VAR_UNK_410C == 2 goto Thanks
  if flag FLAG_GOT_HM03 goto Given
  say "I never miss a performance. My Psyduck enjoys it as much as I do."
  end
label Thanks
  scene "The gentleman asked"
  call GiveSurf
  endscene
  end
label Given
  say "Surf lets a Pokémon carry you over the water. With the Fog Badge, you'll be able to use it outside of battle."
  end

script GiveSurf
  speaker "Gentleman"
  say "That was splendid! You stood up to that ruffian when the rest of us only watched. I'm ashamed of myself."
  say "Please, accept this. It's a technique I treasure, but you'll put it to far better use."
  give "HM03"
  setflag FLAG_GOT_HM03
  setvar VAR_UNK_410C 3
  setvar VAR_UNK_4090 1
  if badge fog setvar VAR_UNK_4079 3
  say "That's Surf. A Pokémon that knows it can carry you over the water once you have Morty's Fog Badge."
  return
