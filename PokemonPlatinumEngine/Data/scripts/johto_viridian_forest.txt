# The place's arrival lays who of the Gym Leaders is here today (Johto's leftovers)
script OnEnter
  # Bugsy's street meeting (Johto's leftovers): Fridays, once Bugsy's number is in the Pokégear
  setflag FLAG_UNK_263
  if weekday friday call ShowMeetBugsy
  end


script ShowMeetBugsy
  if flag FLAG_PHONE_CONTACT_BUGSY clearflag FLAG_UNK_263
  return

# Bugsy, out on Friday (the Leaders' street meetings, Johto's leftovers)
script MeetBugsy
  face d46r0101_gsleader2 player
  speaker "Bugsy"
  random 2
  if result == 1 goto Two
  say "Oh, hi! I come to Viridian Forest on Fridays. The bug Pokémon here are so different from Ilex Forest's!"
  end
label Two
  say "I've filled half a notebook already. Research never ends!"
  end
