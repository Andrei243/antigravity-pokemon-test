# The Battle Maison's lobby in Kiloude City (plan 23, Kalos's last places; plan 37 · FC2's streak house). The lines are
# our own, and so are the people: none of X and Y's hosts is here. Four rooms, the original's counts: single (three
# entered), double (four), triple (six) and rotation (four), each an endless streak a battle at a time against trainers
# of the later sets (Data/facility-sets.json) who grow stronger with the streak, the entered going in as copies at level
# 50, whole again after every battle (Models/Facilities/StreakHouse.cs, StreakChallenge.cs; `facility house ...`). A
# room's host stands in the way at its 20th battle and, beaten, opens its grand room, whose host comes at the 50th. After
# any battle the player goes on, rests (the challenge put down and saved, taken up again here) or retires. The battles
# are KiloudeCity_BattleMaisonHall.txt's arrival script. Battle Points go into the Frontier's one purse, and the counter
# beside the desk trades them for the later games' items (Data/facility-prizes.json).


script Desk
  face self player
  speaker "Receptionist"
  say "Welcome to the Battle Maison! Trainers from every corner of the world come to Kiloude to test themselves here."
  facility house paused
  if result == 1 goto Resume
label Ask
  choose "Would you like to take on the Battle Maison?" "Enter" "The rules" "The records" "No"
  if result == 1 goto Rules
  if result == 2 goto Records
  if result == 3 goto Later
  choose "Which room will you battle in?" "Single" "Double" "Triple" "Rotation" "Cancel"
  if result == 0 goto Single
  if result == 1 goto Double
  if result == 2 goto Triple
  if result == 3 goto Rotation
  goto Later
label Single
  facility house way single
  goto Grand
label Double
  facility house way double
  goto Grand
label Triple
  facility house way triple
  goto Grand
label Rotation
  facility house way rotation
label Grand
  facility house hardopen
  if result == 0 goto Streak
  choose "You've beaten this room's host, so its grand room is open to you. Which will it be?" "The usual room" "The grand room"
  if result == 0 goto Streak
  facility house hard hard
label Streak
  facility house streak
  if result == 0 goto Enough
  say "You've won {number} battles in a row in that room. The regulars have started to talk about you!"
label Enough
  facility house enough
  if result == 1 goto Choose
  say "You need {number} Pokémon that can take part, all of different species and holding different items."
  say "An Egg can't take part, and nor can {banned}."
  end
label Choose
  say "Please choose the {number} Pokémon you'll enter."
  facility house choose
  if result == 0 goto Later
  facility house check
  if result == 1 goto SameSpecies
  if result == 2 goto SameItem
  facility house begin
  say "Thank you! Your first challenger is waiting in the hall. This way, please!"
  fade out 0.6
  warp "KiloudeCity_BattleMaisonHall" 8 10 up
  fade in
  end
label Resume
  say "Welcome back! The challenge you put down is waiting for you, just as you left it."
  facility house resume
  if result == 0 goto Changed
  say "Your streak of {number} still stands. This way, please!"
  fade out 0.6
  warp "KiloudeCity_BattleMaisonHall" 8 10 up
  fade in
  end
label Changed
  say "Oh... the Pokémon you entered aren't with you as they were. I'm sorry, but that challenge is over."
  goto Ask
label SameSpecies
  say "I'm sorry. Those entered must all be of different species."
  goto Later
label SameItem
  say "I'm sorry. No two of those entered may hold the same item."
label Later
  say "We'll be here. The hall never really empties."
  end
label Records
  facility house board
  say "Every streak is written down. Somebody has to keep count!"
  end
label Rules
  say "Your Pokémon battle at Level 50, or at their own level if it's lower. They are made whole after every battle."
  say "Three go into the single room, four into the double and the rotation rooms, and six into the triple room."
  say "Win, and you may go on, rest and come back later, or retire. Lose once, and your streak is over."
  say "The longer your streak, the stronger the Trainers. Reach the twentieth battle and the room's host will meet you."
  end


# The counter of prizes: the later games' items for Battle Points, the Frontier's one purse
script Prizes
  face self player
  speaker "Clerk"
  say "Welcome! Battle Points won here, or anywhere in the world's battle halls, are good at this counter."
  frontier prize maison
  if result == 0 goto Bye
  frontier buy
  if result == 0 goto Short
  sound "select"
  text "{player} traded {price} BP for the {item}."
  say "Thank you! Use it well."
  end
label Short
  say "I'm sorry. You don't have the Battle Points for that."
  end
label Bye
  say "Come back when you've won a few more!"
  end


script Fan
  face self player
  speaker "Fan"
  say "I come every afternoon just to watch. You can tell a Trainer's whole journey from how they send out their first Pokémon."
  end

script Veteran
  face self player
  speaker "Veteran"
  say "Twenty in a row sounds easy until the fifteenth. Then you start to hear your own heartbeat."
  end
