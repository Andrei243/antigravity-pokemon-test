# The Hall of Fame (the room: plan 01 · M9, part 2d; its scene: plan 02 · S14), on the original's plan.
# docs/scripts.md describes the language; the lines are our own, on the original's beats
# (scripts_pokemon_league_hall_of_fame.s, and src/clear_game.c for what follows the machine).
#
# One scene, played as the new Champion comes in. Professor Rowan and Cynthia stand either side of the carpet; they
# have other words for a Champion who has been here before (the original's GetLeagueVictories, which is nought exactly
# until FLAG_GAME_COMPLETED is first set). Cynthia welcomes the player to the machine and says what it keeps. Then, in
# the original's order:
#   - FLAG_UNLOCKED_VS_SEEKER_LVL_4, and the Spring Path on the map for whoever has the National Pokédex;
#   - a Poké Ball for each of the team laid on the machine, and all of them lit to its chime
#     (PlayHallOfFameHealingAnimation: anim halloffame, the machine's own set piece of plan 09 · L12, whose balls the
#     room's pieces draw, GymPieces);
#   - the screen goes dark, and Victory sets what a victory sets (SetHallOfFameVictoryFlagsAndVars);
#   - ClearGame: the Champion Ribbons and the record (halloffame), the game completed and the region done, the scene
#     of the team's entry (halloffame show: each Pokémon in turn, then the Champion among them), the team healed and
#     the game saved in the player's room, where it begins again, and the credits.
# The original then resets the console to its title screen, and CONTINUE wakes the player in their room; here the
# game goes straight on in the room, the save having just been written there (docs/mechanics/rulings.md, "The
# eleventh chapter").
#
# Not here, and why: Terastallization at the League's door (decision 06 · 2; plan 06 · R23). The Four's and
# Cynthia's rematch teams are their rooms' own (plan 02 · S15, part b, by FLAG_ARRESTED_CHARON_STARK_MOUNTAIN).


# PokemonLeagueHallOfFame_OnFrame_EnterHallOfFame
script OnEnter
  # The room's opening words may be skipped (plan 12 · Q20) and nothing after them: the machine, the team's entry,
  # the save and the credits are the player's to see every time
  scene "The Hall of Fame"
  walk player up 8
  # PokemonLeague_HallOfFame_Movement_TurnAroundInPlace: the professor looks the room over
  face prof_rowan up
  wait 0.5
  face prof_rowan right
  if flag FLAG_GAME_COMPLETED goto Again
  # PokemonLeagueHallOfFame_FirstVictoryMessages
  speaker "Professor Rowan"
  say "Hm... It has been a very long time since I last set foot in this room."
  speaker "Cynthia"
  say "If the last time was the day I became Champion, Professor, then it has been a long time indeed."
  goto Welcome
label Again
  # PokemonLeagueHallOfFame_RepeatVictoryMessages
  speaker "Professor Rowan"
  say "However often I come in here, I catch myself standing a little straighter than I do anywhere else."
  speaker "Cynthia"
  say "So does everyone, Professor. Everything Pokémon and their Trainers have hoped for and fought for is kept in this room. It asks no less of whoever walks into it."
label Welcome
  face prof_rowan down
  face cynthia down
  wait 0.25
  speaker "Cynthia"
  say "{player}. Welcome to the Hall of Fame."
  walk player up 2
  wait 0.15
  face prof_rowan right
  face cynthia left
  face prof_rowan up
  face cynthia up
  say "Here your names are kept for good: your own, and those of the Pokémon who came all this way beside you."
  say "What you leave in this room is the memory of your journey so far."
  say "Think of them as you do it. They are your partners. They grew as you grew, through one hard battle after another."
  say "Now, {player}. This machine will make a lasting record of what you have done together."
  endscene

  setflag FLAG_UNLOCKED_VS_SEEKER_LVL_4
  # PokemonLeagueHallOfFame_EnableHiddenLocationSpringPath (the Distortion World's way out shows it already, plan 02 · S12)
  if nationaldex setvar VAR_HIDDEN_LOCATION_SPRING_PATH 786

  # PlayHallOfFameHealingAnimation: the machine's own set piece (plan 09 · L12). A ball for each of the team is set in
  # its sockets, one after another, and all of them are lit to the machine's chime; the script waits for it
  wait 0.4
  anim halloffame
  fade out 0.8
  call Victory

  # ClearGame (src/clear_game.c)
  halloffame
  setflag FLAG_GAME_COMPLETED
  setflag SinnohHallOfFame
  halloffame show
  heal
  # SetPlayerStartLocation: the save is written in the player's own room, where the game begins again
  warp "PlayerHouse2F" 4 6 up
  save
  if result == 0 goto NotSaved
  sound "save"
  text "{player} saved the game."
  goto Roll
label NotSaved
  text "The game could not be saved."
label Roll
  credits
  end


# PokemonLeagueHallOfFame_SetHallOfFameVictoryFlagsAndVars: what every victory sets, each under the original's own
# name. It sets flags and variables and nothing else, so a save that entered the Hall of Fame before this scene was
# written is given it with no screen (Story/StoryMigration.cs, version 17).
#
# The legendary Pokémon are put back where they weren't caught: Heatran in Stark Mountain's heart and Regigigas in
# Snowpoint Temple (their own scripts, plan 02 · S15, part b, hide each after a battle that didn't catch it), the
# lakes' guardians and Giratina, and Cresselia on Fullmoon Island once a roaming one has been knocked out. Dialga and
# Palkia are behind their rifts at Spear Pillar again unless caught (spear_pillar_dialga.txt, spear_pillar_palkia.txt),
# and Heatran only once Buck has been spoken to in the Battleground (Battleground.txt: plan 02 · S15, part c, which
# wrote the last of the scenes this script had asked about ahead of them).
script Victory
  clearflag FLAG_HIDE_DAY_CARE_GYM_GUIDE
  clearflag FLAG_ALT_MUSIC_CHAMPION_ROOM
  # PokemonLeagueHallOfFame_ResetSpearPillarDialgaState and ...PalkiaState
  if not flag FLAG_CAUGHT_DIALGA setvar VAR_SPEAR_PILLAR_DIALGA_STATE 0
  if not flag FLAG_CAUGHT_PALKIA setvar VAR_SPEAR_PILLAR_PALKIA_STATE 0
  # PokemonLeagueHallOfFame_TryShowStarkMountainHeatran: only once Buck has been spoken to at the Battleground
  if flag FLAG_CAUGHT_HEATRAN goto Regigigas
  if not flag FLAG_TALKED_TO_BATTLEGROUND_BUCK goto Regigigas
  clearflag FLAG_HIDE_STARK_MOUNTAIN_ROOM_3_HEATRAN
  setvar VAR_STARK_MOUNTAIN_ROOM_3_STATE 1
label Regigigas
  # PokemonLeagueHallOfFame_TryShowSnowpointTempleRegigigas: only with the National Pokédex
  if flag FLAG_CAUGHT_REGIGIGAS goto Guardians
  if not nationaldex goto Guardians
  clearflag FLAG_HIDE_SNOWPOINT_TEMPLE_B5F_REGIGIGAS
label Guardians
  # The lakes' guardians (plan 02 · S13): Azelf and Uxie are back in their caverns unless they were caught, and
  # Giratina waits in Turnback Cave unless it was (plan 02 · S12)
  if not flag FLAG_CAUGHT_AZELF clearflag FLAG_HIDE_VALOR_CAVERN_AZELF
  if not flag FLAG_CAUGHT_UXIE clearflag FLAG_HIDE_ACUITY_CAVERN_UXIE
  if not flag FLAG_CAUGHT_GIRATINA clearflag FLAG_HIDE_TURNBACK_CAVE_GIRATINA_ROOM_GIRATINA
  # The roamers: one that was knocked out (ROAMER_STATE_DEFEATED, 2) can be set loose again (ROAMER_STATE_RESET, 3).
  # Mesprit is back in its cavern for that, which Verity Cavern's script reads, and Cresselia on Fullmoon Island
  if var VAR_ROAMING_MESPRIT_STATE == 2 clearflag FLAG_HIDE_VERITY_CAVERN_MESPRIT
  if var VAR_ROAMING_MESPRIT_STATE == 2 setvar VAR_ROAMING_MESPRIT_STATE 3
  if var VAR_ROAMING_CRESSELIA_STATE == 2 clearflag FLAG_HIDE_FULLMOON_ISLAND_FOREST_CRESSELIA
  if var VAR_ROAMING_CRESSELIA_STATE == 2 setvar VAR_ROAMING_CRESSELIA_STATE 3
  if var VAR_ROAMING_MOLTRES_STATE == 2 setvar VAR_ROAMING_MOLTRES_STATE 3
  if var VAR_ROAMING_ZAPDOS_STATE == 2 setvar VAR_ROAMING_ZAPDOS_STATE 3
  if var VAR_ROAMING_ARTICUNO_STATE == 2 setvar VAR_ROAMING_ARTICUNO_STATE 3
  # PokemonLeagueHallOfFame_IncreasePlayerHousePostgameState: Mom has news, the first time (PlayerHouse.txt)
  if var VAR_PLAYER_HOUSE_POSTGAME_STATE == 0 setvar VAR_PLAYER_HOUSE_POSTGAME_STATE 1
  # Who the victory brings on: the assistant in Sandgem Town, the professor in the Veilstone Department Store's
  # basement until he has been spoken to there, and Lucian among the Canalave Library's shelves
  clearflag FLAG_HIDE_SANDGEM_TOWN_COUNTERPART
  if not flag FLAG_TALKED_TO_VEILSTONE_STORE_B1F_PROF_ROWAN clearflag FLAG_HIDE_VEILSTONE_STORE_B1F_PROF_ROWAN
  clearflag FLAG_HIDE_CANALAVE_LIBRARY_1F_LUCIAN
  # ...and who it sends away: Cynthia is gone from Sendoff Spring, where she stood before Turnback Cave's mouth (plan
  # 02 · S12: the cave is come to from now on), and will speak again before the painting in Celestic Town's ruins
  clearflag FLAG_TALKED_TO_CELESTIC_TOWN_CAVE_CYNTHIA
  setflag FLAG_HIDE_SENDOFF_SPRING_CYNTHIA
  return
