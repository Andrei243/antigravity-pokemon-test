# Cinnabar Island (plan 19 · K6, part b; Bill and the ferry, plan 19 · K7). docs/scripts.md describes the language; the
# lines are our own, on FireRed's beats (data/maps/CinnabarIsland/scripts.inc).
#
# The Gym's door is locked until the Secret Key has been picked up in the Pokémon Mansion's basement: FireRed asks its ball's flag
# (FLAG_HIDE_POKEMON_MANSION_B1F_SECRET_KEY), and here the bag, which a key item never leaves; the step before the door turns the
# player back while VAR_TEMP_1 is 0.
#
# Once Blaine is beaten (VAR_MAP_SCENE_CINNABAR_ISLAND 1, as FireRed sets it) Bill is waiting outside: he asks the player
# to sail with him to One Island, where his friend Celio needs a hand. Yes, and his friend's ferry, the Seagallop, takes
# them both there (VAR_MAP_SCENE_CINNABAR_ISLAND 2, VAR_MAP_SCENE_ONE_ISLAND_HARBOR 1: one_island.txt plays the rest);
# no, and he waits in the Pokémon Center (FLAG_HIDE_CINNABAR_POKECENTER_BILL cleared, the same 2:
# cinnabar_island_pokemon_center_1f.txt). FireRed draws the Seagallop coming in to the dock; here it comes and goes while
# the screen is dark, and the ship itself stays out of sight (FLAG_HIDE_CINNABAR_SEAGALLOP).


script OnEnter
  if var VAR_MAP_SCENE_CINNABAR_ISLAND == 0 setflag FLAG_HIDE_CINNABAR_BILL
  setflag FLAG_HIDE_CINNABAR_SEAGALLOP
  if item "Secret Key" setvar VAR_TEMP_1 1
  if var VAR_MAP_SCENE_CINNABAR_ISLAND == 1 goto Bill
  if var VAR_MAP_SCENE_CINNABAR_ISLAND == 3 goto Home
  end
label Bill
  call BillScene
  end
label Home
  call BillHome
  end

script LockedGymDoor
  face player up
  speaker none
  text "The door is locked. There's a keyhole, but no key in sight."
  walk player down 1
  end

script GymSign
  speaker none
  if badge volcano goto Won
  text "Cinnabar Island Pokémon Gym\nLeader: Blaine\nRiddles and fire for every challenger."
  end
label Won
  text "Cinnabar Island Pokémon Gym\nLeader: Blaine. Winning trainers: {rival}, {player}."
  end


# CinnabarIsland_EventScript_BillScene: Bill by the dock, once Blaine is beaten.
script BillScene
  scene "Bill on Cinnabar Island"
  clearflag FLAG_HIDE_CINNABAR_BILL
  show cinnabar_bill
  emote cinnabar_bill exclaim
  face cinnabar_bill player
  speaker "Bill"
  say "Hey! If it isn't {player}! Fancy meeting you way out here."
  say "You beat Blaine? Then you've earned a break. I'm off to One Island, out past the sea, to help a friend of mine called Celio."
  say "He's a whiz with machines, and he's building something to link PCs between the islands. How about coming along?"
  ask "Will you come to One Island with me?"
  if no goto Wait
  say "Great! My pal's ferry should be pulling in any minute."
  call SailWithBill
  endscene
  end
label Wait
  say "No? Well, I'll wait for you in the Pokémon Center. Come and find me when you're ready to sail."
  fade out 0.3
  hide cinnabar_bill
  fade in 0.3
  setflag FLAG_HIDE_CINNABAR_BILL
  clearflag FLAG_HIDE_CINNABAR_POKECENTER_BILL
  setvar VAR_MAP_SCENE_CINNABAR_ISLAND 2
  endscene
  end


# CinnabarIsland_EventScript_BillReturnScene (plan 19 · K9, part a): back from One Island once the network works
# (one_island_pokemon_center_1f.txt sets the variable to 3), Bill says goodbye on the dock and goes home. Lostelle,
# rescued, is home on Three Island from now on (4)
script BillHome
  scene "Bill back in Kanto"
  clearflag FLAG_HIDE_CINNABAR_BILL
  show cinnabar_bill
  face cinnabar_bill player
  speaker "Bill"
  say "Phew, back on Cinnabar! Thanks for coming all that way with me, {player}."
  say "Keep the Tri-Pass. Celio said he'd love to see you again, so drop in on him when you're on One Island."
  say "Me, I'm off home to my cottage. My PCs must be missing me! See you around!"
  fade out 0.3
  hide cinnabar_bill
  fade in 0.3
  setflag FLAG_HIDE_CINNABAR_BILL
  setflag FLAG_HIDE_TWO_ISLAND_GAME_CORNER_LOSTELLE
  clearflag FLAG_HIDE_LOSTELLE_IN_HER_HOME
  setvar VAR_MAP_SCENE_CINNABAR_ISLAND 4
  endscene
  end
