# Selphy's house at Resort Gorgeous (plan 19 · K9, parts b and c). docs/scripts.md describes the language; the lines
# are our own, on FireRed's beats (data/maps/FiveIsland_ResortGorgeous_House/scripts.inc and its
# SampleResortGorgeousMonAndReward in src/field_specials.c).
#
# Selphy is at home once she has been brought back from the Lost Cave (five_island_resort_gorgeous.txt clears FLAG_
# HIDE_RESORT_GORGEOUS_INSIDE_SELPHY); before that her butler waits for her alone.
#
# Her wish is a species the Pokédex has seen, drawn on the game's own chance (random seen), kept as its National
# Pokédex number in VAR_RESORT_GORGEOUS_REQUESTED_MON; bring one to her door and the butler hands over the gift drawn
# with it (VAR_RESORT_GORGEOUS_REWARD: 1 a Luxury Ball, which seven draws in ten give, and 2 to 7 one of FireRed's six
# deluxe rewards), after which she wishes for another the next time the house is come into (FLAG_MAP_LOCAL_SELPHY_
# THANKED, FireRed's FLAG_TEMP_2). FireRed forgets a wish 250 steps after it is made; here a wish stands until it is
# granted (docs/mechanics/rulings.md, "The last Sevii Islands").


script OnEnter
  if var VAR_MAP_SCENE_FIVE_ISLAND_RESORT_GORGEOUS < 2 setflag FLAG_HIDE_RESORT_GORGEOUS_INSIDE_SELPHY
  end


script Selphy
  face resort_gorgeous_house_selphy player
  speaker "Selphy"
  if flag FLAG_MAP_LOCAL_SELPHY_THANKED goto Luck
  if var VAR_RESORT_GORGEOUS_REQUESTED_MON == 0 goto Wish
  if has var VAR_RESORT_GORGEOUS_REQUESTED_MON goto Shown
  goto Tell
# FiveIsland_ResortGorgeous_House_EventScript_RequestMon
label Wish
  say "My hero! My butler has been fussing over me ever since I came home." "Oh, but while you are here, would you hear one little wish of mine?"
  random seen
  setvar VAR_RESORT_GORGEOUS_REQUESTED_MON RESULT
  if result == 0 goto NoWish
  random 100
  if result < 30 goto Deluxe
  setvar VAR_RESORT_GORGEOUS_REWARD 1
  goto Tell
# SampleResortGorgeousReward: thirty draws in a hundred give one of the six instead of a Luxury Ball
label Deluxe
  random 6
  if result == 0 setvar VAR_RESORT_GORGEOUS_REWARD 2
  if result == 1 setvar VAR_RESORT_GORGEOUS_REWARD 3
  if result == 2 setvar VAR_RESORT_GORGEOUS_REWARD 4
  if result == 3 setvar VAR_RESORT_GORGEOUS_REWARD 5
  if result == 4 setvar VAR_RESORT_GORGEOUS_REWARD 6
  if result == 5 setvar VAR_RESORT_GORGEOUS_REWARD 7
  goto Tell
# FiveIsland_ResortGorgeous_House_Text_WishToSeeMonPlease
label Tell
  say "I do so long to see a {species:VAR_RESORT_GORGEOUS_REQUESTED_MON} with my own eyes." "Would you be a dear and bring one to my door?"
  end
label NoWish
  say "I should dearly love to see some rare Pokémon, but I know of none at all." "Travel a little and tell me what you find!"
  end
# FiveIsland_ResortGorgeous_House_EventScript_ShowRequestedMon
label Shown
  say "Oh! Oh, is that a {species:VAR_RESORT_GORGEOUS_REQUESTED_MON} with you?" "You brought one all this way, for me? You darling creature!"
  clearflag FLAG_HIDE_SELPHYS_BUTLER
  show resort_gorgeous_house_butler
  wait 0.5
  speaker "Butler"
  say "You called, my lady?"
  speaker "Selphy"
  say "A token of my appreciation for my hero, if you please. The good one."
  speaker "Butler"
  say "I shall do as you bid, my lady." "Her ladyship asks that you accept this with her thanks."
  if var VAR_RESORT_GORGEOUS_REWARD == 2 give "Big Pearl"
  if var VAR_RESORT_GORGEOUS_REWARD == 3 give "Pearl"
  if var VAR_RESORT_GORGEOUS_REWARD == 4 give "Stardust"
  if var VAR_RESORT_GORGEOUS_REWARD == 5 give "Star Piece"
  if var VAR_RESORT_GORGEOUS_REWARD == 6 give "Nugget"
  if var VAR_RESORT_GORGEOUS_REWARD == 7 give "Rare Candy"
  if var VAR_RESORT_GORGEOUS_REWARD <= 1 give "Luxury Ball"
  setflag FLAG_MAP_LOCAL_SELPHY_THANKED
  setvar VAR_RESORT_GORGEOUS_REQUESTED_MON 0
  end
# FiveIsland_ResortGorgeous_House_Text_SelphyWishYouBestOfLuck
label Luck
  say "I shall think of something else I simply must see." "Until then, my hero, I wish you the very best of luck!"
  end
