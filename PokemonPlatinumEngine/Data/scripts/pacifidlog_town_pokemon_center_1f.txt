# The man who teaches Explosion to one Pokémon, once in a game, for nothing (`tutor once`; MoveTutors.Kanto's table
# serves every region's tutor of one move: TM64's move; FLAG_MOVE_TUTOR_TAUGHT_EXPLOSION, Emerald's name)


script ExplosionTutor
  face self player
  speaker "Man"
  if flag FLAG_MOVE_TUTOR_TAUGHT_EXPLOSION goto Taught
  ask "I know a move that goes off with an almighty bang. Shall I teach one of your Pokémon Explosion?"
  if no goto Later
  say "Splendid! Which one?"
  choosepokemon
  tutor once "Explosion"
  if result == 1 goto Later
  if result == 2 goto Egg
  if result == 3 goto Cannot
  if result == 4 goto Knows
  tutor learn
  if result == 0 goto Later
  if result == 2 goto Hm
  if result == 3 say "{wearer} forgot {forgotten}. And..."
  fanfare levelup
  say "{wearer} learned {thing}!"
  setflag FLAG_MOVE_TUTOR_TAUGHT_EXPLOSION
label Taught
  say "Explosion spends everything in one go. Save it for the moment it counts!"
  end
label Later
  say "No? A move like that is best not rushed."
  end
label Egg
  say "An Egg? It hasn't even hatched yet! Bring me a Pokémon."
  end
label Cannot
  say "Ah, that one can't learn Explosion, I'm afraid."
  end
label Knows
  say "That one already knows {thing}."
  end
label Hm
  say "{forgotten} isn't a move that can be given up like that."
  goto Later
