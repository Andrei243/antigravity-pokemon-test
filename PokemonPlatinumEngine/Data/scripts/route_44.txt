# Route 44 (Johto's leftovers: the phone's gifts). docs/scripts.md describes the language; the lines are our own, on
# HeartGold's beats.
#
# Bird Keeper Vance, beaten, gives his number; once he has rung about a present (phone.txt's Vance,
# FLAG_JOHTO_PHONE_GIFT_VANCE, ours) he hands over a Carbos here.


script Vance
  if not defeated self goto Fight
  if rematch self goto Rematch
  speaker "Vance"
  if flag FLAG_JOHTO_PHONE_GIFT_VANCE goto Gift
  if flag FLAG_PHONE_CONTACT_VANCE goto Known
  say "My birds have never been out-flown like that! Let's swap numbers. I'd like to battle you again someday."
  ask "Register Vance's number in your Pokégear?"
  if no goto Refused
  addcontact vance
  say "Good. My birds will remember you."
  end
label Refused
  say "Suit yourself. My birds and I will be watching the road."
  end
label Known
  trainerline after
  end
label Gift
  say "You came! Here, this is what I told you about. Make your Pokémon faster than my birds, if you can."
  give "Carbos"
  clearflag FLAG_JOHTO_PHONE_GIFT_VANCE
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end
