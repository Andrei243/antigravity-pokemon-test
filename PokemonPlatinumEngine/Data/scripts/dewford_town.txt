# Dewford Town (plan 21 · H3). docs/scripts.md describes the language; the lines are our own, on Emerald's beats
# (data/maps/DewfordTown/scripts.inc).
#
# Mr. Briney waits at the jetty while his boat is here (VAR_BRINEY_LOCATION 2, FLAG_HIDE_MR_BRINEY_DEWFORD_TOWN clear):
# back to his cottage on Route 104 until Steven has his letter (FLAG_DELIVERED_STEVEN_LETTER), then to Slateport's
# beach on Route 109 too. The fisherman gives the Old Rod, and the boy by the hall asks after the latest craze.


script OnEnter
  setflag FLAG_VISITED_DEWFORD_TOWN
  end


script Briney
  speaker "Mr. Briney"
  face self player
  if flag FLAG_DELIVERED_STEVEN_LETTER goto Choose
  ask "Ahoy! Shall we sail back to my cottage on Route 104?"
  if yes goto ToPetalburg
  say "Go on and deliver your letter, then. I'll wait right here."
  end
label Choose
  choose "Ahoy! Where shall we sail to?" "Route 104" "Slateport" "Stay here"
  if var RESULT == 0 goto ToPetalburg
  if var RESULT == 1 goto ToSlateport
  say "Just tell me whenever you need to set sail."
  end
label ToPetalburg
  say "Back to the cottage it is! Hold on to your hat!"
  call SailToRoute104
  end
label ToSlateport
  say "Slateport it is! Hoist the sails, Peeko!"
  call SailToSlateport
  end


# Back north across the sea to the cottage on Route 104, where Mr. Briney and Peeko are home again
script SailToRoute104
  scene "Sailing back to Route 104"
  setvar VAR_BRINEY_LOCATION 1
  setflag FLAG_HIDE_MR_BRINEY_DEWFORD_TOWN
  clearflag FLAG_HIDE_BRINEYS_HOUSE_MR_BRINEY
  clearflag FLAG_HIDE_BRINEYS_HOUSE_PEEKO
  music "hoenn/sailing"
  letterbox on
  camera path 70 368 52 352 30 330 16 290 12 250 14
  wait 0.4
  fade out
  camera release 0
  letterbox off 0.1
  warp "Route104_MrBrineysHouse" 5 4 up
  fade in
  music area
  speaker "Mr. Briney"
  say "Home sweet home! Whenever you need the sea again, you know where to find me."
  endscene
  return


# East across the sea of Routes 107 and 108 to Slateport's beach on Route 109
script SailToSlateport
  scene "Sailing to Slateport"
  setvar VAR_BRINEY_LOCATION 3
  setflag FLAG_HIDE_MR_BRINEY_DEWFORD_TOWN
  clearflag FLAG_HIDE_ROUTE_109_MR_BRINEY
  music "hoenn/sailing"
  letterbox on
  camera path 80 370 110 368 150 366 190 360 220 348 14
  wait 0.4
  fade out
  camera release 0
  letterbox off 0.1
  warp "Hoenn" 221 343 down
  fade in
  music area
  speaker "Mr. Briney"
  if flag FLAG_DELIVERED_DEVON_GOODS goto Landed
  say "Land ho! That's Slateport's beach. The city's just to the north. Off you go with your parcel!"
  say "I'll wait here. Whenever you want to go back to Dewford, just say so."
  endscene
  return
label Landed
  say "Land ho! Slateport, as promised. I'll wait here for whenever you need me."
  endscene
  return


# The fisherman who hands out the Old Rod (FLAG_RECEIVED_OLD_ROD), and asks how the fishing goes afterwards
script OldRodFisherman
  face self player
  speaker "Fisherman"
  if flag FLAG_RECEIVED_OLD_ROD goto HowsFishing
  ask "This is a fishing island, you know. Doesn't the sea make you itch to cast a line?"
  if no goto NotItching
  say "That's the spirit! Here, take one of my old rods. I've got plenty."
  give "Old Rod"
  setflag FLAG_RECEIVED_OLD_ROD
  say "Face the water and use the rod. When something bites, reel it in quick!"
  end
label NotItching
  say "No? A shame. The fish around here practically jump onto the hook."
  end
label HowsFishing
  choose "How's the fishing going?" "Excellent" "Not so good"
  if var RESULT == 0 goto Excellent
  say "Keep at it! Different rods bring in different Pokémon. Try every spot you can."
  end
label Excellent
  say "Ha ha! That's what I like to hear. There's a Pokémon out there for every fisher."
  end


# The boy who knows what's popular (Emerald's trendy phrase; ours is the word of the week, which nobody can change)
script TrendyBoy
  face self player
  speaker "Boy"
  ask "Everyone's talking about fishing right now. It's the biggest thing on the island! You knew that, right?"
  if no goto Outdated
  say "Of course you did. Anyone who's anyone knows what's in right now."
  end
label Outdated
  say "You didn't know? Come on! Dewford Hall's the place to hear what's popular. Go and listen in."
  end


# ------------------------------------------------------------------ the Salt Run (plan 41 · EG8)
# The Open Door Agency's case in Hoenn: the customs officer shows the seal with its second stamp, the sailor sings the
# crew's song of lamps, and the old fisherman knows the boat comes in when the customs hut is shut. Once Looker has the
# deductions the skipper, Ottoline Marsh, waits on Dewford's shore (FLAG_CASE_HOENN_OTTOLINE_WAITS).

script CaseHoennCustoms
  face self player
  if flag FLAG_CASE_HOENN_CLOSED goto Thanks
  if flag FLAG_CASE_HOENN_TAKEN goto Case
  say "Customs, Dewford Town. Everything that lands here gets weighed, stamped and written down."
  end
label Case
  if flag FLAG_CASE_HOENN_CLUE_SEAL goto Again
  say "The salt crates? Here's a seal off one of them. That's my stamp, the wave. But see the little star beside it?"
  say "That's not mine. Someone opened the crate after I sealed it, and sealed it again with a stamp of their own."
  setflag FLAG_CASE_HOENN_CLUE_SEAL
  text "{player} noted the customs seal with a second stamp."
  end
label Again
  say "My wave, and a star that shouldn't be there. Somebody's been at those crates."
  end
label Thanks
  say "Every crate weighed, every seal mine. Customs is dull again, and I like it that way."
  end

script CaseHoennSailor
  face self player
  if flag FLAG_CASE_HOENN_CLOSED goto Thanks
  if flag FLAG_CASE_HOENN_TAKEN goto Case
  say "Heave, and haul, and home again! Sorry, I sing when I work. And when I don't."
  end
label Case
  if flag FLAG_CASE_HOENN_CLUE_SONG goto Again
  say "The salt boat's crew? They taught me their song. Salt on the label, light in the hand, lamps for the dark of the land!"
  say "Catchy, isn't it? Though I never did work out what lamps have to do with salt."
  setflag FLAG_CASE_HOENN_CLUE_SONG
  text "{player} noted the crew's song about lamps."
  end
label Again
  say "Salt on the label, light in the hand, lamps for the dark of the land! It's stuck in my head for good."
  end
label Thanks
  say "The salt boat's crew have a new song now. It's about paying your duty. Not as catchy."
  end

script CaseHoennFisherman
  face self player
  if flag FLAG_CASE_HOENN_CLOSED goto Thanks
  if flag FLAG_CASE_HOENN_TAKEN goto Case
  say "I keep a tide table in my pocket. A fisherman who doesn't know the tide goes home with nothing."
  end
label Case
  if flag FLAG_CASE_HOENN_CLUE_TIDE goto Again
  say "The salt boat? It always comes in at the lowest tide of the evening. Look, I've marked the times in my table."
  say "Low tide is when the customs hut shuts for supper. Clever, or lucky. I've never seen her come in at any other time."
  setflag FLAG_CASE_HOENN_CLUE_TIDE
  text "{player} noted the tide table's low tides."
  end
label Again
  say "Lowest tide, every time, while customs has its supper. You could set your clock by it."
  end
label Thanks
  say "The salt boat comes in at high noon now, right under the customs officer's nose. Takes the fun out of it."
  end

script CaseHoennOttoline
  face self player
  speaker "Ottoline"
  if flag FLAG_CASE_HOENN_CULPRIT end
  scene "OttolineFaced"
  say "Looker's sent a Champion after me? I'm flattered. Yes, there's no salt in those crates. Never was."
  say "Lamp parts pay a fortune in duty, and the keepers can't afford it. But I'll not hand my boat over without a fight!"
  battle self as "case_hoenn_ottoline"
  say "...Run aground. Fair's fair. I'll pay the duty, every coin of it. Tell Looker he'll have it in writing."
  setflag FLAG_CASE_HOENN_CULPRIT
  clearflag FLAG_CASE_HOENN_OTTOLINE_WAITS
  hide self
  endscene
  end
