# The S.S. Aqua's deck (plan 20 · J15, part b). docs/scripts.md describes the language; the lines are our own, on
# HeartGold's beats (scr_seq_P01R0301).
#
# The crewman at the head of the gangway makes the crossing: from Olivine to Vermilion City, or back. Where the ship
# lies is FLAG_JOHTO_SS_AQUA_AT_VERMILION (a flag of ours), which the gangway's two ways go by (overlays/ss_aqua_1f.json).
# HeartGold's voyage is a night aboard, the lost granddaughter found and a sleep in the cabin; here the crossing is
# told over the dark, as the region's own crossings are (the rulings' "HeartGold's Kanto, part b").
#
# The cabins, the hold and the captain's room are open since part o. HeartGold's passengers change with the day of the
# voyage, in three groups behind FLAG_UNK_22A, 22B and 22C; here the deck's arrival shows one group by the weekday
# (Monday and Tuesday the first, Wednesday and Thursday the second, Friday to Sunday the third) and hides the others.


script OnEnter
  setflag FLAG_UNK_22A
  setflag FLAG_UNK_22B
  setflag FLAG_UNK_22C
  if weekday monday clearflag FLAG_UNK_22A
  if weekday tuesday clearflag FLAG_UNK_22A
  if weekday wednesday clearflag FLAG_UNK_22B
  if weekday thursday clearflag FLAG_UNK_22B
  if weekday friday clearflag FLAG_UNK_22C
  if weekday saturday clearflag FLAG_UNK_22C
  if weekday sunday clearflag FLAG_UNK_22C
  end


script Crewman
  face p01r0301_seaman_2_2 player
  speaker "Sailor"
  if flag FLAG_JOHTO_SS_AQUA_AT_VERMILION goto ToOlivine
  ask "Welcome aboard the S.S. Aqua! We're bound for Vermilion City in Kanto. Shall we cast off?"
  if no goto Wait
  say "Anchors aweigh! We sail overnight and dock in the morning. Your cabin's made up, so rest easy."
  scene "A night aboard the ship"
  fade out 0.8
  text "The S.S. Aqua slipped out of Olivine's harbour and turned east. The lighthouse's beam shrank away behind it..."
  call Night
  setflag FLAG_JOHTO_SS_AQUA_AT_VERMILION
  call Morning
  endscene
  speaker "Sailor"
  say "Morning! We've docked at Vermilion City, Kanto! Mind your step on the gangway."
  end
label ToOlivine
  ask "Welcome aboard the S.S. Aqua! We're bound for Olivine City in Johto. Shall we cast off?"
  if no goto Wait
  say "Anchors aweigh! We sail overnight and dock in the morning. Your cabin's made up, so rest easy."
  scene "A night aboard the ship homeward"
  fade out 0.8
  text "The S.S. Aqua slipped out of Vermilion's harbour and turned west, the city's lights low on the water behind it..."
  call Night
  clearflag FLAG_JOHTO_SS_AQUA_AT_VERMILION
  call Morning
  endscene
  speaker "Sailor"
  say "Morning! We've docked at Olivine City, Johto! Mind your step on the gangway."
  end
label Wait
  say "No rush. The gangway's behind you, if you'd rather stay ashore a while longer."
  end


# The night aboard (plan 20, part t; the rulings' "Johto's last leftovers, part t"): HeartGold's voyage is a night in a
# cabin; ours is told in a few beats of our own, over the dark and in the south-west cabins (SSAqua_1F_SouthwestRooms,
# the tile inside the east cabin's door): the evening at the porthole, the ship's sounds, the team rested as a bed rests
# it, and the horn in the morning. Nothing in it waits for the player but the lines.
# The cabin is lit as at nine in the evening and the deck as at seven in the morning, whatever the clock says, until the
# player next goes somewhere (part w: `lighting`, the picture's alone).
script Night
  warp "SSAqua_1F_SouthwestRooms" 23 3 down
  lighting 21
  fade in 0.8
  speaker none
  text "{player} was shown to a small cabin. Through the porthole, the last of the sunset sank into the sea."
  text "Somewhere along the corridor a passenger laughed, and a door clicked shut. The engines hummed below the floor."
  fade out 1.2
  text "Rocked by the waves, {player} fell asleep..."
  heal
  return


script Morning
  text "A long blast of the ship's horn! Morning light poured in through the porthole."
  warp "SSAqua_1F" 24 18 down
  lighting 7
  fade in 0.8
  return
