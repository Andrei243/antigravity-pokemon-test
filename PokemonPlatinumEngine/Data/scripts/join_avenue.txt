# Join Avenue (plan 22 · U6, part c; its growth plan 22 · U11). docs/scripts.md describes the language; the lines are our
# own.
#
# Three of the avenue's shopfronts are open, each a shopkeeper of ours behind a counter in one of the street's bays:
# the market, the flower shop's mulches and the café's drinks. The counters' ends read as the shops' signs.
#
# Its growth (Models/JoinAvenue.cs; the rulings, "Join Avenue's growth"): the owner asks the player to manage the
# avenue, and from then on the first coming of each day brings visitors (two more than the avenue's rank), and every ten
# visitors raise its rank, up to 10. Since U11's last pieces the shops widen their stock at levels of their own, by the
# money spent in each (the market at levels 2 and 4, the flowers and the café at 2), and the two shop girls open shops of
# their own at ranks 7 (evolution stones) and 9 (vitamins).
# Black 2's avenue grows with the passers-by met over the console's wireless; ours grows by days, its numbers ours.


script OnEnter
  joinavenue visit
  end


script Owner
  face self player
  speaker "Owner"
  if flag FLAG_UNOVA_JOIN_AVENUE_MANAGER goto Report
  say "Welcome to Join Avenue! As you see, the street is quiet yet. One day it will be the liveliest in Unova!"
  say "What it needs is a manager: someone to bring the visitors in, so the shops can grow with them."
  ask "You have the look of a Trainer people follow. Will you be Join Avenue's manager?"
  if no goto Later
  setflag FLAG_UNOVA_JOIN_AVENUE_INVITATIONS
  setflag FLAG_UNOVA_JOIN_AVENUE_MANAGER
  joinavenue visit
  say "Splendid! From today the avenue is yours to run."
  say "Visitors will come by each day you drop in. The more come, the higher the avenue's rank, and the better its shops!"
  say "Some of them may even ask to open a shop in one of our empty bays. Say yes, and the street fills up!"
  say "And every shop grows with what its customers spend there. Spend freely, Manager!"
  end
label Later
  say "No? Well, the offer stands. Come and see me if you change your mind."
  end
label Report
  joinavenue visitors
  say "Manager! Join Avenue has had {number} visitors so far."
  joinavenue rank
  say "That makes it a rank {number} avenue."
  if result >= 10 goto Top
  joinavenue next
  say "{number} more visitors and it will rank up. Keep dropping in!"
  end
label Top
  say "That's as high as an avenue goes. Unova has never seen a street like it!"
  end

# The avenue's other shops (plan 22 · U11's leftovers; rulings, "Join Avenue's other shops"): Black 2's raffle, dojo, salon,
# nursery and antique shop, each in a bay of the street with its keeper behind a counter (Models/JoinAvenue.cs; every number
# ours). Since U11's last pieces each opens when a visitor the player invites sets it up (the visitors' Invite, below; its
# keeper is shown by the shop's FLAG_UNOVA_JOIN_AVENUE_<SHOP>_OPEN), and every shop rises a level with the money spent in it
# (joinavenue till, tally): its stock, the dojo's effort, the salon's friendship and the raffle's prizes go by the level. The
# raffle draws on the game's own chance once a day.
script Raffle
  face self player
  speaker "Raffle Clerk"
  ask "Welcome to the raffle! One free ticket a day, and the busier the avenue, the better the prizes. Draw today's?"
  if no goto Bye
  joinavenue raffle
  if result == 0 goto Drawn
  say "Congratulations! Come back tomorrow for another ticket."
  end
label Drawn
  say "You've drawn today's ticket already. There'll be a new one for you tomorrow!"
  end
label Bye
  say "No? Your ticket will wait until the end of the day."
  end


script Dojo
  face self player
  speaker "Dojo Master"
  say "Hah! Welcome to the dojo. One session is $1,000, and your Pokémon comes out tougher in the stat you pick."
  if money < 1000 goto Poor
  choose "Which stat shall we train?" "HP" "ATTACK" "DEFENSE" "SP. ATK" "SP. DEF" "SPEED"
  if result == 0 goto Hp
  if result == 1 goto Attack
  if result == 2 goto Defense
  if result == 3 goto SpAtk
  if result == 4 goto SpDef
  if result == 5 goto Speed
  goto Bye
label Hp
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo hp
  goto Trained
label Attack
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo attack
  goto Trained
label Defense
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo defense
  goto Trained
label SpAtk
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo spatk
  goto Trained
label SpDef
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo spdef
  goto Trained
label Speed
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo speed
label Trained
  if result >= 254 goto Egg
  if result == 0 goto Full
  joinavenue till
  takemoney 1000
  joinavenue tally dojo
  sound "select"
  say "Hup! Hah! {wearer} trained hard and came out tougher than it went in!"
  end
label Full
  say "{wearer} can't grow any tougher there. Its training is complete! I won't take your money for nothing."
  end
label Egg
  say "An Egg? Ha! Bring it back once it's hatched and has some fight in it."
  end
label Poor
  say "You're short of the fee, I'm afraid. Come back with $1,000!"
  end
label Bye
  say "Come back when you're ready to sweat!"
  end


script Salon
  face self player
  speaker "Stylist"
  ask "Welcome to the salon! A grooming is $500, and your Pokémon will love you all the more. Shall we?"
  if no goto Bye
  if money < 500 goto Poor
  choosepokemon
  if result == 255 goto Bye
  joinavenue salon
  if result >= 254 goto Egg
  if result == 0 goto Full
  joinavenue till
  takemoney 500
  joinavenue tally salon
  sound "select"
  say "All done! {wearer} looks wonderful, and it knows exactly who to thank for it."
  end
label Full
  say "{wearer} couldn't love you any more than it already does! No grooming needed today."
  end
label Egg
  say "An Egg? There's nothing to brush yet! Bring it back once it's hatched."
  end
label Poor
  say "Oh dear, you're a little short. A grooming is $500."
  end
label Bye
  say "Do come back! A brushed Pokémon is a happy Pokémon."
  end


script Nursery
  face self player

  end

  end

# The visitors (plan 22 · U11's leftovers): three of the passers-by of ours stand in the street once the avenue is managed,
# picked by its visitors so far, standing in for those met over the console's wireless. Each greets the player in their
# own words (ui.json's joinAvenue.visitor.N). Since U11's last pieces each would open a shop of their own in an empty bay
# (JoinAvenue.Offer): asked, and let, they open it, as many as the avenue's rank has room for.
script Visitor1
  face self player
  joinavenue visitor 1
  joinavenue invite 1
  call Invite
  end


script Visitor2
  face self player
  joinavenue visitor 2
  joinavenue invite 2
  call Invite
  end


script Visitor3
  face self player
  joinavenue visitor 3
  joinavenue invite 3
  call Invite
  end


# What a visitor's offer comes to (RESULT of joinavenue invite): 0 their shop is open, 9 no room, 1 to 5 the shop
script Invite
  if result == 0 return
  if result == 9 goto NoRoom
  if result == 1 goto Raffle
  if result == 2 goto Dojo
  if result == 3 goto Salon
  if result == 4 goto Nursery
  ask "You're the manager? I've always dreamt of running an antique shop. Might I open one in an empty bay here?"
  if no goto Declined
  setflag FLAG_UNOVA_JOIN_AVENUE_ANTIQUES_OPEN
  goto Opened
label Raffle
  ask "You're the manager? This street needs a raffle, and I'd love to run it. Might I open one in an empty bay here?"
  if no goto Declined
  setflag FLAG_UNOVA_JOIN_AVENUE_RAFFLE_OPEN
  goto Opened
label Dojo
  ask "You're the manager? I've trained Pokémon all my life. Might I open a dojo in an empty bay here?"
  if no goto Declined
  setflag FLAG_UNOVA_JOIN_AVENUE_DOJO_OPEN
  goto Opened
label Salon
  ask "You're the manager? I'm a Pokémon groomer by trade. Might I open a salon in an empty bay here?"
  if no goto Declined
  setflag FLAG_UNOVA_JOIN_AVENUE_SALON_OPEN
  goto Opened
label Nursery
  ask "You're the manager? I've looked after Eggs for years. Might I open a nursery in an empty bay here?"
  if no goto Declined
  setflag FLAG_UNOVA_JOIN_AVENUE_NURSERY_OPEN
  goto Opened
label Opened
  sound "select"
  say "Thank you, Manager! I'll have it ready in no time. Do come and see how it does!"
  return
label Declined
  say "Oh, that's a shame. If you change your mind, you know where to find me."
  return
label NoRoom
  say "I'd love to open a shop here one day, but the avenue isn't busy enough for another yet. Bring in more visitors!"
  return
