# Join Avenue (plan 22 · U6, part c; its growth plan 22 · U11). docs/scripts.md describes the language; the lines are our
# own.
#
# Three of the avenue's shopfronts are open, each a shopkeeper of ours behind a counter in one of the street's bays:
# the market, the flower shop's mulches and the café's drinks. The counters' ends read as the shops' signs.
#
# Its growth (Models/JoinAvenue.cs; the rulings, "Join Avenue's growth"): the owner asks the player to manage the
# avenue, and from then on the first coming of each day brings visitors (two more than the avenue's rank), and every ten
# visitors raise its rank, up to 10. The three shops widen their stock at ranks 3 and 6 (the market), 4 (the flowers)
# and 5 (the café), and the two shop girls open shops of their own at ranks 7 (evolution stones) and 9 (vitamins).
# Black 2's avenue grows with the passers-by met over the console's wireless; ours grows by days, its numbers ours.


script OnEnter
  joinavenue visit
  end


script Owner
  face self player
  speaker "Owner"
  if flag FLAG_UNOVA_JOIN_AVENUE_MANAGER goto Report
  say "Welcome to Join Avenue! As you see, the street is quiet yet. One day it will be the liveliest in Unova!"
  say "What it needs is a manager: someone to bring the visitors in, so the shops can grow with them."
  ask "You have the look of a Trainer people follow. Will you be Join Avenue's manager?"
  if no goto Later
  setflag FLAG_UNOVA_JOIN_AVENUE_MANAGER
  joinavenue visit
  say "Splendid! From today the avenue is yours to run."
  say "Visitors will come by each day you drop in. The more come, the higher the avenue's rank, and the better its shops!"
  end
label Later
  say "No? Well, the offer stands. Come and see me if you change your mind."
  end
label Report
  joinavenue visitors
  say "Manager! Join Avenue has had {number} visitors so far."
  joinavenue rank
  say "That makes it a rank {number} avenue."
  if result >= 10 goto Top
  joinavenue next
  say "{number} more visitors and it will rank up. Keep dropping in!"
  end
label Top
  say "That's as high as an avenue goes. Unova has never seen a street like it!"
  end

# The avenue's other shops (plan 22 · U11's leftovers; rulings, "Join Avenue's other shops"): Black 2's raffle, dojo, salon,
# nursery and antique shop, each in a bay of the street with its keeper behind a counter, open from their ranks and doing more
# the higher the avenue's rank (Models/JoinAvenue.cs; every number ours). The raffle draws on the game's own chance once a day.
script Raffle
  face self player
  speaker "Raffle Clerk"
  ask "Welcome to the raffle! One free ticket a day, and the busier the avenue, the better the prizes. Draw today's?"
  if no goto Bye
  joinavenue raffle
  if result == 0 goto Drawn
  say "Congratulations! Come back tomorrow for another ticket."
  end
label Drawn
  say "You've drawn today's ticket already. There'll be a new one for you tomorrow!"
  end
label Bye
  say "No? Your ticket will wait until the end of the day."
  end


script Dojo
  face self player
  speaker "Dojo Master"
  joinavenue open dojo
  if result == 0 goto Closed
  say "Hah! Welcome to the dojo. One session is $1,000, and your Pokémon comes out tougher in the stat you pick."
  if money < 1000 goto Poor
  choose "Which stat shall we train?" "HP" "ATTACK" "DEFENSE" "SP. ATK" "SP. DEF" "SPEED"
  if result == 0 goto Hp
  if result == 1 goto Attack
  if result == 2 goto Defense
  if result == 3 goto SpAtk
  if result == 4 goto SpDef
  if result == 5 goto Speed
  goto Bye
label Hp
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo hp
  goto Trained
label Attack
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo attack
  goto Trained
label Defense
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo defense
  goto Trained
label SpAtk
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo spatk
  goto Trained
label SpDef
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo spdef
  goto Trained
label Speed
  choosepokemon
  if result == 255 goto Bye
  joinavenue dojo speed
label Trained
  if result >= 254 goto Egg
  if result == 0 goto Full
  takemoney 1000
  sound "select"
  say "Hup! Hah! {wearer} trained hard and came out tougher than it went in!"
  end
label Full
  say "{wearer} can't grow any tougher there. Its training is complete! I won't take your money for nothing."
  end
label Egg
  say "An Egg? Ha! Bring it back once it's hatched and has some fight in it."
  end
label Poor
  say "You're short of the fee, I'm afraid. Come back with $1,000!"
  end
label Bye
  say "Come back when you're ready to sweat!"
  end
label Closed
  say "The dojo isn't open yet. Once the avenue's busier, I'll have the room to train your Pokémon!"
  end


script Salon
  face self player
  speaker "Stylist"
  joinavenue open salon
  if result == 0 goto Closed
  ask "Welcome to the salon! A grooming is $500, and your Pokémon will love you all the more. Shall we?"
  if no goto Bye
  if money < 500 goto Poor
  choosepokemon
  if result == 255 goto Bye
  joinavenue salon
  if result >= 254 goto Egg
  if result == 0 goto Full
  takemoney 500
  sound "select"
  say "All done! {wearer} looks wonderful, and it knows exactly who to thank for it."
  end
label Full
  say "{wearer} couldn't love you any more than it already does! No grooming needed today."
  end
label Egg
  say "An Egg? There's nothing to brush yet! Bring it back once it's hatched."
  end
label Poor
  say "Oh dear, you're a little short. A grooming is $500."
  end
label Bye
  say "Do come back! A brushed Pokémon is a happy Pokémon."
  end
label Closed
  say "The salon isn't open yet. We're waiting for the avenue to bring the customers in!"
  end
