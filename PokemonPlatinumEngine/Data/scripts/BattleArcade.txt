# The Battle Arcade's lobby (plan 06 · R18d). docs/scripts.md describes the language; the lines are our own, on the
# original's beats (scripts_battle_arcade.s).
#
# Two attendants stand before the doorway in the back wall, beside the records' machine and the PC. The first takes
# Single and Double challenges: three Pokémon of different kinds are entered, the team is healed, the game is saved,
# and she leads the way. Whatever the three hold is kept at the door and handed back at the end. The second would
# take a Multi challenge beside a friend, which isn't held (decision 06 · 7).
#
# VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION says how the player comes back into the lobby, as in the original: 1 with a
# round of seven won (the print is handed over), 2 with the challenge put down between two battles, 3 with it lost
# or given up, and 255 while a challenge is under way, which is only ever found here by someone who closed the game
# in the middle of one: that streak is over.
#
# Not here: the reporter who interviews whoever has a streak to talk about (plan 08), and a challenge beside a
# friend over a link (plan 07).


script OnEnter
  # Whoever led the way to the game room is back at her post
  place single_attendant 14 5 down
  place multi_attendant 15 5 down
  if var VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION == 1 goto Won
  if var VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION == 2 goto PutDown
  if var VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION == 3 goto Over
  if var VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION == 255 goto LeftOpen
  end
label Won
  call RoundWon
  end
label PutDown
  call Paused
  end
label Over
  call ChallengeEnded
  end
label LeftOpen
  call WalkedOut
  end


# The first attendant: Single and Double challenges (BattleArcade_SingleAttendant).
script SingleAttendant
  if var VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION == 2 goto Waiting
  arcade end
  say "Welcome to the Battle Arcade, where the game board has the last word! I look after the Single and Double challenges."
  choose "Which challenge would you like to take?" "SINGLE BATTLE" "DOUBLE BATTLE" "INFO" "EXIT"
  if result == 0 goto Single
  if result == 1 goto Double
  if result == 2 goto Info
  goto Bye
label Waiting
  call Paused
  end
label Info
  say "You enter three Pokémon and battle seven trainers. Before every battle the game board is set going, and whatever its cursor stops on happens."
  say "It may fall on you, on your opponent or on both of you, and it may be a help or a hindrance. Your Pokémon are made whole again before each battle, and whatever they hold is looked after at the door."
  say "Stop the cursor yourself if you can. Leave it for thirty seconds and it stops where it pleases."
  end
label Single
  setvar VAR_BATTLE_ARCADE_CHALLENGE_TYPE 0
  goto Choose
label Double
  setvar VAR_BATTLE_ARCADE_CHALLENGE_TYPE 1
label Choose
  arcade enough
  if result == 0 goto TooFew
  say "Please choose the three Pokémon you will enter."
  arcade choose
  if result == 0 goto Bye
  arcade check
  if result == 1 goto SameKind
  call GoIn
  arcade active
  if result == 0 end
  # BattleArcade_WalkToCorridorSingleChallenge: the attendant leads the way round to the doorway behind her
  walk self up 1 right 1 up 1
  hide self
  walk player up 2 right 1 up 1
  sound "stairs"
  warp "BattleArcadeCorridor" 11 11 up
  end
label TooFew
  say "You need three Pokémon, each of a different kind, to take this challenge."
  say "Eggs can't enter, and neither can {banned}."
  goto Bye
label SameKind
  say "I'm sorry. The three Pokémon must each be of a different kind."
label Bye
  say "We hope to see you again!"
  end


# The team is healed and the game saved before a challenge goes in (BattleArcade_HealAndSaveBeforeChallenge).
script GoIn
  heal
  ask "Your game must be saved before the challenge begins. Shall I save it?"
  if no goto NoSave
  if var VAR_BATTLE_ARCADE_CHALLENGE_TYPE == 1 goto Double
  arcade begin single
  goto Begun
label Double
  arcade begin double
label Begun
  call Prepare
  frontier save
  text "{player} saved the game."
  arcade enter
  say "Thank you. Right this way, please!"
  return
label NoSave
  say "Then please come and see me when you are ready."
  return


# The game room as nobody is in it: what the lobby sets before a challenge goes in.
script Prepare
  setflag FLAG_HIDE_BATTLE_ARCADE_ROOM_OPPONENT
  setvar VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION 255
  return


# The second attendant: a Multi challenge, which two players take side by side (BattleArcade_MultiAttendant).
script MultiAttendant
  say "Welcome to the Battle Arcade! I look after the Multi challenge, which two trainers take side by side with two Pokémon each."
  say "I'm sorry to say nobody can be linked up with. Only the Single and Double challenges are held."
  end


# Back from a round of seven won (BattleArcade_OnFrame_ChallengeEndedCompletedRound): the print for beating the
# Arcade Star is handed over, and the game is saved.
script RoundWon
  setvar VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION 0
  arcade active
  if result == 0 end
  speaker "Attendant"
  say "Seven games, seven wins! Congratulations!"
  frontier print arcade
  if result == 1 call SilverPrint
  if result == 2 call GoldPrint
  arcade end
  text "The result is being recorded. {player} saved the game."
  frontier save
  say "We hope to see you again!"
  end

script SilverPrint
  say "And you beat the Arcade Star herself! Please take this to remember it by."
  text "A Silver Print was added to {player}'s record of the Battle Arcade!"
  fanfare keyitem
  return

script GoldPrint
  say "You beat the Arcade Star with every light on the board against you! Nobody will believe it without this."
  text "A Gold Print was added to {player}'s record of the Battle Arcade!"
  fanfare keyitem
  return


# Back from a challenge lost or given up (BattleArcade_OnFrame_ChallengeEnded).
script ChallengeEnded
  setvar VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION 0
  arcade active
  if result == 0 end
  arcade end
  speaker "Attendant"
  say "Thank you for playing."
  text "The result is being recorded. {player} saved the game."
  frontier save
  say "We hope to see you again!"
  end


# A challenge put down between two battles (BattleArcade_OnFrame_ResumeChallenge): the game is saved with it, and
# it is taken up again when the player says so, by the three it was entered with.
script Paused
  speaker "Attendant"
  arcade paused
  if result == 0 goto Nothing
  text "{player} saved the game."
  frontier save
  ask "We have missed you! Your challenge is waiting where you put it down. Would you like to go on with it now?"
  if no goto Later
  arcade resume
  if result == 0 goto Gone
  call Prepare
  arcade enter
  say "Then right this way, please!"
  warp "BattleArcadeCorridor" 11 11 up
  end
label Later
  say "Very well. Speak to me whenever you are ready."
  end
label Gone
  say "The Pokémon you entered are no longer with you as they were, so the challenge can't be taken up again. I'm afraid its streak ends there."
label Nothing
  setvar VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION 0
  end


# The game was closed in the middle of a challenge (BattleArcade_OnFrame_DidntSaveBeforeQuit).
script WalkedOut
  setvar VAR_BATTLE_ARCADE_LOBBY_LOAD_ACTION 0
  speaker "Attendant"
  say "Excuse me! The last challenge was left before it was over, so it can't be taken up again."
  say "I'm afraid its streak ends there."
  arcade forfeit
  say "We hope to see you again!"
  end


# The machine beside the PC: the Arcade's records.
script RecordsMachine
  sound "select"
  frontier records arcade
  end
