# The Nature Preserve (plan 22 · U10, part h). docs/scripts.md describes the language; the lines are our own. The Cargo
# Service's plane lands a Champion of Unova here from Mistralton (mistralton_city_3.txt's NaturePreserveFlight); the pilot (0)
# by the plane flies them back. Black 2's shiny Haxorus (1) waits in the clearing by the lake once Unova's Hall of Fame is
# entered (shown by UnovaHallOfFame), level 60 and shiny as Black 2's is: caught or beaten it is gone for good
# (FLAG_UNOVA_FOUGHT_HAXORUS); run from, it slips off into the trees and is back the next time the preserve is come to (the
# rulings' "The Nature Preserve").


script Pilot
  face 0 player
  speaker "Pilot"
  say "This island's kept wild on purpose. Nobody lives here, so the Pokémon grow as they please."
  ask "Shall I fly you back to Mistralton City?"
  if no goto Stay
  say "Buckle up! Mistralton Cargo Service, homeward bound."
  call FlightHome
  end
label Stay
  say "Take your time. I'll wait by the plane."
  end


script FlightHome
  scene "The flight home from the Nature Preserve"
  fade out 0.8
  music stop
  sound "crossing_engine"
  text "The plane lifts off the island's strip and swings out over the sea..."
  text "The preserve's woods shrink behind the tail, and Mistralton's runway comes up to meet the wheels."
  endscene
  warp "Unova" 108 305 down
  fade in
  return


script Haxorus
  if not flag UnovaHallOfFame end
  scene "Haxorus in the Nature Preserve"
  music stop
  face self player
  letterbox on 0.4
  camera zoom 1.3 0.8
  shake self 6
  cry "Haxorus"
  speaker none
  text "The Haxorus's scales flash gold in the light!\nIt bares its tusks and charges!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Haxorus" 60 shiny
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_HAXORUS
  hide self
  text "The clearing is quiet again. Only the wind moves in the trees..."
  end
label Fled
  hide self
  text "The Haxorus crashed off into the trees..."
  end
