# Ultra Space Wilds, the yellow light's world (plan 23, Alola finished: the Ultra Warp Ride; docs/mechanics/rulings.md,
# "Alola finished"). The lines are our own; which legendary Pokémon live in which of the four worlds, and that three need
# others of their kind in the team, follows Ultra Sun and Ultra Moon as they are publicly described (Serebii's and
# Smogon's lists), both versions' together. No picture, map, line or script of the games is read.
# Each arrival draws one of the world's legendary Pokémon not yet caught here (FLAG_ALOLA_WARP_<NAME>_CAUGHT) and shows it
# on the dais (FLAG_MAP_LOCAL_WARP_<NAME>, which lasts only while the player stays); the draw begins at a die of 8 and
# goes on round the list past those caught or not yet called. Each is met at level 60 after a beat of its own, and is
# back on a later ride until it is caught.


script OnEnter
  effect rift 32 55 hold
  random 8
  setvar VAR_ALOLA_WARP_PICK RESULT
  setvar VAR_ALOLA_WARP_TRIES 0
label Try
  if var VAR_ALOLA_WARP_TRIES >= 8 goto Nobody
  addvar VAR_ALOLA_WARP_TRIES 1
  if var VAR_ALOLA_WARP_PICK == 0 goto P0
  if var VAR_ALOLA_WARP_PICK == 1 goto P1
  if var VAR_ALOLA_WARP_PICK == 2 goto P2
  if var VAR_ALOLA_WARP_PICK == 3 goto P3
  if var VAR_ALOLA_WARP_PICK == 4 goto P4
  if var VAR_ALOLA_WARP_PICK == 5 goto P5
  if var VAR_ALOLA_WARP_PICK == 6 goto P6
  if var VAR_ALOLA_WARP_PICK == 7 goto P7
  goto Nobody
label P0
  if flag FLAG_ALOLA_WARP_REGIROCK_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_REGIROCK
  end
label P1
  if flag FLAG_ALOLA_WARP_REGICE_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_REGICE
  end
label P2
  if flag FLAG_ALOLA_WARP_REGISTEEL_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_REGISTEEL
  end
label P3
  if flag FLAG_ALOLA_WARP_GROUDON_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_GROUDON
  end
label P4
  if flag FLAG_ALOLA_WARP_PALKIA_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_PALKIA
  end
label P5
  if flag FLAG_ALOLA_WARP_HEATRAN_CAUGHT goto Next
  setflag FLAG_MAP_LOCAL_WARP_HEATRAN
  end
label P6
  if flag FLAG_ALOLA_WARP_REGIGIGAS_CAUGHT goto Next
  if not has "Regirock" goto Next
  if not has "Regice" goto Next
  if not has "Registeel" goto Next
  setflag FLAG_MAP_LOCAL_WARP_REGIGIGAS
  end
label P7
  if flag FLAG_ALOLA_WARP_GIRATINA_CAUGHT goto Next
  if not has "Dialga" goto Next
  if not has "Palkia" goto Next
  setflag FLAG_MAP_LOCAL_WARP_GIRATINA
  end
label Next
  addvar VAR_ALOLA_WARP_PICK 1
  if var VAR_ALOLA_WARP_PICK >= 8 setvar VAR_ALOLA_WARP_PICK 0
  goto Try
label Nobody
  speaker none
  text "A cave of yellow stone, warm to the touch. Deep inside, something breathes like a slow furnace."
  end


script Regirock
  scene "Regirock in the Ultra Space Wilds"
  music stop
  letterbox on 0.4
  # (one tile south of it, the player would hide it behind their head: they step back first)
  if var PLAYER_X != 32 goto Look
  if var PLAYER_Y != 17 goto Look
  walk player down 2
  face player up
label Look
  camera pan 32 16 1.0
  camera zoom 1.4 0.8
  shake self 6
  cry "Regirock"
  speaker none
  text "The rock wall shook, and a giant of stone pulled itself free of it. Regirock!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Regirock" 60
  if result == 3 setflag FLAG_ALOLA_WARP_REGIROCK_CAUGHT
  hide self
  music area
  end
