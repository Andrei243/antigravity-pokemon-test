# Petalburg City's Gym, the player's father's (plan 21 · H2, plan 21 · H4, part b). docs/scripts.md describes the
# language; the lines are our own, on Emerald's beats (data/maps/PetalburgCity_Gym/scripts.inc).
#
# VAR_PETALBURG_GYM_STATE, as Emerald counts it: 0 Norman met at the door, 1 back from Wally's catch, 2 told to win four
# Badges first, and one more for each of the Badges of Rustboro, Dewford, Mauville and Lavaridge, so 6 is four Badges
# won and Norman's battle waiting, 7 Norman beaten. Until 6 Norman stands in the hall at the door (Emerald's
# MoveNormanToEntrance) and the hall's two doors are shut; from 6 he waits in his own room at the head of the Gym.
#
# The rooms lie one behind another, each with a Cooltrainer who battles only when spoken to and two doors in its north
# wall. A door is read (Emerald's door scripts): shut until the room's trainer is beaten (the hall's until four
# Badges), and then it asks before it lets the player through into the room beyond, onto the floor before the mat that
# leads back. Norman's Badge, the Balance Badge, lets Surf be used outside battle; he gives TM42, Facade
# (FLAG_RECEIVED_TM_FACADE), counts his Gym's trainers beaten, sends Wattson out into Mauville's street, and Wally's
# father comes in to take the player home with him for HM03, Surf (petalburg_city_wallys_house.txt).


script OnEnter
  if var VAR_PETALBURG_GYM_STATE >= 6 end
  place petalburg_gym_norman 4 107 down
  if var VAR_PETALBURG_GYM_STATE != 1 end
  scene "Wally's thanks"
  clearflag FLAG_HIDE_PETALBURG_GYM_WALLY
  place petalburg_gym_wally 5 108 up
  show petalburg_gym_wally
  speaker "Wally"
  say "Mr. Norman! I caught a Ralts! Thank you so much for lending me your Zigzagoon."
  face petalburg_gym_wally player
  say "And thank you too, {player}. I'll never forget it. I'm going to be a trainer, just like you!"
  walk petalburg_gym_wally down 3
  hide petalburg_gym_wally
  setflag FLAG_HIDE_PETALBURG_GYM_WALLY
  face petalburg_gym_norman player
  speaker "Norman"
  say "That boy's eyes were shining. You did a good thing today, {player}."
  say "Now, about you. You want to battle me, don't you? I can see it. But I'm a Gym Leader, and I have my rules."
  say "Win four Gym Badges first. Then come back, and we'll see what you've learned. Start with Rustboro City, north of the woods."
  setvar VAR_PETALBURG_GYM_STATE 2
  endscene
  end


# Norman (Emerald's PetalburgCity_Gym_EventScript_Norman): at the first visit, Wally comes in with his father's word
# and Norman sends the player out with him to catch his first Pokémon; with four Badges, his battle
script Norman
  if flag FLAG_PHONE_REMATCH_NORMAN goto Rematch
  if var VAR_PETALBURG_GYM_STATE >= 7 goto After
  if var VAR_PETALBURG_GYM_STATE >= 6 goto Battle
  if var VAR_PETALBURG_GYM_STATE >= 2 goto Rules
  scene "Norman and Wally"
  speaker "Norman"
  say "{player}! So you made it to Petalburg on your own. And with a Pokémon of your own, too. I'm proud of you."
  say "It's a long way from Littleroot. Your mother must have told you I'm the Leader of this Gym now."
  clearflag FLAG_HIDE_PETALBURG_GYM_WALLY
  place petalburg_gym_wally 4 111 up
  show petalburg_gym_wally
  walk petalburg_gym_wally up 2
  speaker "Wally"
  say "Um... excuse me. My name is Wally. I'm going to live with my relatives in Verdanturf Town for a while."
  say "I thought... if I had a Pokémon with me, I wouldn't feel so lonely. But I've never caught one."
  speaker "Norman"
  say "I see. Here, Wally, take my Zigzagoon. Weaken a wild Pokémon with it, then throw a Poké Ball."
  say "{player}, go with Wally and keep an eye on him. There's tall grass just east of town."
  speaker "Wally"
  say "Th-thank you, Mr. Norman! Let's go, {player}!"
  setflag FLAG_HIDE_PETALBURG_CITY_WALLYS_MOM
  setflag FLAG_HIDE_LITTLEROOT_TOWN_BIRCHS_LAB_RIVAL
  clearflag FLAG_HIDE_PETALBURG_CITY_WALLY
  setvar VAR_PETALBURG_GYM_STATE 1
  setvar VAR_PETALBURG_CITY_STATE 2
  fade out
  hide petalburg_gym_wally
  setflag FLAG_HIDE_PETALBURG_GYM_WALLY
  warp "Hoenn" 55 241 down
  endscene
  end
label Rules
  face petalburg_gym_norman player
  speaker "Norman"
  if var VAR_PETALBURG_GYM_STATE >= 4 goto Stronger
  if var VAR_PETALBURG_GYM_STATE >= 3 goto Dewford
  say "Four Gym Badges, {player}. Then come back and face me. Start with Roxanne in Rustboro City."
  end
label Dewford
  say "You've won Rustboro's Badge? Well done! Brawly's Gym is next, in Dewford Town across the sea. Mr. Briney sails there."
  end
label Stronger
  say "You've grown stronger, {player}. I can see it in the way your Pokémon look at you. Keep going: four Badges."
  end
label Battle
  face petalburg_gym_norman player
  scene "Norman's battle"
  speaker "Norman"
  say "So. You really have won four Gym Badges. Then, as I promised, we'll battle."
  say "{player}... I can't tell you how glad I am to face my own child in a real battle."
  say "But a battle is a battle. As a Gym Leader, I'll give it everything I have!"
  battle self as "hoenn_norman_1"
  say "I... lost? To you, {player}? I can hardly believe it."
  say "But rules are rules. You've earned this. Take it."
  givebadge balance
  text "{player} received the Balance Badge from Norman!"
  say "With the Balance Badge, your Pokémon can use Surf outside of battle to cross the water."
  setflag FLAG_DEFEATED_PETALBURG_GYM
  setvar VAR_PETALBURG_GYM_STATE 7
  setflag FLAG_HIDE_MAUVILLE_GYM_WATTSON
  clearflag FLAG_HIDE_MAUVILLE_CITY_WATTSON
  defeat "hoenn_randall"
  defeat "hoenn_mary"
  defeat "hoenn_parker"
  defeat "hoenn_alexia"
  defeat "hoenn_george"
  defeat "hoenn_jody"
  defeat "hoenn_berke"
  say "And take this as well."
  give "TM42"
  setflag FLAG_RECEIVED_TM_FACADE
  say "TM42 holds Facade. When its user is poisoned, burned or paralysed, it hits twice as hard. Turn a bad spot into strength!"
  say "As a Gym Leader, I'm more upset than I can say. But as your father... it makes me happy, and a little sad. Strange, isn't it?"
  wait 0.6
  sound "door_open"
  clearflag FLAG_HIDE_PETALBURG_GYM_WALLYS_DAD
  place petalburg_gym_wallys_dad 1 7 up
  show petalburg_gym_wallys_dad
  walk petalburg_gym_wallys_dad up 1
  emote petalburg_gym_wallys_dad exclaim
  walk petalburg_gym_wallys_dad right 2
  walk petalburg_gym_wallys_dad up 3
  face petalburg_gym_wallys_dad player
  speaker "Wally's Father"
  say "Ah, there you are, {player}! Would you come with me? There's something I'd like you to have."
  face petalburg_gym_wallys_dad up
  say "Norman, you won't mind if I borrow {player} for a minute or two, will you?"
  setvar VAR_PETALBURG_CITY_STATE 4
  fade out
  hide petalburg_gym_wallys_dad
  setflag FLAG_HIDE_PETALBURG_GYM_WALLYS_DAD
  warp "PetalburgCity_WallysHouse" 2 4 right
  endscene
  end
label After
  face petalburg_gym_norman player
  speaker "Norman"
  if flag FLAG_RECEIVED_TM_FACADE goto Training
  give "TM42"
  setflag FLAG_RECEIVED_TM_FACADE
  say "TM42 holds Facade. When its user is poisoned, burned or paralysed, it hits twice as hard. Turn a bad spot into strength!"
  end
label Training
  say "Go and see your mother now and then, {player}. I'll be here, training harder than ever. I can't stay beaten by my own child!"
  # Spoken to with his Badge and a PokéNav, he gives his number (plan 21 · H12, part g; ours)
  if flag FLAG_ENABLE_NORMAN_MATCH_CALL end
  if not flag FLAG_SYS_POKENAV_GET end
  setflag FLAG_ENABLE_NORMAN_MATCH_CALL
  say "And let me have your PokéNav's number. A father should be able to call his child."
  addcontact norman
  text "Registered Norman in the PokéNav."
  end
# The rematch a Match Call asked for (plan 21 · H12, part g; Emerald's PetalburgCity_Gym_EventScript_NormanRematch, a
# double battle): each time one team stronger, Emerald's TRAINER_NORMAN_2 to _5, the last again once all are beaten
label Rematch
  face petalburg_gym_norman player
  speaker "Norman"
  if party < 2 goto NeedTwo
  say "You came, {player}. This time I won't hold anything back, not even as your father. Two Pokémon each!"
  if not defeated "hoenn_norman_2" goto Second
  if not defeated "hoenn_norman_3" goto Third
  if not defeated "hoenn_norman_4" goto Fourth
  battle self as "hoenn_norman_5"
  goto Rematched
label Second
  battle self as "hoenn_norman_2"
  goto Rematched
label Third
  battle self as "hoenn_norman_3"
  goto Rematched
label Fourth
  battle self as "hoenn_norman_4"
label Rematched
  speaker "Norman"
  clearflag FLAG_PHONE_REMATCH_NORMAN
  say "Beaten again by my own child. I couldn't be prouder, {player}. Call me whenever you want another."
  end
label NeedTwo
  say "A rematch is two Pokémon at a time, {player}. Come back with at least two on your team."
  end

script GymGuide
  speaker "Gym Guide"
  if flag FLAG_DEFEATED_PETALBURG_GYM goto Won
  if var VAR_PETALBURG_GYM_STATE >= 6 goto Advice
  say "Hey there! This is the Petalburg Gym. Norman's rooms each test a trainer's skill a different way."
  say "But the doors stay shut until you've got four Badges. Rules are rules!"
  end
label Advice
  say "Four Badges! The doors are open to you now. Each one beyond opens once you beat the trainer in its room."
  say "The left door goes to the Speed Room, the right to the Accuracy Room. Norman's Pokémon are Normal types: Fighting moves hit them hard!"
  end
label Won
  say "You beat your own father! Whoa. I don't think anybody in Petalburg will ever forget that battle."
  end
