# Goldenrod City (plan 20 · J11, part b). docs/scripts.md describes the language; the lines are our own, on
# HeartGold's beats (scr_seq_T25).
#
# Team Rocket's takeover of the Radio Tower is HeartGold's one count, VAR_SCENE_ROCKET_TAKEOVER: 1 once the Storm,
# Mineral and Glacier Badges are all held, 2 once Professor Elm has rung (common.JohtoTakeoverCall), 3 once the
# player has borrowed a uniform in the passage under the city (goldenrod_tunnel_b1f.txt), 4 once the grunt at the
# tower's stairs has seen through it (goldenrod_radio_tower_1f.txt), and 5 once Archer is beaten on the observation
# deck (goldenrod_radio_tower_observation_deck.txt). The town's arrival shows and hides its people by it, as
# HeartGold's T25_018 does: the townspeople (FLAG_HIDE_ROCKET_TAKEOVER_2) keep indoors from 2 to 4, the man in black
# before the tower (_3) stands there until the takeover is over, the grunts about the streets come out at 1 (_4) and
# at 2 (_1), and at 3 two more guard the way north and one the way south (_5).


script OnEnter
  setflag FLAG_HIDE_ROCKET_TAKEOVER_3
  setflag FLAG_HIDE_ROCKET_TAKEOVER_4
  setflag FLAG_HIDE_ROCKET_TAKEOVER_1
  setflag FLAG_HIDE_ROCKET_TAKEOVER_5
  setflag FLAG_HIDE_ROCKET_TAKEOVER_2
  if var VAR_SCENE_ROCKET_TAKEOVER == 0 goto Quiet
  if var VAR_SCENE_ROCKET_TAKEOVER == 1 goto Gathering
  if var VAR_SCENE_ROCKET_TAKEOVER == 3 goto Disguised
  if var VAR_SCENE_ROCKET_TAKEOVER == 5 goto Over
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_3
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_4
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_1
  end
label Quiet
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_2
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_3
  end
label Gathering
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_2
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_3
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_4
  end
label Disguised
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_3
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_4
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_1
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_5
  end
label Over
  clearflag FLAG_HIDE_ROCKET_TAKEOVER_2
  end


# T25_001: the woman who saw the man in black
script ManInBlack
  face t25_gswoman2_2 player
  speaker "Woman"
  if var VAR_SCENE_ROCKET_TAKEOVER >= 2 goto Real
  say "Did you see that man in black by the Radio Tower? He's dressed up as one of Team Rocket. What a strange costume!"
  end
label Real
  say "That man in black really was one of Team Rocket? And I thought it was a costume!"
  end


# T25_007: the man in black before the Radio Tower
script TowerGrunt
  face t25_rocketm player
  speaker "Man in Black"
  say "So this is the Radio Tower..."
  if var VAR_SCENE_ROCKET_TAKEOVER == 3 goto Uniform
  say "What are you gawping at, kid? Beat it!"
  face t25_rocketm up
  end
label Uniform
  say "Hey, you! What are you standing about for? Get to your post!"
  face t25_rocketm up
  end
