# Clay Tunnel's first floor (plan 22 · U10, part b). docs/scripts.md describes the language; the lines are our own, on Black 2's
# beats (the decompilation's res/field/scripts/clay_tunnel.s, read for the order alone).
#
# Three workers dig at a heap of rock that shuts the way on to the second floor. The one in charge (8) asks for a Pokémon that
# knows Rock Smash; with one, the heap is broken together, the heap on the second floor's side with it, and the workers go home
# (FLAG_UNOVA_CLAY_TUNNEL_DUG). The workers by the mining carts (16 to 18) ride the player to another floor.
script Diggers
  face self player
  speaker "Worker"
  say "Phew! We've dug and dug at this heap all week, and it won't budge... Hang on. Is one of your Pokémon good at Rock Smash?"
  if knows "Rock Smash" goto Able
  if not granted "Rock Smash" goto Cannot
label Able
  say "It is? Then let's all have a go at once! Ready... smash!"
  usemove "Rock Smash"
  fade out 0.4
  setflag FLAG_UNOVA_CLAY_TUNNEL_DUG
  fade in
  speaker "Worker"
  say "We did it! The way through to the next floor is open at last. That's the end of the job. I'll ring the cart man and head home!"
  end
label Cannot
  say "No? Ah well. We'll keep at it with our picks, then..."
  end


script Digging
  speaker "Worker"
  say "Chip, chip, chip! Chip, chip, chip! This heap is tougher than it looks!"
  end


script DugRocks
  speaker none
  text "A heap of broken rock fills the way."
  end


script CartToSecond
  call CartQuestion
  if no end
  call CartRide
  warp "ClayTunnel_2" 27 14 left
  end


script CartToThirdEast
  call CartQuestion
  if no end
  call CartRide
  warp "ClayTunnel_3" 56 84 down
  end


script CartToThirdWest
  call CartQuestion
  if no end
  call CartRide
  warp "ClayTunnel_3" 19 35 down
  end


script CartQuestion
  face self player
  speaker "Worker"
  ask "You there! Fancy a ride in the mining cart?"
  if yes return
  say "Whenever you like, just come and ask me."
  return


script CartRide
  speaker "Worker"
  say "Hop in and hold on tight!"
  return

# The holes of this floor a boulder pushed with Strength fills (Black 2's StrengthHole tiles, 2 by 2; plan 22 · U10, part b), as the
# Relic Passage's are: each pit's script lets the boulder fall in and the hole is floor from then on
script FillHole1
  call BoulderFallsIn
  setflag FLAG_UNOVA_CLAY_TUNNEL_HOLE_1
  hide self
  end

script FillHole2
  call BoulderFallsIn
  setflag FLAG_UNOVA_CLAY_TUNNEL_HOLE_2
  hide self
  end
