# The Villa in the Resort Area (plan 06 · R16b), rebuilt to the original's plan. docs/scripts.md describes the
# language; the lines are our own, on the original's beats (scripts_villa.s).
#
# VAR_VILLA_STATE: 0 before the player has been inside, 1 while the boy who gave the Villa away waits for the free
# table to be ordered (he won't let the player leave: DontGoYet), 2 from the table's delivery on. Furniture is
# ordered from the catalogue on the table (OrderForm, on the table's tiles; villa catalogue, villa buy), paid for,
# and delivered at once: villa furnish stands it in its own place. Each piece owned is a flag
# (FLAG_VILLA_FURNITURE_...), and looking at one says what it is (Look...).
#
# The day's visitor (VAR_RESORT_VILLA_VISITOR, drawn each new day by VillaRules.NewDay; 255 for nobody) is shown as
# the player comes in (OnEnter, the original's OnTransition): the rival, the assistant, Mom, Volkner, Flint with
# Volkner and Maylene with Candice are found inside; the professor, Roark, Gardenia, Maylene and Byron wait at the
# door outside and come in when asked (FLAG_VILLA_VISITOR_INSIDE; resort_area.txt); Fantina, Crasher Wake, Candice
# and Cynthia walk in once the player has stepped up to the table (the triggers on 10,7, by
# VAR_MAP_LOCAL_VILLA_CALLER; FLAG_VILLA_VISITOR_OUTSIDE then keeps them where they are). Each says one of five
# things, by VAR_RESORT_VILLA_VISITOR_MESSAGE_NUM. Five pieces bring someone round the moment they are delivered:
# the table the rival, the small sofa the assistant, the bookshelf the professor, the houseplants Gardenia, the
# piano Cynthia.


script OnEnter
  if var VAR_VILLA_STATE == 0 goto First
  if flag FLAG_VILLA_VISITOR_OUTSIDE end
  call SendEveryoneHome
  setvar VAR_MAP_LOCAL_VILLA_CALLER 0
  if var VAR_RESORT_VILLA_VISITOR == 255 end
  if var VAR_RESORT_VILLA_VISITOR == 6 setvar VAR_MAP_LOCAL_VILLA_CALLER 6
  if var VAR_RESORT_VILLA_VISITOR == 8 setvar VAR_MAP_LOCAL_VILLA_CALLER 8
  if var VAR_RESORT_VILLA_VISITOR == 10 setvar VAR_MAP_LOCAL_VILLA_CALLER 10
  if var VAR_RESORT_VILLA_VISITOR == 12 setvar VAR_MAP_LOCAL_VILLA_CALLER 12
  if var VAR_RESORT_VILLA_VISITOR == 0 clearflag FLAG_HIDE_VILLA_RIVAL
  if var VAR_RESORT_VILLA_VISITOR == 1 clearflag FLAG_HIDE_VILLA_COUNTERPART
  if var VAR_RESORT_VILLA_VISITOR == 3 clearflag FLAG_HIDE_VILLA_MOM
  if var VAR_RESORT_VILLA_VISITOR == 11 clearflag FLAG_HIDE_VILLA_VOLKNER
  if var VAR_RESORT_VILLA_VISITOR == 13 goto FlintAndVolkner
  if var VAR_RESORT_VILLA_VISITOR == 14 goto MayleneAndCandice
  if not flag FLAG_VILLA_VISITOR_INSIDE end
  if var VAR_RESORT_VILLA_VISITOR == 2 clearflag FLAG_HIDE_VILLA_PROF_ROWAN
  if var VAR_RESORT_VILLA_VISITOR == 4 clearflag FLAG_HIDE_VILLA_ROARK
  if var VAR_RESORT_VILLA_VISITOR == 5 clearflag FLAG_HIDE_VILLA_GARDENIA
  if var VAR_RESORT_VILLA_VISITOR == 7 clearflag FLAG_HIDE_VILLA_MAYLENE
  if var VAR_RESORT_VILLA_VISITOR == 9 clearflag FLAG_HIDE_VILLA_BYRON
  end
label FlintAndVolkner
  clearflag FLAG_HIDE_VILLA_FLINT
  clearflag FLAG_HIDE_VILLA_VOLKNER
  end
label MayleneAndCandice
  clearflag FLAG_HIDE_VILLA_MAYLENE
  clearflag FLAG_HIDE_VILLA_CANDICE
  place maylene 6 8 right
  place candice 7 8 left
  end
label First
  # The first time in: the boy who gave the Villa away is waiting, and a new game's flags are set here
  setflag FLAG_HIDE_VILLA_DELIVERY_GUY
  call SendEveryoneHome
  setvar VAR_RESORT_AREA_STATE 1
  setvar VAR_RESORT_VILLA_VISITOR 255
  setvar VAR_VILLA_STATE 1
  speaker "Boy"
  say "Well, here it is. Yours now, every board of it!" "...Ah. I may have sold the furniture. All of it. I needed the money, you see."
  say "But a house needs a table, so the first one's on me. The catalogue's right here."
  call OfferTable
  end


script SendEveryoneHome
  setflag FLAG_HIDE_VILLA_RIVAL
  setflag FLAG_HIDE_VILLA_COUNTERPART
  setflag FLAG_HIDE_VILLA_PROF_ROWAN
  setflag FLAG_HIDE_VILLA_MOM
  setflag FLAG_HIDE_VILLA_ROARK

  setflag FLAG_HIDE_VILLA_FANTINA
  setflag FLAG_HIDE_VILLA_MAYLENE
  setflag FLAG_HIDE_VILLA_CRASHER_WAKE
  setflag FLAG_HIDE_VILLA_BYRON

  return


script OfferTable
  villa choose "Table"
  ask "Order the Table? It's free."
  if no goto NotYet
  villa buy
  call Deliver
  speaker "Boy"
  say "There. A table! It's a home already." "That's the last of it off my hands. Do look after the place, won't you?"
  fade out 0.4
  hide school_kid_m
  setflag FLAG_HIDE_VILLA_SCHOOL_KID_M
  setvar VAR_VILLA_STATE 2
  fade in 0.4
  call RivalDropsIn
  return
label NotYet
  speaker "Boy"
  say "Go on, order the table. I can't hand over an empty house and sleep at night."
  return


script SchoolKid
  call OfferTable
  end


script DontGoYet
  face school_kid_m down
  say "Wait, don't go yet! Order the table first. Please."
  walk player up 1
  end


# ---- the catalogue

script OrderForm
  text "{player} opened the furniture catalogue. Money: ${money}."
  villa catalogue
  if result == 0 end
  if result == 2 goto Sold
  ask "Order the {thing} for ${price}?"
  if no end
  villa buy
  if result == 0 goto Poor
  call Deliver
  villa is "Small Sofa"
  if result == 1 goto Sofa
  villa is "Bookshelf"
  if result == 1 goto Shelf
  villa is "Houseplant"
  if result == 1 goto Plant
  villa is "Piano"
  if result == 1 goto Piano
  end
label Sofa
  call CounterpartDropsIn
  end
label Shelf
  call RowanDropsIn
  end
label Plant
  call GardeniaDropsIn
  end
label Piano
  call CynthiaDropsIn
  end
label Sold
  text "The {thing} has been bought already. The page is stamped: sold out."
  end
label Poor
  text "There isn't enough money for the {thing}."
  end


script Deliver
  fade out 0.4
  villa furnish
  place player 10 6 down
  clearflag FLAG_HIDE_VILLA_DELIVERY_GUY
  place delivery_guy 10 8 up
  show delivery_guy
  fade in 0.4
  speaker "Delivery Man"
  villa owned
  if result == 20 goto Last
  say "Delivery! One {thing}, put right where it belongs. Thank you for your order!"
  goto Leaves
label Last
  say "Delivery! One {thing}... and that's the whole catalogue. You've bought us out!"
label Leaves
  fade out 0.4
  hide delivery_guy
  setflag FLAG_HIDE_VILLA_DELIVERY_GUY
  fade in 0.4
  return

script RivalDropsIn
  call SendEveryoneHome
  clearflag FLAG_HIDE_VILLA_RIVAL
  place rival 11 11 up
  show rival
  sound "stairs"
  walk rival left 1 up 3
  face player down
  setflag FLAG_VILLA_VISITOR_INSIDE
  setvar VAR_RESORT_VILLA_VISITOR 255
  speaker "{rival}"
  say "{player}! You got yourself a whole Villa?! No fair!" "...One table. That's it? OK, I'm fining you a million if this place isn't full the next time I come by!"
  walk rival down 3 right 1
  sound "stairs"
  hide rival
  setflag FLAG_HIDE_VILLA_RIVAL
  return
