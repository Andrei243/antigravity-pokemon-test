# Bill's Sea Cottage on Route 25 (plan 19 · K4). docs/scripts.md describes the language; the lines are our own, on
# FireRed's beats (data/maps/Route25_SeaCottage/scripts.inc).
#
# Bill has mixed himself up with a Clefairy in his own teleporter. Spoken to, the Clefairy asks the player to run the
# cell separator on the PC and walks into the machine (FLAG_MAP_LOCAL_BILL_IN_TELEPORTER, a flag of the place, as
# FireRed's FLAG_TEMP_2 is); the PC then brings Bill out as himself (FLAG_HELPED_BILL_IN_SEA_COTTAGE, and the human
# Bill no longer hidden by FLAG_HIDE_BILL_HUMAN_SEA_COTTAGE). He gives the S.S. Ticket (FLAG_GOT_SS_TICKET), which
# takes the Rocket recruiter off Nugget Bridge.


# FireRed's OnTransition: the Clefairy is there until Bill has been helped.
script OnEnter
  if flag FLAG_HELPED_BILL_IN_SEA_COTTAGE end
  clearflag FLAG_HIDE_BILL_CLEFAIRY
  # Bill as himself is hidden until then, as a new game hides him (and a save from before plan 19 · K4)
  setflag FLAG_HIDE_BILL_HUMAN_SEA_COTTAGE
  end


script Bill
  if flag FLAG_GOT_SS_TICKET goto Party
  if flag FLAG_HELPED_BILL_IN_SEA_COTTAGE goto Ticket
  face bill_clefairy player
  speaker "Bill"
  scene "Bill in a muddle"
  say "Hiya! I'm a Pokémon... No, wait! I'm Bill, the Pokémaniac! Honestly, I'm a person!"
  say "I was testing my teleporter and got mixed up with a Clefairy. Can you help me out of this?"
  ask "Please? You'll help me, won't you?"
  if yes goto Plan
  say "Aw, come on, don't leave me like this! Please?"
label Plan
  say "Thanks! When I step into the teleporter, run the cell separator on the PC over there. That'll sort us out."
  fade out
  hide bill_clefairy
  setflag FLAG_HIDE_BILL_CLEFAIRY
  setflag FLAG_MAP_LOCAL_BILL_IN_TELEPORTER
  fade in
  endscene
  end
label Ticket
  face bill_human player
  speaker "Bill"
  scene "The ticket"
  say "Phew, thanks a million! I'm myself again. I owe you one."
  say "Here, take this. It's a ticket for the S.S. Anne's party. I can't make it, so you should go instead."
  give "S.S. Ticket"
  setflag FLAG_GOT_SS_TICKET
  setflag FLAG_HIDE_NUGGET_BRIDGE_ROCKET
  say "The ship's in Vermilion City harbour. Bring your Pokémon: there'll be trainers from all over the world aboard!"
  endscene
  end
label Party
  face bill_human player
  speaker "Bill"
  say "Have fun on the S.S. Anne! I hear the captain's a real Pokémon lover."
  end


# The PC beside the teleporters.
script Computer
  if flag FLAG_GOT_SS_TICKET goto List
  if flag FLAG_MAP_LOCAL_BILL_IN_TELEPORTER goto Separate
  text "The screen shows a diagram of the teleporter, with a button marked CELL SEPARATOR."
  end
label Separate
  scene "The cell separator"
  text "{player} started the teleporter's cell separator!"
  clearflag FLAG_MAP_LOCAL_BILL_IN_TELEPORTER
  setflag FLAG_HELPED_BILL_IN_SEA_COTTAGE
  sound "select"
  wait 0.4
  sound "select"
  wait 0.4
  camera shake 0.5
  fade out
  clearflag FLAG_HIDE_BILL_HUMAN_SEA_COTTAGE
  show bill_human
  fade in
  fanfare levelup
  endscene
  end
# Bill's list of his favourite Pokémon, a picture of each (plan 19 · K4, part b; FireRed's
# MULTICHOICE_EEVEELUTIONS, each choice showing a picture): the picture told in words, and the Pokédex told it was seen.
label List
  choose "Bill's favourite Pokémon. Which picture will you look at?" "Eevee" "Flareon" "Jolteon" "Vaporeon" "Log off"
  if result == 1 goto Flareon
  if result == 2 goto Jolteon
  if result == 3 goto Vaporeon
  if result == 4 end
  seen "Eevee"
  text "Eevee: a little brown Pokémon with a fluffy collar. Its genes let it evolve in many different ways."
  end
label Flareon
  seen "Flareon"
  text "Flareon: an Eevee touched by a Fire Stone, its mane a warm, fiery yellow."
  end
label Jolteon
  seen "Jolteon"
  text "Jolteon: an Eevee touched by a Thunder Stone, its fur standing up in sharp spikes."
  end
label Vaporeon
  seen "Vaporeon"
  text "Vaporeon: an Eevee touched by a Water Stone, with fins and a tail like a mermaid's."
  end
