# Veilstone's Department Store, fourth floor (plan 01 · M11, part d), on the original's plan. docs/scripts.md
# describes the language; the lines are our own, on the original's beats (scripts_veilstone_store_4f.s).
#
# Two clerks sell things for a Secret Base, for money (the original's PokeMartDecor with MART_DECOR_ID_VEILSTONE_4F_UP
# and _DOWN: five furnishings and six dolls, at the prices of the Underground's table of goods, underground.json's
# "money"). What is bought is sent to the base's PC (`underground give good`), which holds two hundred. A question
# takes six answers, so the dolls are asked for in two. The attendant recommends the first of her four, as on the
# floors below (VeilstoneStore2F.txt).


script OnEnter
  setvar VAR_VEILSTONE_STORE_LIFT_FLOOR 5
  call VeilstoneStoreElevator.Stands
  end


# VeilstoneStore4F_Attendant
script Attendant
  choose "Hello. May I help you with anything?" "WHAT'S GOOD?" "NOTHING, THANKS"
  if result == 0 goto Cushion
  say "Please enjoy your visit."
  end
label Cushion
  say "The Yellow Cushion is a great favourite: neither too soft nor too firm."
  end


# VeilstoneStore4F_TopDecorVendor
script Goods
  say "Welcome! Furnishings for a Secret Base, sent straight to its PC."
  underground room
  if result == 0 goto Full
  choose "Which would you like?" "YELLOW CUSHION  500" "CUPBOARD  1,000" "TV  4,500" "REFRIGERATOR  1,000" "PRETTY SINK  3,000" "NOTHING"
  if result == 0 goto Cushion
  if result == 1 goto Cupboard
  if result == 2 goto Tv
  if result == 3 goto Fridge
  if result == 4 goto Sink
  say "Please come again!"
  end
label Cushion
  takemoney 500
  if result == 0 goto Poor
  underground give good "Yellow Cushion"
  goto Sold
label Cupboard
  takemoney 1000
  if result == 0 goto Poor
  underground give good "Cupboard"
  goto Sold
label Tv
  takemoney 4500
  if result == 0 goto Poor
  underground give good "TV"
  goto Sold
label Fridge
  takemoney 1000
  if result == 0 goto Poor
  underground give good "Refrigerator"
  goto Sold
label Sink
  takemoney 3000
  if result == 0 goto Poor
  underground give good "Pretty Sink"
label Sold
  sound "select"
  say "Thank you! It is on its way to the PC of your Secret Base."
  end
label Poor
  say "I'm so sorry, but you haven't the money for that."
  end
label Full
  say "I'm so sorry. The PC of your Secret Base has no room for anything more."
  end


# VeilstoneStore4F_BottomDecorVendor
