# The Pewter Museum of Science in HeartGold's Kanto (plan 20, part q). docs/scripts.md describes the language; the lines
# are our own, on HeartGold's beats (scr_seq_T03R0101).
#
# Steven's visits (plan 20, parts q and t; the rulings' "Johto's last leftovers, part t"): once he has given his gift in
# Silph Co.'s lobby (FLAG_JOHTO_GOT_HOENN_STARTER, ours, saffron_silph_co_hq.txt) he stands before the space wing's stone
# in HeartGold's first place. Spoken to, he shows the player the Enigma Stone the museum lent him (a stone he holds, not
# an item of the bag: HeartGold's was a gift of an event of its own) and goes out to fetch the friend who travels with
# him (FLAG_JOHTO_STEVEN_SHOWED_STONE, ours), who waits in Pewter City (pewter.txt). Once the player has met it
# (FLAG_JOHTO_MET_STEVENS_EON, ours) he stands in HeartGold's second place, by the fossils, for good: HeartGold's
# FLAG_HIDE_STEVEN_IN_HOUSE_AFTER_LATIS, laid here. The first place's own flag stays ours,
# FLAG_JOHTO_HIDE_PEWTER_MUSEUM_STEVEN, since HeartGold's FLAG_HIDE_STEVEN_IN_HOUSE_BEFORE_LATIS waits on its event.


script OnEnter
  setflag FLAG_JOHTO_HIDE_PEWTER_MUSEUM_STEVEN
  setflag FLAG_HIDE_STEVEN_IN_HOUSE_AFTER_LATIS
  if not flag FLAG_JOHTO_GOT_HOENN_STARTER end
  if flag FLAG_JOHTO_MET_STEVENS_EON goto After
  if flag FLAG_JOHTO_STEVEN_SHOWED_STONE end
  clearflag FLAG_JOHTO_HIDE_PEWTER_MUSEUM_STEVEN
  end
label After
  clearflag FLAG_HIDE_STEVEN_IN_HOUSE_AFTER_LATIS
  end


# Steven before the space wing (ours): the Enigma Stone shown, and his friend fetched
script Steven
  face t03r0101_daigo player
  speaker "Steven"
  say "{player}! Good timing. The museum let me hold one of its stones for a moment. Come and look."
  speaker none
  text "Steven held out a smooth, pale stone. Deep inside it, something like an eye seemed to glint."
  speaker "Steven"
  say "They call it the Enigma Stone. Nobody knows where it came from. But my travelling friend felt it from far away."
  say "My friend is a Pokémon of the skies over Hoenn. I think it would like to meet you, too."
  say "I'll go and fetch it. Meet us outside, by the houses at the north of the city."
  setflag FLAG_JOHTO_STEVEN_SHOWED_STONE
  setflag FLAG_JOHTO_HIDE_PEWTER_MUSEUM_STEVEN
  hide self
  end


# Steven by the fossils, after his friend has met the player (ours)
script StevenAfter
  face t03r0101_daigo_2 player
  speaker "Steven"
  random 2
  if result == 1 goto Two
  say "I gave the Enigma Stone back to the museum. Some stones are meant to be looked at, not kept."
  say "Old bones and fallen stars under one roof. I could spend a whole day in here."
  end
label Two
  say "Hoenn's skies are wide, {player}. If you ever cross the sea that way, look up now and then."
  end


# Plan 40 · GE3, our words on HeartGold's beats: the museum restores fossils as HeartGold's does (the Helix and Dome
# Fossils the Ruins of Alph's researchers dig up, ruins_of_alph_research_center.txt, and the Old Amber of the museum's own
# back room, which its other scientist hands over here, once: FLAG_JOHTO_GOT_OLD_AMBER, ours). The scientist takes a
# Fossil and has the Pokémon ready once the player has gone out and come back, as Oreburgh's and Cinnabar's do (the
# game's own Fossils, VAR_REVIVED_POKEMON_SPECIES, and a local flag that lasts while the player stays). HeartGold hands
# the Pokémon over at level 20, and so does this museum.
script Curator
  speaker "Scientist"
  say "Welcome to the museum. Mind the bones: some of them are older than the mountains outside."
  if flag FLAG_JOHTO_GOT_OLD_AMBER end
  say "Oh, you look like a Trainer who travels. We found a lump of amber in the back room, more than we can study."
  give "Old Amber"
  setflag FLAG_JOHTO_GOT_OLD_AMBER
  say "There's something caught inside it, from long, long ago. My colleague over there can tell you more."
  end


script FossilScientist
  speaker "Scientist"
  if flag FLAG_MAP_LOCAL_PEWTER_MUSEUM_REVIVING goto Working
  fossil ready
  if result == 1 goto Ready
  say "We study how to bring Pokémon back from fossils here. And at last, we've managed it!"
  fossil count
  if result == 0 goto None
  ask "You've brought a Fossil, haven't you? Shall I bring it back to life for you?"
  if no goto Declined
  fossil choose
  if result == 0 goto Declined
  fossil revive
  if result == 0 goto Declined
  say "Leave it with me. The machine needs a little while."
label Working
  setflag FLAG_MAP_LOCAL_PEWTER_MUSEUM_REVIVING
  say "Take a walk round Pewter City, then come back. It'll be ready by then."
  end
label Ready
  say "There you are! Your Fossil came back to life as {thing}!"
  if not party < 6 goto Full
  fossil give
  fanfare pokemon
  text "{player} received {thing}!"
  say "It's been asleep in that stone for longer than anyone can say. Show it the world."
  end
label Full
  say "But your team is full! Make room, and come back for it."
  end
label None
  say "If you ever find a Fossil, bring it here. I'll show you what it used to be!"
  end
label Declined
  say "No? Well, the machine will be here if you change your mind."
  end
