# beats (data/maps/SevenIsland/scripts.inc).

# The trainer by the Pokémon Center teaches Swords Dance once, for nothing (`tutor once`; Models/MoveTutors.cs, Kanto's
# table of who may learn it, our own). FLAG_TUTOR_SWORDS_DANCE, FireRed's name: he has taught it.


script Tutor
  speaker "Cooltrainer"
  if flag FLAG_TUTOR_SWORDS_DANCE goto Taught
  ask "A battle can turn on a single dance. Shall I teach one of your Pokémon Swords Dance?"
  if no goto Later
  say "Good. Which Pokémon is to learn it?"
  choosepokemon
  tutor once "Swords Dance"
  if result == 1 goto Later
  if result == 2 goto Egg
  if result == 3 goto Cannot
  if result == 4 goto Knows
  tutor learn
  if result == 0 goto Later
  if result == 2 goto Hm
  if result == 3 say "{wearer} forgot {forgotten}. And..."
  fanfare levelup
  say "{wearer} learned {thing}!"
  setflag FLAG_TUTOR_SWORDS_DANCE
label Taught
  say "Swords Dance sharpens a Pokémon's Attack a great deal. Dance first, then strike!"
  end
label Later
  say "Come back if you want to learn. The dance will wait."
  end
label Egg
  say "An Egg can't dance! Hatch it first."
  end
label Cannot
  say "That one can't learn Swords Dance, I'm afraid."
  end
label Knows
  say "That one already knows {thing}!"
  end
label Hm
  say "{forgotten} isn't a move that can be given up like that."
  goto Later
