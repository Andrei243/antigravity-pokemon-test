# Reversal Mountain's lair (plan 22 · U7, part b). docs/scripts.md describes the language; the lines are our own, on Black 2's
# beats (the decompilation's res/field/scripts/reversal_mountain_5.s, read for the order alone). VAR_UNOVA_REVERSAL_LAIR is the
# errand's count (reversal_mountain_4.txt has it whole): 1 as the player comes up to the lair with Bianca along, 2 once she has
# had her say.
#
# Black 2's Heatran waits in this room for the Magma Stone and the post-game (plan 22 · U10). Its person (0) stays hidden
# (FLAG_UNOVA_HIDE_HEATRAN) until then.


script OnEnter
  call UnovaCatchUpB
  call UnovaBiancaWaits
  end


# The passage up to the lair (our trigger across it, 9 to 11, 18, at 1)
script BiancaInTheLair
  if var VAR_UNOVA_REVERSAL != 1 end
  scene "Bianca in the lair"
  face bianca_along player
  speaker "Bianca"
  say "What is this place? It feels very strange."
  say "Could this be where Reversal Mountain started from? The lair of the Pokémon Heatran?"
  say "They say Heatran is a Pokémon with magma-like blood flowing through it! Just imagine!"
  setvar VAR_UNOVA_REVERSAL_LAIR 2
  endscene
  end


# The lair's heart (read at 10, 12; plan 22 · U10, part a). Once Unova's Hall of Fame is entered the Magma Stone lies on the ring
# path (an item's ball, magma_stone); brought to the heart it stirs, and set down (Black 2's question, reversal_mountain_5.s) it is
# gone and Heatran (0) falls into the lair (FLAG_UNOVA_MAGMA_STONE_SET). Heatran is battled at Black 2's level 68; caught or beaten
# it is gone for good (FLAG_UNOVA_FOUGHT_HEATRAN), run from, it is back the next time the lair is come to.
script MagmaStone
  if flag FLAG_UNOVA_MAGMA_STONE_SET goto Warm
  if not item "Magma Stone" goto Warm
  speaker none
  text "The Magma Stone is glowing hot, as if something here calls to it..."
  ask "Will you set it down here?"
  if no end
  take "Magma Stone"
  setflag FLAG_UNOVA_MAGMA_STONE_SET
  cry "Heatran"
  text "A deep rumble rose from below...\nSomething heavy dropped into the lair from above!"
  show 0
  end
label Warm
  speaker none
  text "The rock here is warm to the touch."
  end


script Heatran
  if not flag FLAG_UNOVA_MAGMA_STONE_SET end
  face self player
  cry "Heatran"
  speaker none
  text "Heatran let out a grinding roar!\nThe air of the lair shimmers with heat!"
  wildbattle "Heatran" 68
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_HEATRAN
  hide self
  text "The lair is cooling..."
  end
label Fled
  hide self
  text "Heatran sank into the depths of the mountain..."
  end


# The Magma Stone's ball (magma_stone), on the ring path once Unova's Hall of Fame is entered (shown by UnovaHallOfFame): an
# item's ball as any other, with a script of its own because it is there only after the Hall of Fame
script MagmaStoneBall
  setflag own
  find own
  end
