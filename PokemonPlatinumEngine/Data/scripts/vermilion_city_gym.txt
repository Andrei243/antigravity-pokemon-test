# The Vermilion City Gym (plan 19 · K4, part b). docs/scripts.md describes the language; the lines are our own, on
# FireRed's beats (data/maps/VermilionCity_Gym/scripts.inc).
#
# Lt. Surge stands behind two locked electric gates. Two of the fifteen bins in the hall hide their switches: the
# first is any bin, drawn on the game's own chance each time the Gym is come into (FireRed's SetVermilionTrashCans,
# its first switch at random), the second a bin beside it, north, south, east or west, drawn the same way. The first
# found opens the first lock (FLAG_MAP_LOCAL_VERMILION_GYM_FIRST_SWITCH, a flag of the place, as FireRed's FLAG_TEMP_1
# is); a wrong bin after it resets both locks and draws the switches again; the second opens the gates for good
# (FLAG_FOUND_BOTH_VERMILION_GYM_SWITCHES, which hides the gates' beams, overlays/vermilion_city_gym.json).
# VAR_VERMILION_GYM_SWITCH_1 and _2 are the bins of the switches (1 to 15, five to a row, west to east, north to
# south, as FireRed numbers them), VAR_VERMILION_GYM_CAN the bin last looked into.
#
# Lt. Surge battles for the Thunder Badge (FLAG_BADGE03_GET is the badge itself, `givebadge thunder`, which opens Fly
# in the field), counts his three trainers beaten (FireRed's set_gym_trainers 3) and gives TM34, Shock Wave. For a
# trainer who never had Flash from Route 2's aide, Oak's other aide waits in Vermilion City after (vermilion_city.txt).


# FireRed's OnTransition: the switches are drawn again each time the Gym is come into, until both are found.
script OnEnter
  if flag FLAG_FOUND_BOTH_VERMILION_GYM_SWITCHES end
  call DrawSwitches
  end


# The first switch on any of the fifteen bins, the second on a bin beside it.
script DrawSwitches

  addvar VAR_VERMILION_GYM_SWITCH_1 1
  setvar VAR_VERMILION_GYM_SWITCH_2 VAR_VERMILION_GYM_SWITCH_1

  if var VAR_VERMILION_GYM_SWITCH_1 == 2 goto Beside2
  if var VAR_VERMILION_GYM_SWITCH_1 == 3 goto Beside3
  if var VAR_VERMILION_GYM_SWITCH_1 == 4 goto Beside4
  if var VAR_VERMILION_GYM_SWITCH_1 == 5 goto Beside5
  if var VAR_VERMILION_GYM_SWITCH_1 == 6 goto Beside6
  if var VAR_VERMILION_GYM_SWITCH_1 == 7 goto Beside7
  if var VAR_VERMILION_GYM_SWITCH_1 == 8 goto Beside8
  if var VAR_VERMILION_GYM_SWITCH_1 == 9 goto Beside9
  if var VAR_VERMILION_GYM_SWITCH_1 == 10 goto Beside10
  if var VAR_VERMILION_GYM_SWITCH_1 == 11 goto Beside11
  if var VAR_VERMILION_GYM_SWITCH_1 == 12 goto Beside12
  if var VAR_VERMILION_GYM_SWITCH_1 == 13 goto Beside13
  if var VAR_VERMILION_GYM_SWITCH_1 == 14 goto Beside14
  if var VAR_VERMILION_GYM_SWITCH_1 == 15 goto Beside15
  return
label Beside1

  if result == 0 setvar VAR_VERMILION_GYM_SWITCH_2 2
  if result == 1 setvar VAR_VERMILION_GYM_SWITCH_2 6
  return
label Beside2

# A bin looked into.
script Can
  if flag FLAG_FOUND_BOTH_VERMILION_GYM_SWITCHES goto Trash
  if flag FLAG_MAP_LOCAL_VERMILION_GYM_FIRST_SWITCH goto Second
  if var VAR_VERMILION_GYM_CAN == VAR_VERMILION_GYM_SWITCH_1 goto First
label Trash
  text "Nope! There's only rubbish in here."
  return
label First
  text "Hey! There's a switch under the rubbish! {player} pressed it.\nThe first electric lock opened!"
  sound "select"
  setflag FLAG_MAP_LOCAL_VERMILION_GYM_FIRST_SWITCH
  return
label Second
  if var VAR_VERMILION_GYM_CAN == VAR_VERMILION_GYM_SWITCH_2 goto Both
  text "Nope! There's only rubbish in here.\nHey! The electric locks were reset!"
  sound "error"
  clearflag FLAG_MAP_LOCAL_VERMILION_GYM_FIRST_SWITCH
  call DrawSwitches
  return
label Both
  text "There's another switch under the rubbish! {player} pressed it.\nThe second electric lock opened, and the gates are down!"
  sound "door_slide"
  setflag FLAG_FOUND_BOTH_VERMILION_GYM_SWITCHES
  return


script LtSurge
  face lt_surge player
  speaker "Lt. Surge"
  if badge thunder goto Won
  scene "Surge"
  say "Hey, kid! What do you think you're doing here? You won't last long in battle, not with your puny power!"
  say "I tell you, kid, electric Pokémon saved me during the war! Now let's see you survive a jolt like that!"
  battle self as "kanto_leader_lt_surge"
  say "Whoa! You're the real deal, kid! Fine then, take the Thunder Badge!"
  givebadge thunder
  text "{player} received the Thunder Badge from Lt. Surge!"
  say "The Thunder Badge lets your Pokémon use Fly outside of battle, once you have the move."
  setflag FLAG_DEFEATED_LT_SURGE
  defeat "kanto_engineer_baily"
  defeat "kanto_sailor_dwayne"
  defeat "kanto_gentleman_tucker"
  if not flag FLAG_GOT_HM05 clearflag FLAG_HIDE_VERMILION_CITY_OAKS_AIDE
label Tm
  say "And here, take this as well. It's my special move!"
  give "TM34"
  setflag FLAG_GOT_TM34_FROM_SURGE
  say "TM34 is Shock Wave. It strikes so fast that it never misses. Electric, like me!"
  endscene
  end
label Won
  if not flag FLAG_GOT_TM34_FROM_SURGE goto Tm
  say "A little word of advice, kid. Electricity is sure powerful, but it's useless against Ground types!"
  end


# The guide by the door.
script GymGuy
  speaker "Gym Guide"
  if badge thunder goto Won
  say "Yo! Champ in the making! Lt. Surge has locked himself in behind electric gates."
  say "The switches are hidden in the bins round the hall. Find the first, and the second is always right beside it!"
  say "His Pokémon are Electric types, so bring a Ground type. Electric moves can't touch them!"
  end
label Won
  say "Whew! That was an electrifying battle! You really showed Lt. Surge."
  end
