script BerrySearcher
  if poketchapp BerrySearcher goto Water
  ask "Are you the forgetful sort, who plants berries and then can't remember where?"
  if no goto Fine
  say "I knew it! So am I! We're two of a kind, you and me. You can have this app for your Pokétch!"
  if not poketch goto NoWatch
  poketchapp BerrySearcher
  fanfare item
  text "{player} received the Berry Searcher for the Pokétch!"
  say "The Berry Searcher shows you where berries are ripe and waiting to be picked."
  end
label NoWatch
  say "Oh... You haven't got a Pokétch to put it on. Come back when you have one!"
  end
label Fine
  say "Oh, is that so. Then you'll manage perfectly well without this lovely thing."
  end
label Water
  say "Some kinds of berries want lots and lots of water."
  end


script Mulch
  choose "Um... Would you like some mulch? It's good for growing berries. You have ${money}." "Growth Mulch: $200" "Damp Mulch: $200" "Stable Mulch: $200" "Gooey Mulch: $200" "No, thanks"
  if result == 0 goto Growth
  if result == 1 goto Damp
  if result == 2 goto Stable
  if result == 3 goto Gooey
  say "Um... Please come again."
  end
label Growth
  takemoney 200
  if result == 0 goto Short
  give "Growth Mulch"
  goto Made
label Damp
  takemoney 200
  if result == 0 goto Short
  give "Damp Mulch"
  goto Made
label Stable
  takemoney 200
  if result == 0 goto Short
  give "Stable Mulch"
  goto Made
label Gooey
  takemoney 200
  if result == 0 goto Short
  give "Gooey Mulch"
label Made
  say "I make it myself, from the best soil and, um... what Pokémon leave behind. You know." "It helps berries grow up healthy."
  end
label Short
  say "Um... You don't seem to have enough money..."
  end


script Book
  sayown
  choose "Which mulch would you like to read about?" "Growth Mulch" "Damp Mulch" "Stable Mulch" "Gooey Mulch" "Close the book"
  if result == 0 goto Growth
  if result == 1 goto Damp
  if result == 2 goto Stable
  if result == 3 goto Gooey
  end
label Growth
  text "Growth Mulch makes berries grow faster. The soil dries out faster too, so water it more often."
  end
label Damp
  text "Damp Mulch keeps the soil from drying out. The berries grow more slowly for it."
  end
label Stable
  text "Stable Mulch keeps ripe berries on the plant for longer. Some growers lay it where they seldom pass."
  end
label Gooey
  text "Gooey Mulch lets a plant come up again more times from where the old one withered."
  end
