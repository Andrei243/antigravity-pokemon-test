# The Pokégear's calls (plan 20 · J3). docs/scripts.md describes the language; the lines are our own, on the
# subjects of HeartGold's calls (src/application/pokegear/phone/scripts). A script is a contact of the phone book
# (Data/world/johto/phone.json) by its id, the first letter of each word up (prof_elm: ProfElm); a contact with no
# script of their own gets Nobody's. The game sets VAR_PHONE_CALL before a call plays (Models/Pokegear.cs): 0 the
# player rang them, 1 they rang to talk, 2 they rang or were rung on their day and hour and now want a rematch, and
# wait on their map for it, 3 a rematch they asked for is still waiting.


# Someone with nothing of their own to say
script Nobody
  text "…There's no answer. Maybe they're busy."
  end


# Mom: news from home, and how the player's money is doing
script Mother
  speaker "Mom"
  if var VAR_PHONE_CALL == 1 goto Rang
  random 3
  if result == 1 goto Two
  if result == 2 goto Three
  say "Hello, dear! It's so good to hear from you. Are you eating properly out there?"
  say "Don't push yourself too hard. Come home and rest whenever you need to."
  call MomShops
  end
label Two
  say "Oh, {player}! The house is so quiet without you. Professor Elm stops by now and then to ask how you're doing."
  call MomShops
  end
label Three
  say "You sound well! Keep your Pokémon close and you'll be fine. I'm proud of you, you know."
  call MomShops
  end
label Rang
  say "{player}? It's Mom. I just wanted to hear your voice. Take care of yourself, all right?"
  call MomShops
  end

# Mom's buys with the savings (plan 43 · LF7): now and then, once she holds enough, a thing for the bag or a piece for
# the home, and the call says what. Her savings are short by what it cost.
script MomShops
  if var VAR_JOHTO_MOM_SAVINGS < 1200 goto Done
  random 4
  if result == 1 goto Doll
  if result == 2 goto Repel
  if result == 3 goto Done
  addvar VAR_JOHTO_MOM_SAVINGS -700
  give "Super Potion"
  say "Oh, and I used a little of your savings on a Super Potion. I sent it along to you!"
  return
label Doll
  addvar VAR_JOHTO_MOM_SAVINGS -1000
  openbase give "Pikachu Doll"
  say "Oh! I saw the sweetest Pikachu Doll and bought it with some of your savings. It's waiting in your decoration PC."
  return
label Repel
  addvar VAR_JOHTO_MOM_SAVINGS -350
  give "Repel"
  say "I bought you a Repel with some of your savings. Wild Pokémon can be such a bother!"
label Done
  return

# Joey, like the six callers after him with a gift in the phone book (Data/world/johto/phone.json's "gift"), now and
# then rings to say he has something for the player (one call in three that he makes to talk), and hands it over on
# his route once spoken to (route_30.txt): HeartGold's phone gifts, our words. FLAG_JOHTO_PHONE_GIFT_<ID> (ours)
# keeps a gift waiting; it comes again after it is collected.
script Joey
  speaker "Joey"
  if var VAR_PHONE_CALL == 2 goto Rematch
  if var VAR_PHONE_CALL == 3 goto Waiting
  if flag FLAG_JOHTO_PHONE_GIFT_JOEY goto GiftWaiting
  if var VAR_PHONE_CALL != 1 goto Chat
  random 3
  if result == 0 goto Gift
label Chat
  random 2
  if result == 1 goto Two
  say "Hey, it's Joey! My Ratchisel ate a whole berry by itself today. It's growing up so fast!"
  end
label Two
  say "Joey here! I was just thinking about our battle. My Ratchisel's been training non-stop since then."
  end
label Gift
  setflag FLAG_JOHTO_PHONE_GIFT_JOEY
  say "Joey here! I found something great, but my Ratchisel can't use it. Come and get it from me on Route 30!"
  end
label GiftWaiting
  say "It's Joey! Did you forget? I've still got that present for you. I'm on Route 30!"
  end
label Rematch
  say "It's Joey! My Ratchisel's in even better shape now. Come and battle me again on Route 30!"
  end
label Waiting
  say "Joey here. I'm still waiting for you on Route 30! My Ratchisel is getting restless."
  end
