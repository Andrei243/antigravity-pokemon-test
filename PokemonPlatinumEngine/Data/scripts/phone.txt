# The Pokégear's calls (plan 20 · J3). docs/scripts.md describes the language; the lines are our own, on the
# subjects of HeartGold's calls (src/application/pokegear/phone/scripts). A script is a contact of the phone book
# (Data/world/johto/phone.json) by its id, the first letter of each word up (prof_elm: ProfElm); a contact with no
# script of their own gets Nobody's. The game sets VAR_PHONE_CALL before a call plays (Models/Pokegear.cs): 0 the
# player rang them, 1 they rang to talk, 2 they rang or were rung on their day and hour and now want a rematch, and
# wait on their map for it, 3 a rematch they asked for is still waiting.


# Someone with nothing of their own to say
script Nobody
  text "…There's no answer. Maybe they're busy."
  end


# Mom: news from home, and how the player's money is doing
script Mother
  speaker "Mom"
  if var VAR_PHONE_CALL == 1 goto Rang
  random 3
  if result == 1 goto Two
  if result == 2 goto Three
  say "Hello, dear! It's so good to hear from you. Are you eating properly out there?"
  say "Don't push yourself too hard. Come home and rest whenever you need to."
  end
label Two
  say "Oh, {player}! The house is so quiet without you. Professor Elm stops by now and then to ask how you're doing."
  end
label Three
  say "You sound well! Keep your Pokémon close and you'll be fine. I'm proud of you, you know."
  end
label Rang
  say "{player}? It's Mom. I just wanted to hear your voice. Take care of yourself, all right?"
  end


# Professor Elm: his research, and the Egg

  end

  end

# Warden Baoba (plan 20 · J8, part b): his Safari Zone, and the test he has set the player (VAR_UNK_4057, the
# original's: safari_zone_entrance_interior.txt says what each value is). His first call is Olivine City's, as the
# player comes down from the lighthouse (olivine.txt), and his second the gate's, once a day has passed after the
# first test (safari_zone_gate.txt)
script Baoba
  speaker "Baoba"
  if var VAR_UNK_4057 >= 6 goto Owner
  if var VAR_UNK_4057 == 5 goto Sandshrew
  if var VAR_UNK_4057 == 4 goto Ready
  if var VAR_UNK_4057 == 3 goto Thinking
  if var VAR_UNK_4057 == 2 goto Geodude
  say "Baoba here! My Safari Zone is open. Go through the gate in the cliff by Cianwood City and follow the path."
  end
label Geodude
  say "Baoba here! Caught that Geodude in my Safari Zone yet? It's in the grass nearest the way in. You can do it!"
  end
label Thinking
  say "Ah, it's you! I'm still working out the second test. Give an old warden a little more time, will you?"
  end
label Ready
  say "Perfect timing! The second test is ready at last. Come to the Safari Zone and I'll tell you all about it."
  end
label Sandshrew
  say "Baoba here! Found a Sandshrew yet? Look in the Desert, the far corner of the zone. Good luck!"
  end
label Owner
  say "How's my Safari Zone's owner doing? I'm dreaming up new ways to play. I'll ring you the moment I have one!"
  end

# Brawly, Dewford's Gym Leader, registered with his Badge. Like every Gym Leader of Hoenn he asks a Champion of Hoenn for
# a rematch in his Gym on his day (plan 21 · H12, part g; Data/world/hoenn/phone.json, rematches.json): the call only
# sets the rematch waiting (VAR_PHONE_CALL 2, then 3 until it is fought), and the Gym's script battles the team.

  if var VAR_PHONE_CALL == 2 goto Rematch
  if var VAR_PHONE_CALL == 3 goto Waiting
  say "Yo, {player}! The waves were huge today. I wiped out three times, but that's how you learn!"
  end
label Rematch
  setflag FLAG_PHONE_REMATCH_BRAWLY
  say "Yo, {player}! I've been riding bigger waves than ever. Come to my Gym in Dewford and let's go again, two on two!"
  end

  end


# Wattson, Mauville's Gym Leader, registered with his Badge. A Champion of Hoenn is asked on his day for a rematch in his
# Gym (plan 21 · H12, part f), but only once he is back there after New Mauville (FLAG_WATTSON_REMATCH_AVAILABLE, as
# Emerald's): before that the call that would ask takes the waiting back and is only a talk.
script Wattson
  speaker "Wattson"
  if var VAR_PHONE_CALL == 2 goto Rematch
  if var VAR_PHONE_CALL == 3 goto Waiting
label Talk
  say "Wahahaha! {player}, is that you? I just rewired the whole Gym. The lights are brighter than ever!"
  end
label Rematch
  if not flag FLAG_WATTSON_REMATCH_AVAILABLE goto NotYet
  setflag FLAG_PHONE_REMATCH_WATTSON
  say "Wahahaha! {player}! My Pokémon have been charging up all week. Come to my Gym in Mauville and let's have a rematch!"
  end
label NotYet
  clearflag FLAG_PHONE_REMATCH_WATTSON
  goto Talk
label Waiting
  say "Wattson here! The Gym's lights are on and my Pokémon are buzzing. Don't keep us waiting too long, wahahaha!"
  end

# Kurt (plan 20 · J6): he rings when his Poké Balls are ready, and talks of his work when rung. `apricorn status` says
# whether an order is in hand: 0 none, 1 at work, 2 ready.
script Kurt
  speaker "Kurt"
  apricorn status
  if result == 2 goto Ready
  if result == 1 goto Working
  say "Kurt here. I've nothing in the works. Bring me Apricorns whenever you have them."
  end
label Ready
  say "It's Kurt! Your {ball} are done. Come by the workshop in Azalea Town and collect them."
  end
label Working
  say "Kurt here. Your {ball} are coming along. Give me until tomorrow."
  end
