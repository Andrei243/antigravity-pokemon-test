# The Lilycove Department Store's roof (plan 21 · H7, part a). docs/scripts.md describes the language; the lines are
# our own, on Emerald's beats (data/maps/LilycoveCity_DepartmentStoreRooftop/scripts.inc).
#
# The drinks machine sells Fresh Water, Soda Pop and Lemonade at 200, 300 and 350, and now and then drops a second
# (one time in sixty-four, Emerald's die). The woman by the railing teaches Substitute once
# (FLAG_MOVE_TUTOR_TAUGHT_SUBSTITUTE; the move's row of MoveTutors.Kanto, which serves every region's one-move tutors).


script Vending
  speaker none
  choose "A drinks machine. Which drink would you like?" "FRESH WATER  200" "SODA POP  300" "LEMONADE  350" "NONE"
  if result == 0 goto Water
  if result == 1 goto Soda
  if result == 2 goto Lemonade
  text "Decided not to buy a drink."
  end
label Water
  takemoney 200
  if result == 0 goto Poor
  sound "bump"
  text "Clang! A bottle of Fresh Water dropped down."
  give "Fresh Water"
  random 64
  if result != 0 end
  text "Lucky! Another bottle of Fresh Water dropped down."
  give "Fresh Water"
  end
label Soda
  takemoney 300
  if result == 0 goto Poor
  sound "bump"
  text "Clang! A can of Soda Pop dropped down."
  give "Soda Pop"
  random 64
  if result != 0 end
  text "Lucky! Another can of Soda Pop dropped down."
  give "Soda Pop"
  end
label Lemonade
  takemoney 350
  if result == 0 goto Poor
  sound "bump"
  text "Clang! A can of Lemonade dropped down."
  give "Lemonade"
  random 64
  if result != 0 end
  text "Lucky! Another can of Lemonade dropped down."
  give "Lemonade"
  end
label Poor
  text "Not enough money..."
  end


script SubstituteTutor
  face self player
  speaker "Woman"
  if flag FLAG_MOVE_TUTOR_TAUGHT_SUBSTITUTE goto Taught
  ask "A Pokémon that can make a decoy of itself is a tricky one to beat. Shall I teach one of yours Substitute?"
  if no goto Later
  say "Lovely! Which one?"
  choosepokemon
  tutor once "Substitute"
  if result == 1 goto Later
  if result == 2 goto Egg
  if result == 3 goto Cannot
  if result == 4 goto Knows
  tutor learn
  if result == 0 goto Later
  if result == 2 goto Hm
  if result == 3 say "{wearer} forgot {forgotten}. And..."
  fanfare levelup
  say "{wearer} learned {thing}!"
  setflag FLAG_MOVE_TUTOR_TAUGHT_SUBSTITUTE
label Taught
  say "Substitute costs a little HP, but the decoy takes the hits in its place. Use it wisely!"
  end
label Later
  say "Oh, never mind, then. I'll be up here enjoying the breeze."
  end
label Egg
  say "That's an Egg! It can't learn anything until it hatches."
  end
label Cannot
  say "Hmm, that one can't learn Substitute, I'm afraid."
  end
label Knows
  say "That one already knows {thing}."
  end
label Hm
  say "{forgotten} isn't a move that can be given up like that."
  goto Later
