# The Champion's room of the Alola Pokémon League (plan 23 · Z7). The lines are our own, on Sun and Moon's beats as they
# are publicly described (docs/mechanics/rulings.md, "Ula'ula, Poni, Aether Paradise and the League"): Professor Kukui
# has built the League, and its Champion's seat is empty until someone sits in it. His team is alola_kukui_1 to 3, the
# last of each being the starter strong against the one the player chose. Beaten, FLAG_ALOLA_DEFEATED_KUKUI opens the
# way to the Hall of Fame.


script Kukui
  face kukui player
  speaker "Prof. Kukui"
  if flag FLAG_ALOLA_DEFEATED_KUKUI goto Won
  if flag AlolaHallOfFame goto Defence
  scene "Kukui"
  say "Woo! Welcome, {player}! You made it, cousin. The Elite Four are down, and nobody sits in this room's chair yet."
  say "I built this League with Hala and the others, because I wanted you kids to have something to aim for."
  say "I wanted to be the first to try you out. Let's go. Show me every move you've learned!"
  if starter "Rowlet" goto AgainstRowlet
  if starter "Litten" goto AgainstLitten
  battle self as "alola_kukui_3"
  goto Beaten
label AgainstRowlet
  battle self as "alola_kukui_1"
  goto Beaten
label AgainstLitten
  battle self as "alola_kukui_2"
label Beaten
  speaker "Prof. Kukui"
  say "...Woo! Woooo! Yeah! That's what I'm talking about! You took me full on, and you came out on top!"
  say "Alola has its first Champion, {player}. The chair's yours. Come on, the Hall of Fame is just through here."
  setflag FLAG_ALOLA_DEFEATED_KUKUI
  endscene
  warp "AlolaLeague_HallOfFame" 6 10 up
  end
label Won
  say "The Hall of Fame is through here."
  warp "AlolaLeague_HallOfFame" 6 10 up
  end


# The Champion's title defended (plan 23 · Z8; rulings, "Ultra Space and the Battle Royal Dome"): once Alola has its
# Champion, whoever beats the Elite Four challenges the one in the chair, as in Sun and Moon. Ours: the challenger is
# drawn from three who know the player (Hau, Gladion and Hapu), each with a team of our own (defence_*), and Professor
# Kukui stands aside to announce them. A defence won goes on to the Hall of Fame as the first victory did.
label Defence
  say "Champion! A challenger has made it through all four of the Elite Four. You know what that means. Defend that title!"
  random 3
  if result == 1 goto Gladion
  if result == 2 goto Hapu
  call DefendAgainstHau
  goto Defended
label Gladion
  call DefendAgainstGladion
  goto Defended
label Hapu
  call DefendAgainstHapu
label Defended
  speaker "Prof. Kukui"
  say "...Woo! The Champion of Alola keeps the title! Come on, the Hall of Fame wants to hear about this."
  setflag FLAG_ALOLA_DEFEATED_KUKUI
  warp "AlolaLeague_HallOfFame" 6 10 up
  end


script DefendAgainstHau
  scene "Hau's challenge"
  setflag FLAG_ALOLA_DEFENCE_HAU
  show challenger_hau
  face challenger_hau player
  speaker "Hau"
  say "Surprise! I trained every single day since the last time. Today I'm taking that chair, {player}. For real!"
  battle challenger_hau as "defence_hau"
  speaker "Hau"
  say "Aw, man... I really thought this was the day. Okay! Malasadas, then training. In that order."
  hide challenger_hau
  clearflag FLAG_ALOLA_DEFENCE_HAU
  endscene
  return


script DefendAgainstGladion
  scene "Gladion's challenge"
  setflag FLAG_ALOLA_DEFENCE_GLADION
  show challenger_gladion
  face challenger_gladion player
  speaker "Gladion"
  say "I didn't come here to make friends. I came here because you're the one to beat. Let's go."
  battle challenger_gladion as "defence_gladion"
  speaker "Gladion"
  say "...Fine. You're still the strongest. I'll be back, and next time it won't be fine."
  hide challenger_gladion
  clearflag FLAG_ALOLA_DEFENCE_GLADION
  endscene
  return


script DefendAgainstHapu
  scene "Hapu's challenge"
  setflag FLAG_ALOLA_DEFENCE_HAPU
  show challenger_hapu
  face challenger_hapu player
  speaker "Hapu"
  say "A kahuna should know where she stands. So I climbed the mountain to find out. Don't you dare hold back."
  battle challenger_hapu as "defence_hapu"
  speaker "Hapu"
  say "Hmph. Poni's ground is solid, and so are you. I'll tell the island the Champion is in good hands."
  hide challenger_hapu
  clearflag FLAG_ALOLA_DEFENCE_HAPU
  endscene
  return
