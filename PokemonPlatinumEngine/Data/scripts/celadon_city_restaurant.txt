# The man who lost everything at the slots gives his Coin Case away (FLAG_GOT_COIN_CASE).


script CoinCaseMan
  speaker "Man"
  if flag FLAG_GOT_COIN_CASE goto Done
  say "Hic! I lost every last coin at the slots. I'm finished with gambling, I tell you!"
  say "Here, you take this. I don't want to look at it ever again."
  give "Coin Case"
  setflag FLAG_GOT_COIN_CASE
  end
label Done
  say "Slot machines are a mug's game. Mark my words!"
  end


# A meal for the team (plan 39 · CR7): three dishes of ours from recipes.json's `stands`, each eaten there and then
# (`craft meal`), its price taken first; and the pantry's goods for the camp's pot. Shown once the crafting kit is in
# the bag (FLAG_GOT_CRAFTING_KIT).
script MealStand
  face self player
  speaker "Waiter"
  say "The chef's kitchen isn't open yet, but our little pot is! Today's specials are for your whole team, eaten here and now."
  choose "What will it be?" "Bean Hotpot, 1,600" "Potato Hash, 1,800" "Honey Oat Cakes, 2,000" "The pantry" "Nothing, thanks"
  if result == 4 goto Bye
  if result == 3 goto Pantry
  setvar VAR_TEMP_1 RESULT
  if var VAR_TEMP_1 == 0 takemoney 1600
  if var VAR_TEMP_1 == 1 takemoney 1800
  if var VAR_TEMP_1 == 2 takemoney 2000
  if result == 0 goto Short
  if var VAR_TEMP_1 == 0 craft meal "beanHotpot"
  if var VAR_TEMP_1 == 1 craft meal "potatoHash"
  if var VAR_TEMP_1 == 2 craft meal "honeyOatCakes"
  say "Enjoy! Every plate came back clean."
  end
label Pantry
  say "We sell the same goods we cook with. Take what you like for your own pot."
  craft pantry
  end
label Short
  say "Oh dear, that's a little more than you have on you."
label Bye
  say "Come back when you're hungry!"
  end
