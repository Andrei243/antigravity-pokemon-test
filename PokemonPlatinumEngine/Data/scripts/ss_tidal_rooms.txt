# The S.S. Tidal's cabins (plan 21 · H12, part g). docs/scripts.md describes the language; the lines are our own, on
# Emerald's beats (data/maps/SSTidalRooms/scripts.inc).
#
# The bed of the player's cabin: a rest heals the team (Emerald's Common_EventScript_OutOfCenterPartyHeal) and passes
# half the voyage, as Emerald's does (plan 21 · H12, part k): the first takes the ship halfway (VAR_SS_TIDAL_STATE 1,
# bound for Lilycove, becomes 5; 2, bound for Slateport, becomes 6) and the second brings her in (5 becomes 3, 6 becomes
# 4; ss_tidal_corridor.txt has the values). Steps taken aboard bring her in as well (GameEngine.CruiseStep).


script Bed
  speaker none
  ask "A soft bed. Lie down for a rest?"
  if no end
  fade out 0.8
  heal
  text "{player} took a rest. The ship's engines hummed below..."
  if var VAR_SS_TIDAL_STATE == 1 goto HalfwayLilycove
  if var VAR_SS_TIDAL_STATE == 2 goto HalfwaySlateport
  if var VAR_SS_TIDAL_STATE == 5 goto Lilycove
  if var VAR_SS_TIDAL_STATE == 6 goto Slateport
  fade in 0.8
  end
label HalfwayLilycove
  setvar VAR_SS_TIDAL_STATE 5
  fade in 0.8
  text "Attention, passengers. We are halfway to Lilycove City."
  end
label HalfwaySlateport
  setvar VAR_SS_TIDAL_STATE 6
  fade in 0.8
  text "Attention, passengers. We are halfway to Slateport City."
  end
label Lilycove
  setvar VAR_SS_TIDAL_STATE 3
  setvar VAR_CRUISE_STEP_COUNT 0
  fade in 0.8
  text "Attention, passengers. We have arrived at Lilycove City."
  end
label Slateport
  setvar VAR_SS_TIDAL_STATE 4
  setvar VAR_CRUISE_STEP_COUNT 0
  fade in 0.8
  text "Attention, passengers. We have arrived at Slateport City."
  end
