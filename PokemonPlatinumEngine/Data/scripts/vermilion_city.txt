# Vermilion City (plan 19 · K4, part b). docs/scripts.md describes the language; the lines are our own, on FireRed's
# beats (data/maps/VermilionCity/scripts.inc).
#
# VAR_MAP_SCENE_VERMILION_CITY, as FireRed counts it: 0 until the S.S. Anne's captain gives HM01 Cut, 1 once he has,
# 3 once the ship has sailed (ss_anne_exterior.txt, as the player comes off her; FireRed's 2 is the step between,
# which the warp does at once here). VAR_VERMILION_CITY_TICKET_CHECK_TRIGGER: 0 until the sailor at the head of the
# pier has seen the S.S. Ticket, 1 while the player stands past him, and 0 again once they step back north.


# FireRed's OnTransition: Oak's aide is gone once spoken to. He comes only after Lt. Surge, for a trainer who never
# had Flash from his colleague on Route 2 (vermilion_city_gym.txt); a new game hides him (common.KantoNewGame), and so
# does this for a save from before plan 19 · K4, part b, which never came here.
script OnEnter
  if not flag FLAG_DEFEATED_LT_SURGE setflag FLAG_HIDE_VERMILION_CITY_OAKS_AIDE
  if flag FLAG_TALKED_TO_OAKS_AIDE_IN_VERMILION setflag FLAG_HIDE_VERMILION_CITY_OAKS_AIDE
  end


# The two steps at the head of the pier (triggers 0 and 1): the sailor asks for a ticket.
script CheckTicket
  face vermilion_ferry_sailor left
  face player right
  speaker "Sailor"
  if var VAR_MAP_SCENE_VERMILION_CITY == 3 goto Sailed
  say "Welcome to the S.S. Anne! May I see your ticket, please?"
  if not flag FLAG_GOT_SS_TICKET goto NoTicket
  text "{player} showed the S.S. Ticket."
  say "Thank you! Mind your step on the gangway, and enjoy the party!"
  setvar VAR_VERMILION_CITY_TICKET_CHECK_TRIGGER 1
  end
label NoTicket
  say "Oh dear, no ticket? I'm afraid I can't let you aboard without one."
  walk player up 1
  end
label Sailed
  say "The S.S. Anne has already set sail. She won't be back until next year."
  walk player up 1
  end


# The two steps north of them (triggers 2 and 3): back on the quay, the check starts again.
script ExitedTicketCheck
  setvar VAR_VERMILION_CITY_TICKET_CHECK_TRIGGER 0
  end


# The sailor spoken to.
script FerrySailor
  face vermilion_ferry_sailor player
  speaker "Sailor"
  # Plan 19 · K7: with the Tri-Pass, the Seagallop sails from here to One Island (FireRed's
  # VermilionCity_EventScript_ChooseSeagallopDestTriPass), and since plan 19 · K9, part a to Two and Three Island too,
  # with the Tri-Pass; since part b the Rainbow Pass sails to every port (common.SeagallopRainbow)
  if item "Rainbow Pass" goto Rainbow
  if item "Tri-Pass" goto Seagallop
  if var VAR_MAP_SCENE_VERMILION_CITY == 3 goto Sailed
  say "Welcome to the S.S. Anne! Have your ticket ready at the gangway, please."
  end
label Sailed
  say "The S.S. Anne has already set sail. She won't be back until next year."
  end
label Seagallop
  say "Ah, a pass for the Seagallop! She's ready whenever you are."
  choose "Where would you like to sail?" "ONE ISLAND" "TWO ISLAND" "THREE ISLAND" "NOT NOW"
  if result == 0 goto Sail
  if result == 1 goto Two
  if result == 2 goto Three
  say "Come back any time."
  end
label Sail
  say "All aboard! Next stop, One Island!"
  call SeagallopToOneIsland
  end
label Two
  say "All aboard! Next stop, Two Island!"
  call SeagallopToTwoIsland
  end
label Three
  say "All aboard! Next stop, Three Island!"
  call SeagallopToThreeIsland
  end
label Rainbow
  say "Ah, a pass for the Seagallop! She's ready whenever you are."
  setvar VAR_SEAGALLOP_HERE 0
  call SeagallopRainbow
  end


# The old man who watches the harbour.
script OldMan
  speaker "Old Man"
  if var VAR_MAP_SCENE_VERMILION_CITY == 3 goto Sailed
  say "Did you see the S.S. Anne down in the harbour? What a beauty she is!"
  end
label Sailed
  say "So the S.S. Anne has gone. She'll be sailing the world for a whole year before she's back."
  end

# ------------------------------------------------------------------ the Ferry's Extra Passenger (plan 41 · EG8)
# The Open Door Agency's case in Kanto: the ticket clerk on the quay finds a stub under the empty seat torn by no hand of
# the office's, the old fisherman sees the dockhand wave one last traveller up the cargo ramp after the count, and once
# Looker has the deductions the dockhand, Dov Harlan, waits on the quay (FLAG_CASE_KANTO_DOV_WAITS) to be faced.

script CaseKantoClerk
  face self player
  if flag FLAG_CASE_KANTO_CLOSED goto Thanks
  if flag FLAG_CASE_KANTO_TAKEN goto Case
  say "Tickets for the Sevii ferry! Every seat counted, every stub torn. That's my job."
  end
label Case
  if flag FLAG_CASE_KANTO_CLUE_STUB goto Again
  say "Seat thirteen, the ghost's seat? Paid for every sailing, empty every sailing. I found this stub under it last night."
  say "Look at the tear, though. I tear every stub on the left. This one's torn on the right. It isn't one of mine."
  setflag FLAG_CASE_KANTO_CLUE_STUB
  text "{player} noted the ticket stub torn by someone else's hand."
  end
label Again
  say "Torn on the right. Whoever sold that ticket, it wasn't this office."
  end
label Thanks
  say "No more ghost on the ferry. Seat thirteen has a real passenger now, with a real ticket. One of mine!"
  end

script CaseKantoFisherman
  face self player
  if flag FLAG_CASE_KANTO_CLOSED goto Thanks
  if flag FLAG_CASE_KANTO_TAKEN goto Case
  say "I fish off this quay every evening. The ferry comes and goes, and the fish don't mind it a bit."
  end
label Case
  if flag FLAG_CASE_KANTO_CLUE_REMARK goto Again
  say "The ferry? Funny thing. Every evening, after the crew have counted heads, Dov the dockhand waves one more traveller aboard."
  say "Not up the gangway, mind. Up the cargo ramp, with the crates. And he pockets something as they pass."
  setflag FLAG_CASE_KANTO_CLUE_REMARK
  text "{player} noted the fisherman's remark about the cargo ramp."
  end
label Again
  say "One more up the cargo ramp, after the count. Every evening, regular as the tide."
  end
label Thanks
  say "Dov's on the gangway now, checking tickets like a proper sailor. The fish miss him, I expect."
  end

script CaseKantoDov
  face self player
  speaker "Dov"
  if flag FLAG_CASE_KANTO_CULPRIT end
  scene "DovFaced"
  say "Looker's friend, eh? So somebody found my stub. Thought a ghost story would keep folk from asking."
  say "Fine. If I'm to be caught, I'll be caught fighting. Right here on the quay!"
  battle self as "case_kanto_dov"
  say "...Sunk. All right. I'll pay back every fare I took twice. Tell Looker I'll be at his door, ghost and all."
  setflag FLAG_CASE_KANTO_CULPRIT
  clearflag FLAG_CASE_KANTO_DOV_WAITS
  hide self
  endscene
  end
