# The Lottery Corner: once a day the clerk reads out the day's number (Models/Lottery.cs) and matches it, digit by
# digit from the right, against the ID numbers of the player's Pokémon, on the team and in the boxes. The more digits
# match, the better the prize: one an Ultra Ball, two a PP Up, three an Exp. Share, four a Max Revive, all five a
# Master Ball. FLAG_DAILY_CHECKED_LUCKY_NUMBER is cleared each new day.


script LotteryClerk
  speaker "Lottery Clerk"
  if flag FLAG_DAILY_CHECKED_LUCKY_NUMBER goto Tomorrow
  say "Hello, and welcome to the Jubilife TV Lottery Corner!"
  say "Every day we draw a lucky number. If it matches the ID number of any of your Pokémon, you win a prize!"
  ask "Shall I check today's number against your Pokémon?"
  if no goto Later
  setflag FLAG_DAILY_CHECKED_LUCKY_NUMBER
  say "Today's lucky number is {ticket}! Now then, let me look through your Pokémon's ID numbers..."
  lottery draw
  if result == 5 goto Five
  if result == 4 goto Four
  if result == 3 goto Three
  if result == 2 goto Two
  if result == 1 goto One
  say "Oh dear... Not a single digit matched today."
  say "Every day brings a new number, though. Do come back tomorrow!"
  end
label One
  call Whose
  say "The last digit is a match! That wins you our fifth prize."
  give "Ultra Ball"
  goto Thanks
label Two
  call Whose
  say "The last two digits are a match! That wins you our fourth prize."
  give "PP Up"
  goto Thanks
label Three
  call Whose
  say "The last three digits are a match! That wins you our third prize."
  give "Exp. Share"
  goto Thanks
label Four
  call Whose
  say "The last four digits are a match! That wins you our second prize!"
  give "Max Revive"
  goto Thanks
label Five
  call Whose
  say "All five digits are a match! Incredible! That wins you our first prize!"
  give "Master Ball"
  goto Thanks
label Thanks
  speaker "Lottery Clerk"
  say "Congratulations! Do try your luck again tomorrow."
  end
label Tomorrow
  say "Today's draw is done, I'm afraid. A new lucky number comes up tomorrow!"
  end
label Later
  say "Of course. Come back whenever you feel lucky!"
  end


# Whose ID number won: on the team, or in the boxes.
script Whose
  lottery boxed
  if result == 1 goto Boxed
  say "Oh! The ID number of {winner}, on your team, matches!"
  return
label Boxed
  say "Oh! The ID number of {winner}, in your PC Boxes, matches!"
  return

# The gift box's questionnaire (plan 40 · GE7; the user's 17b): Madame Ottilie Fray asks two questions answered with the
# phrase picker, as Platinum's own gift menu was opened by two phrases given to a man on the station's third floor. Her
# second question has an answer of ours, SPARROW and CLOCKWORK (19b), which opens the box (FLAG_GIFT_BOX_OPEN): the Gift
# Runners in every region's Pokémon Centers hand out what waits from then on. A Gift Runner who isn't busy yet says both
# words, so nobody need find them by chance.
script Questionnaire
  face self player
  speaker "Madame Ottilie Fray"
  if flag FLAG_GIFT_BOX_OPEN goto Opened
  say "Good day! I am Ottilie Fray, and I ask a question or two of everyone who comes through these doors."
  ask "Would you answer my questionnaire? Two questions, two words each."
  if no goto Declined
  say "The first. What do you like best about travelling with Pokémon? Two words, please."
  phrase GIFT_QUESTION_ONE
  speaker "Madame Ottilie Fray"
  if result == 0 goto Declined
  say "\"{phrase:GIFT_QUESTION_ONE}\"... Charming. I shall write that down."
  say "And the second. What sings at the break of day, and what never stops ticking?"
  phrase GIFT_QUESTION_TWO
  speaker "Madame Ottilie Fray"
  if result == 0 goto Declined
  if phrase GIFT_QUESTION_TWO "small:sparrow" "small:clockwork" goto Right
  if phrase GIFT_QUESTION_TWO "small:clockwork" "small:sparrow" goto Right
  say "\"{phrase:GIFT_QUESTION_TWO}\"? Not what I had in mind, I'm afraid. Do try again another time."
  end
label Right
  say "\"{phrase:GIFT_QUESTION_TWO}\"! Exactly so. You've answered the questionnaire, and the gift box is open to you."
  setflag FLAG_GIFT_BOX_OPEN
  say "The Gift Runners in their yellow scarves stand in the Pokémon Centers. They'll hand you whatever is waiting."
  end
label Declined
  say "Another time, then. My questions will keep."
  end
label Opened
  say "Have you been to see the Gift Runners? Gifts come to the box from all over, on days and for deeds."
  say "And words, too. Say the right two words to a Runner, and you never know what might turn up."
  end
