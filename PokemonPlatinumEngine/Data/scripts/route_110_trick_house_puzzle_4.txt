# The Trick House's fourth puzzle (plan 21 · H12, part j). docs/scripts.md describes the language; the lines are our own,
# on Emerald's beats (data/maps/Route110_TrickHousePuzzle4/scripts.inc).
#
# Boulders to push with Strength (things of the map, put back as the room is come into). VAR_TRICK_HOUSE_PUZZLE_4_STATE:
# 0 the scroll not found, 1 the code learnt, 2 written on the door, which opens it (FLAG_HOENN_TRICK_HOUSE_PUZZLE_4_UNLOCKED,
# ours).

script Scroll
  if var VAR_TRICK_HOUSE_PUZZLE_4_STATE != 0 goto Again
  setvar VAR_TRICK_HOUSE_PUZZLE_4_STATE 1
  fanfare item
  text "{player} found a scroll!"
  text "{player} learnt the secret code written on it by heart."
  end
label Again
  text "The secret code is written on it: \"The Trick Master is a genius of disguise.\""
  end


script Door
  if var VAR_TRICK_HOUSE_PUZZLE_4_STATE == 2 goto Open
  if var VAR_TRICK_HOUSE_PUZZLE_4_STATE == 1 goto Write
  text "The door is locked. Something is written on it: \"Write the secret code here.\""
  end
label Write
  text "{player} wrote the secret code on the door. \"The Trick Master is a genius of disguise.\""
  sound "door_open"
  text "Click! The lock opened!"
  setvar VAR_TRICK_HOUSE_PUZZLE_4_STATE 2
  setflag FLAG_HOENN_TRICK_HOUSE_PUZZLE_4_UNLOCKED
  end
label Open
  text "The door stands open."
  end
