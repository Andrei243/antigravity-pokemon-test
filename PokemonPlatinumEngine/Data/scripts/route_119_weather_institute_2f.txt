# The Weather Institute's upper floor on Route 119 (plan 21 · H6). docs/scripts.md describes the language; the lines
# are our own, on Emerald's beats (data/maps/Route119_WeatherInstitute_2F/scripts.inc).
#
# Team Aqua holds the floor (VAR_WEATHER_INSTITUTE_STATE 0): three grunts, and Shelly, their admin, before the
# cornered scientist. Beaten, she is about to go on when a grunt runs up from the stairs: Team Magma is on its way to
# Mt. Pyre. Team Aqua leaves (FLAG_HIDE_ROUTE_119_TEAM_AQUA, state 1), the workers come out, and the scientist gives
# the player Castform holding a Mystic Water, as Emerald gives it (FLAG_RECEIVED_CASTFORM).


script OnEnter
  if var VAR_WEATHER_INSTITUTE_STATE == 0 goto Held
  place weather_institute_2f_scientist 4 6 down
  end
label Held
  place weather_institute_2f_scientist 1 6 right
  end


script Shelly
  face self player
  speaker "Shelly"
  if var VAR_WEATHER_INSTITUTE_STATE != 0 end
  scene "Shelly at the Weather Institute"
  say "Ahahaha! So you're the kid who keeps getting in Team Aqua's way. Archie told me all about you."
  say "We came for the research on rain. With it, we'll fill the world with sea. You won't stop us, sweetie!"
  battle self
  say "Ahaha... well, aren't you a tough one. Fine, I lost. But I've still got what I came for."
  clearflag FLAG_HIDE_WEATHER_INSTITUTE_2F_AQUA_GRUNT_M
  show weather_institute_2f_grunt_3
  if facing left walk player up 1
  walk weather_institute_2f_grunt_3 left 11 fast
  speaker "Team Aqua Grunt"
  say "Admin Shelly! Bad news! Team Magma's been seen heading for Mt. Pyre!"
  speaker "Shelly"
  emote weather_institute_2f_shelly exclaim
  say "Magma? At Mt. Pyre? Then they're after the orbs! Everyone out, we're going to Mt. Pyre!"
  say "As for you, kid... stay out of grown-ups' business, or you'll get swept away."
  setvar VAR_WEATHER_INSTITUTE_STATE 1
  clearflag FLAG_HIDE_WEATHER_INSTITUTE_2F_WORKERS
  fade out
  setflag FLAG_HIDE_ROUTE_119_TEAM_AQUA
  setflag FLAG_HIDE_WEATHER_INSTITUTE_2F_AQUA_GRUNT_M
  hide weather_institute_2f_shelly
  hide weather_institute_2f_grunt_1
  hide weather_institute_2f_grunt_2
  hide weather_institute_2f_grunt_3
  hide weather_institute_2f_grunt_4
  show man_4
  show man_4_2
  # Where OnEnter holds him; said again so the scene stands on its own, wherever he was left
  place weather_institute_2f_scientist 1 6 right
  place player 5 6 left
  fade in
  walk weather_institute_2f_scientist right 3
  face weather_institute_2f_scientist player
  call Castform
  endscene
  end


# The scientist's thanks: a Castform of the institute's own
script Castform
  speaker "Scientist"
  say "Thank you! Thank you so much! I thought they'd never leave."
  say "Please, take this Pokémon as our thanks. It's Castform. Its shape changes with the weather!"
  givepokemon "Castform" 25 holding "Mystic Water"
  setflag FLAG_RECEIVED_CASTFORM
  fanfare pokemon
  text "{player} received Castform!"
  if result == 2 text "Castform was sent to the PC."
  say "It's holding a Mystic Water, too. Raise it well, and watch how it changes in the sun and the rain."
  return


script WeatherScientist
  face self player
  speaker "Scientist"
  if var VAR_WEATHER_INSTITUTE_STATE == 0 goto Held
  if not flag FLAG_RECEIVED_CASTFORM goto Give
  if flag HoennHallOfFame goto Report
  say "Strange weather has been reported in Hoenn of late. Heavy rain in one place, blazing sun in another."
  say "Castform would know all about it. Take good care of it!"
  end
label Give
  call Castform
  end
label Held
  say "Th-they want our data on rain! What could anyone want with that much rain?"
  end
# Once the player is Champion (plan 21 · H12, part e; Emerald's Route119_WeatherInstitute_2F_EventScript_TryStartAbnormalWeather):
# the scientist tracks a drought or a downpour that has settled over one route, which is where Groudon's or Kyogre's cave
# has opened. Each visit's first word of it draws a new place (Emerald's CreateAbnormalWeatherEvent behind FLAG_TEMP_2, a
# flag of the visit); once both have been beaten or caught the weather is ordinary again
label Report
  if not flag FLAG_DEFEATED_KYOGRE goto Tracking
  if not flag FLAG_DEFEATED_GROUDON goto Tracking
  say "The abnormal weather has stopped. Nothing strange has been recorded anywhere in Hoenn for a while now."
  say "A little rain now and then is a blessing, really."
  end
label Tracking
  if not flag FLAG_MAP_LOCAL_WEATHER_REPORTED call Draw
  say "I keep track of the weather all over Hoenn. Lately, something very odd has been showing up on our charts."
  if var VAR_ABNORMAL_WEATHER_LOCATION >= 9 goto Rain
  if var VAR_ABNORMAL_WEATHER_LOCATION <= 2 say "A fierce drought has settled over Route 114, and it won't move on."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 3 say "A fierce drought has settled over Route 115, and it won't move on."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 4 say "A fierce drought has settled over Route 115, and it won't move on."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 5 say "A fierce drought has settled over Route 116, and it won't move on."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 6 say "A fierce drought has settled over Route 116, and it won't move on."
  if var VAR_ABNORMAL_WEATHER_LOCATION >= 7 say "A fierce drought has settled over Route 118, and it won't move on."
  say "Sunshine that harsh, in one spot only... Could something be sleeping somewhere close by?"
  end
label Rain
  if var VAR_ABNORMAL_WEATHER_LOCATION <= 10 say "Heavy rain has been pouring down over Route 105, and it won't let up."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 11 say "Heavy rain has been pouring down over Route 125, and it won't let up."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 12 say "Heavy rain has been pouring down over Route 125, and it won't let up."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 13 say "Heavy rain has been pouring down over Route 127, and it won't let up."
  if var VAR_ABNORMAL_WEATHER_LOCATION == 14 say "Heavy rain has been pouring down over Route 127, and it won't let up."
  if var VAR_ABNORMAL_WEATHER_LOCATION >= 15 say "Heavy rain has been pouring down over Route 129, and it won't let up."
  say "A downpour that never moves on... Could something be sleeping deep under the sea there?"
  end


# Where the weather settles this visit (Emerald's CreateAbnormalWeatherEvent): Terra Cave's eight places while Kyogre is
# beaten, the Marine Cave's while Groudon is, otherwise either kind as a coin falls, and one of its eight as a die does
script Draw
  setflag FLAG_MAP_LOCAL_WEATHER_REPORTED
  if flag FLAG_DEFEATED_KYOGRE goto Terra
  if flag FLAG_DEFEATED_GROUDON goto Marine
  random 2
  if var RESULT == 0 goto Terra
label Marine
  random 8
  setvar VAR_ABNORMAL_WEATHER_LOCATION RESULT
  addvar VAR_ABNORMAL_WEATHER_LOCATION 9
  call HoennAbnormalWeather
  return
label Terra
  random 8
  setvar VAR_ABNORMAL_WEATHER_LOCATION RESULT
  addvar VAR_ABNORMAL_WEATHER_LOCATION 1
  call HoennAbnormalWeather
  return
