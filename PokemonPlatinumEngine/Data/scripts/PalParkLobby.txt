# Pal Park's lobby on Route 221 (plan 06 · R16b). docs/scripts.md describes the language; the lines are our own, on
# the original's beats (scripts_pal_park_lobby.s).
#
# The attendant in the gap of the counter starts a Catching Show (palpark check, palpark start) for whoever has six
# Pokémon waiting in the park and six free places in the PC boxes, steps aside and lets the player through the door
# to the park. VAR_PAL_PARK_STATE says how the player comes back: 1 with all six caught (the score is told, the
# record kept, a berry given, and the six sent to the boxes or left in the park for another try), 2 having given
# up (the six go on waiting). Nothing is migrated (plan 08 · P20 to P23 were dropped): when nothing waits, palpark check
# lets six visitors come of the species Sinnoh has no other way to (plan 03 · D12a, Models/Pastimes.cs
# PalParkState.Visitors, a stand-in the user can overturn).
#
# Professor Oak stands in the middle of the lobby the first time the player comes in (plan 02 · S15, part a; the
# original's OnFrame while VAR_PAL_PARK_LOBBY_STATE is 0: nobody comes in before the National Pokédex, since Route
# 221's two workers stand before the doors until then). The player goes up to him (the original's professor comes
# down to the door, where the text would cover the two of them). He says what the park is for, puts the Trainer Counter on the
# Pokétch, says he will be in Eterna City for a while (FLAG_HIDE_ETERNA_CITY_SOUTH_HOUSE_PROF_OAK cleared) and goes;
# and the record to beat is the rival's 2,000 points from then on (VAR_CATCHING_SHOW_RECORD).
#
# The viewing gallery upstairs with its four spectators is a map of its own, PalParkGallery (plan 01 · M11, part d),
# come to by the escalator.


script OnEnter
  # Professor Oak, the first time. Only someone with the National Pokédex comes in at all (Route 221's workers keep
  # the doors until then), and the scene asks for it too, so that nobody meets him who came some other way
  if var VAR_PAL_PARK_LOBBY_STATE != 0 goto Park
  if nationaldex goto Oak
label Park
  if var VAR_PAL_PARK_STATE == 0 end
  place worker 8 9 left
  if var VAR_PAL_PARK_STATE == 1 goto Tally
  # Given up: nothing is told, and the six go on waiting in the park
  setvar VAR_PAL_PARK_STATE 0
  palpark end
  goto Out
label Tally
  setvar VAR_PAL_PARK_STATE 0
  walk player down 2
  face player right
  speaker "Park Attendant"
  say "All six caught! Now, let's see how your Catching Show went."
  palpark points catching
  text "Catching points: {number}"
  palpark points time
  text "Time points: {number}"
  palpark points type
  text "Type points: {number}"
  palpark points total
  text "Total: {number} points!"
  palpark record
  if result == 1 goto Record
  say "A fine show. Here is something for your trouble."
  goto Prize
label Record
  say "That beats the record! Outstanding! This is for you."
label Prize
  palpark prize
label AskStore
  ask "Shall I send the six you caught to your PC boxes?"
  if yes goto Stored
  ask "Then you'd like them left in the park, to put on the show again?"
  if no goto AskStore
  say "That's the spirit! They'll be waiting for you."
  palpark end
  goto Out
label Stored
  palpark store
  say "Your Pokémon have been sent to your PC boxes." "Do come and see us again!"
label Out
  face player down
  walk player down 5
  walk worker left 1 down 2
  face worker down
  end
label Oak
  call OakWelcomes
  end


# PalParkLobby_OnFrame_ProfOak: he sees whoever has just come in by the door, and they go up to him
script OakWelcomes
  scene "Professor Oak at Pal Park"
  clearflag FLAG_HIDE_ETERNA_CITY_SOUTH_HOUSE_PROF_OAK
  setvar VAR_CATCHING_SHOW_RECORD 2000
  emote prof_oak exclaim
  walk player up 3
  speaker "Prof. Oak"
  say "Ah, {player}! So you came. Welcome to Pal Park!"
  say "Pokémon are brought here from all over the country and let loose in the park, and their Trainers catch them again in a Catching Show. I had a hand in setting it up."
  emote prof_oak exclaim
  say "Now that I think of it, I have just the thing for a Trainer with a National Pokédex to fill. Hold out your Pokétch!"
  poketchapp TrainerCounter
  fanfare keyitem
  text "The Trainer Counter was added to the Pokétch!"
  say "It keeps count of the Pokémon you meet one after another. You will find a use for it, I am sure."
  say "I shall be staying in Eterna City for some time. Do call on me there. Good luck, {player}!"
  wait 0.25
  walk player right 1
  face player left
  move prof_oak down 4
  wait 0.4
  face player down
  waitmoves
  sound "stairs"
  hide prof_oak
  setflag FLAG_HIDE_PAL_PARK_LOBBY_PROF_OAK
  setvar VAR_PAL_PARK_LOBBY_STATE 1
  endscene
  return


script Worker
  if flag FLAG_TALKED_TO_PAL_PARK_LOBBY_RECEPTIONIST goto Greeting
  setflag FLAG_TALKED_TO_PAL_PARK_LOBBY_RECEPTIONIST
  say "Welcome to Pal Park, {player}!" "Pokémon from far away come to stay in the park, and Trainers catch them in a Catching Show."
label Greeting
  choose "What will it be today?" "ENTER" "INFO" "EXIT"
  if result == 0 goto Check
  if result == 1 goto Info
  say "Aww, that's too bad. Another day, then!"
  end
label Info
  choose "What shall I read out of the manual?" "The park" "The Catching Show" "Bringing Pokémon" "That's enough"
  if result == 0 goto Park
  if result == 1 goto Show
  if result == 2 goto Bringing
  say "Ask me whenever you like!"
  end
label Park
  say "Pal Park is where Pokémon from faraway regions come to stay. Six at a time come over, Pokémon that live nowhere else in Sinnoh, and wait in the park until someone catches them."
  end
label Show
  say "In a Catching Show you walk the park's grass and water with six Park Balls. Each of the six keeps to its own corner, and a Park Ball never misses."
  say "The score is the Pokémon's own points, plus points for catching them quickly, and for never catching two of a type in a row. A good score earns a berry!"
  end
label Bringing
  say "Six Pokémon come over together, and no more until those six have been caught and sent to your PC boxes."
  end
label Check
  palpark check
  if result == 1 goto BoxesFull
  if result == 2 goto NoneBrought
  palpark start
  say "Here are your six Park Balls. The score to beat is {var:VAR_CATCHING_SHOW_RECORD} points. Good luck, {player}!"
  walk self up 2 right 1
  face self left
  walk player up 5
  warp "PalPark" 24 47 up
  end
label NoneBrought
  say "Oh! But there are no Pokémon waiting in the park." "Six must come over from far away before a show can be held. None have come yet."
  end
label BoxesFull
  say "Your PC boxes haven't room for six more Pokémon. Please make some room first."
  end


script RecordGuy
  if var VAR_CATCHING_SHOW_RECORD == 0 goto None
  # PalParkLobby_BufferRivalRecord: the record Professor Oak leaves behind is the rival's
  if var VAR_CATCHING_SHOW_RECORD == 2000 goto Rivals
  say "The record for a Catching Show stands at {var:VAR_CATCHING_SHOW_RECORD} points. It's written up behind the counter."
  end
label Rivals
  say "The record for a Catching Show stands at 2000 points. Somebody called {rival} set it, and then ran out of the door before I could shake his hand."
  end
label None
  say "Nobody has put on a Catching Show here yet. The first score will be the record!"
  end


script PoketchAppLady
  if var VAR_MAP_LOCAL_PAL_PARK_APP == 1 goto GotTimer
  if var VAR_MAP_LOCAL_PAL_PARK_APP == 2 goto GotColours
  if poketchapp ColorChanger goto All
  say "Isn't Pal Park wonderful? Pokémon I have never seen, from places I'll never go!"
  if poketchapp KitchenTimer goto Kecleon
  say "There is one I long to see: it does nothing but eat and sleep, they say, and it is as big as a house."
  if not has "Snorlax" goto No
  say "That's it! That's the very one! Oh, thank you. Here, take this for your Pokétch."
  poketchapp KitchenTimer
  setvar VAR_MAP_LOCAL_PAL_PARK_APP 1
  end
label Kecleon
  say "And there's another: it changes colour to match whatever it stands on. Imagine!"
  if not has "Kecleon" goto No
  say "Oh! It went green, and then it went pink! Thank you! This is for your Pokétch."
  poketchapp ColorChanger
  setvar VAR_MAP_LOCAL_PAL_PARK_APP 2
  end
label GotTimer
  say "The Kitchen Timer makes you want to cook something, doesn't it?"
  end
label GotColours
  say "Changing the Pokétch's colour changes my mood for the whole day."
  end
label No
  say "What was its name, now? If you ever have one with you, do show me."
  end
label All
  say "Are you putting on a Catching Show today? I'll be cheering!"
  end
