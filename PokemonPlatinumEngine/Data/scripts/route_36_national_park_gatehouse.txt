# counter, and coming in from the park while a contest is under way asks whether the player is done.

  setvar VAR_BUG_CONTEST_GATE 1
  call BugContestDesk
  end

script OnEnter
  bugcontest on
  if result == 0 end
  setvar VAR_BUG_CONTEST_GATE 1
