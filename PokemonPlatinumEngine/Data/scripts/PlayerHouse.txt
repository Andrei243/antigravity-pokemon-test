# Downstairs at home (plan 02 · S4). docs/scripts.md describes the language; the lines are our own, on the
# original's beats (scripts_twinleaf_town_player_house_1f.s).
#
# Mom: the rival has been and gone; a warning about tall grass at the door; the Running Shoes once the player has a
# Pokémon; and, once the Pokédex is the player's, the Journal and the rival's mother with a parcel for her son.
# VAR_PLAYER_HOUSE_STATE counts the visits as the original's does: 1 told, 2 warned, 3 back from the rival's
# battle, 4 the shoes given, 6 the parcel given.
#
# After the Hall of Fame (plan 02 · S14; the original's OnFrame_Postgame): the first time the new Champion comes
# downstairs Mom has news. The rival has been round looking for them, shouting about a ship at Snowpoint City; and
# until the National Pokédex is the player's she asks how the professor's Pokédex is coming along.
# VAR_PLAYER_HOUSE_POSTGAME_STATE is 1 from the Hall of Fame's first victory and 2 once she has said it.


script OnEnter
  if var VAR_PLAYER_HOUSE_STATE == 0 goto RivalLeft
  if var VAR_PLAYER_HOUSE_POSTGAME_STATE == 1 goto Postgame
  if var VAR_PLAYER_HOUSE_STATE == 3 goto RunningShoes
  end
label Postgame
  # TwinleafTownPlayerHouse1F_OnFrame_Postgame, with the original's two flags (it asks them although, the state
  # having moved on to 2, the scene never comes a second time)
  setvar VAR_PLAYER_HOUSE_POSTGAME_STATE 2
  if flag FLAG_TALKED_TO_MOM_ABOUT_RIVAL_SNOWPOINT_CITY end
  if not flag FLAG_TALKED_TO_MOM_ABOUT_NATIONAL_DEX_PROGRESS goto MomHasNews
  if not nationaldex end
label MomHasNews
  wait 0.15
  face player mom
  face mom player
  emote mom exclaim
  speaker "Mom"
  say "{player}! There you are, sleepyhead. {rival} was round here a little while ago, looking for you."
  say "I couldn't make out what it was all about. He kept shouting that you had to get on a ship at Snowpoint City!"
  say "You know how he is. He was out of the door again before I could ask him a single thing."
  if nationaldex goto RivalCame
  # TwinleafTownPlayerHouse1F_IsYourProjectComingAlong
  setflag FLAG_TALKED_TO_MOM_ABOUT_NATIONAL_DEX_PROGRESS
  say "Well, never mind him. How are you getting on, Champion? Is that Pokédex of Professor Rowan's coming along?"
  face mom left
  end
label RivalCame
  # TwinleafTownPlayerHouse1F_RivalCameLookingForYou
  setflag FLAG_TALKED_TO_MOM_ABOUT_RIVAL_SNOWPOINT_CITY
  face mom left
  end
label RivalLeft
  walk mom right 1
  face mom player
  speaker "Mom"
  say "Oh, you're up! {rival} was just here looking for you."
  say "He went straight back out again. Something about Professor Rowan. That boy is always in such a hurry!"
  walk mom left 1
  face mom left
  setflag FLAG_TALKED_TO_MOM
  setvar VAR_PLAYER_HOUSE_STATE 1
  end
label RunningShoes
  walk mom down 2 left 3
  face mom player
  face player mom
  speaker "Mom"
  say "{player}! Welcome home. ...Is that a Pokémon? Professor Rowan gave you a Pokémon of your own?"
  say "Oh, I'm so glad he happened to be there. You should go to his lab in Sandgem Town and thank him properly."
  say "And if you're going to be out and about, you'll want these. Your very own Running Shoes!"
  setflag FLAG_HAS_RUNNING_SHOES
  fanfare keyitem
  text "{player} received the Running Shoes!"
  say "Hold down the run button as you walk, and off you'll go! Now, be careful out there."
  walk mom right 3 up 2
  face mom left
  setvar VAR_PLAYER_HOUSE_STATE 4
  end


# The tile in front of the door, the first time the player heads out.
script DoorWarning
  face mom player
  speaker "Mom"
  say "Going out after {rival}? Remember, wild Pokémon live in the tall grass."
  say "Never walk into it without a Pokémon of your own!"
  setvar VAR_PLAYER_HOUSE_STATE 2
  end


script Mom
  if flag FLAG_HAS_POKEDEX goto Pokedex
  if var VAR_PLAYER_HOUSE_STATE >= 4 goto Rest
  if party > 0 goto Rest
  speaker "Mom"
  say "{rival} is always in such a rush. You'd better not keep him waiting!"
  end
label Pokedex
  if flag FLAG_RECEIVED_PARCEL goto Rest
  goto Parcel
label Rest
  # What the common Healer does (her own words, then the team is well), written out so that she can go on
  sayown
  heal
  fanfare heal
  # Plan 02 · S10: once Lake Valor has blown she asks after the tremor, until the Distortion World is behind the
  # player (the original's TwinleafTownPlayerHouse1F_IsEverythingAllRight: VAR_EXITED_DISTORTION_WORLD_STATE 2, which
  # Sendoff Spring sets; plan 02 · S13)
  if var VAR_CANALAVE_LIBRARY_STATE < 2 end
  if var VAR_EXITED_DISTORTION_WORLD_STATE >= 2 end
  speaker "Mom"
  say "{player}, is everything all right? The whole house shook a while ago, and the television won't say what it was."
  say "You're mixed up in it somehow, aren't you? ...I thought so. Then be careful, and come home when it's done."
  end
label Parcel
  speaker "Mom"
  say "Welcome home! Let me look after your Pokémon first."
  heal
  fanfare heal
  say "...So Professor Rowan wants you to fill in his Pokédex? That's wonderful! Then you'll want this."
  give "Journal"
  say "It's a Journal. It keeps a note of everything you do, so you can always look back on your adventure."
  say "Honestly, I'd love to go on a journey like that myself. You take care of yourself out there."
  sound "door_open"
  show rival_mom
  speaker "{rival}'s Mom"
  say "Hello? Is {rival} here? ...Oh, he's gone already? That boy!"
  walk rival_mom up 2
  say "He ran off without the parcel I'd put by for him. I so wanted him to have it..."
  speaker "Mom"
  say "Then {player} can take it to him! You will, won't you?"
  speaker "{rival}'s Mom"
  say "Would you really? Thank you, dear!"
  give "Parcel"
  setflag FLAG_RECEIVED_PARCEL
  say "I think he said he was going to Jubilife City. Please give it to him when you catch up with him."
  walk rival_mom down 2
  hide rival_mom
  sound "door_close"
  setvar VAR_PLAYER_HOUSE_STATE 6
  end
