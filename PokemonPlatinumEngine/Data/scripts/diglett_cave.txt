# The place's arrival lays who of the Gym Leaders is here today (Johto's leftovers)
script OnEnter
  # Brock's street meeting (Johto's leftovers): Wednesdays, once Brock's number is in the Pokégear
  setflag FLAG_UNK_262
  if weekday wednesday call ShowMeetBrock
  end


script ShowMeetBrock
  if flag FLAG_PHONE_CONTACT_BROCK clearflag FLAG_UNK_262
  return

# Brock, out on Wednesday (the Leaders' street meetings, Johto's leftovers)
script MeetBrock
  face d01r0101_gsleader14 player
  speaker "Brock"
  random 2
  if result == 1 goto Two
  say "Hey! I come down here on Wednesdays. The Diglett keep the cave's rocks in great shape."
  end
label Two
  say "If you want a rock-solid team, spend some time underground. Trust me."
  end
