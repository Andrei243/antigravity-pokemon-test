# Twist Mountain's chamber (plan 22 · U10, part e). docs/scripts.md describes the language; the lines are our own.
#
# Regigigas sleeps at the chamber's head. It stirs only when the player carries Regirock, Regice and Registeel (as Platinum's
# Snowpoint Temple asks), and is battled at level 68. Caught or beaten it is gone for good (FLAG_UNOVA_FOUGHT_REGIGIGAS); run
# from, it is back the next time the chamber is come to.
script Regigigas
  if not has "Regirock" goto Sleeps
  if not has "Regice" goto Sleeps
  if not has "Registeel" goto Sleeps
  scene "Regigigas wakes"
  music stop
  face player self
  letterbox on 0.4
  camera pan 14 5 1.2
  camera zoom 1.4 0.8
  effect beam 14 5
  shake self 8
  camera shake 0.8
  speaker none
  text "The three giants in the team grew restless...\nThe colossus stirred awake!"
  cry "Regigigas"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Regigigas" 68
  if result == 3 setflag FLAG_UNOVA_CAUGHT_REGIGIGAS
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_REGIGIGAS
  hide self
  text "The chamber is still again..."
  end
label Fled
  hide self
  text "Regigigas sank back into the mountain's depths..."
  end
label Sleeps
  speaker none
  text "A colossus of stone sleeps here, unmoving.\nIt seems to be waiting for something."
  end


script Plaque
  speaker none
  text "Twist Mountain Chamber"
  end


script Inscription
  speaker none
  text "Gather the giants of rock, ice and steel, and the one who shaped them will stir."
  end
