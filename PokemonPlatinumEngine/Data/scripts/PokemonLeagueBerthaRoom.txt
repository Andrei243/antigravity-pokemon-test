# PokemonLeagueBerthaRoom_OnFrame_EnterRoom, by the door's own flag (see Aaron's room)
script OnEnter
  if not flag FLAG_HIDE_POKEMON_LEAGUE_BERTHA_ROOM_ENTRANCE_DOOR end
  walk player up 2
  sound "lift_stop"
  clearflag FLAG_HIDE_POKEMON_LEAGUE_BERTHA_ROOM_ENTRANCE_DOOR
  show entrance_door
  end


script Bertha
  speaker "Bertha"
  if flag FLAG_DEFEATED_BERTHA goto Beaten
  music "common/eye_elite_four"
  say "Well now. What a young thing to have come so far! But there's steel in the way you stand, isn't there, dear?"
  say "My name is Bertha. I've spent a long life with Ground Pokémon: patient, steady, and stronger than anyone expects."
  say "Now then. Would you show an old woman how much you've learned on your way here?"
  battle self as "elite_four_bertha"
  setflag FLAG_DEFEATED_BERTHA
  sound "lift_stop"
  setflag FLAG_HIDE_POKEMON_LEAGUE_BERTHA_ROOM_EXIT_DOOR
  hide exit_door
  say "Well, I never! You and your Pokémon are quite something, dear. I'm glad I lived to see it."
  say "Off you go, then. The next one burns a good deal hotter than I do."
  end
label Beaten
  say "You're a strong one, dear, and I can see your Pokémon believe in you. Keep each other that way."
  say "Go on. Don't keep the others waiting!"
  end
