# The place's arrival lays who of the Gym Leaders is here today (Johto's leftovers)
script OnEnter
  # Jasmine's street meeting (Johto's leftovers): Fridays, once Jasmine's number is in the Pokégear
  setflag FLAG_UNK_2CB
  if weekday friday call ShowMeetJasmine
  end


script ShowMeetJasmine
  if flag FLAG_PHONE_CONTACT_JASMINE clearflag FLAG_UNK_2CB
  return

# Jasmine, out on Friday (the Leaders' street meetings, Johto's leftovers)
script MeetJasmine
  face t26r0701_gsleader6 player
  speaker "Jasmine"
  random 2
  if result == 1 goto Two
  say "Oh... hello. I eat here on Fridays. The sailors' portions are very big..."
  end
label Two
  say "Amphy's light keeps the ships safe while I eat. I'm grateful to it."
  end
