# Route 20 (plan 22 · U5). docs/scripts.md describes the language; the lines are our own, on Black 2's beats (the
# decompilation's res/field/scripts/route_20.s, read for the order of the scenes alone).
#
# VAR_UNOVA_ROUTE_20, the route's count as Black 2 keeps it: 0 the hiker turns back anyone without a Badge; 1 the Basic
# Badge won (aspertia_city_gym.txt, aspertia_city.txt), and he wants a battle; 2 the hiker beaten; 3 Cheren's advice
# on the dark grass and his Pecha Berries, Hugh gone on ahead to Virbank City; (Black 2's 4 is not used) 5 the grunts
# beaten on Virbank's dock, Hugh searching the route for the one who fled this way (virbank_city.txt); 6 that grunt
# beaten and gone to sea, Hugh back to Virbank's docks (VAR_UNOVA_VIRBANK 7). The people: 0 the hiker Jerome, 21 Hugh
# (FLAG_UNOVA_HIDE_HUGH_ROUTE_20), 22 the grunt (FLAG_UNOVA_HIDE_ROUTE_20_GRUNT), 23 Cheren (FLAG_UNOVA_HIDE_CHEREN_ROUTE_20).


script OnEnter
  call UnovaCatchUp
  if var VAR_UNOVA_ROUTE_20 == 5 place 21 153 645 left
  end


# The grunt who fled from Virbank's dock, in the dark grass (person 22, there on 5): the battle, the grunt off to sea,
# and Hugh's guess that Team Plasma travels by boat
script Grunt
  face 22 player
  speaker "Grunt"
  scene "The grunt on Route 20"
  music "unova/eye_plasma"
  say "Ugh, you again?! I really hate it when people won't let things go!"
  battle 22 as "unova_team_plasma_grunt_48"
  say "Hmm... if I keep running this way, there's nobody to back me up. I'd better head for the sea!"
  say "So with that settled... I'm running away again! Goodbye, Virbank City!"
  hide 22
  setflag FLAG_UNOVA_HIDE_ROUTE_20_GRUNT
  music area
  clearflag FLAG_UNOVA_HIDE_HUGH_ROUTE_20
  place 21 155 663 down
  show 21
  face player 21
  speaker "{rival}"
  say "Well, they're good at running away, I'll give them that..."
  say "But she just told us something useful! They must be getting around by boat."
  say "Castelia City has a big port. Maybe they headed there!"
  say "Right! Back to Virbank, {player}. Someone at the docks must know about ships. There was a place to board them, I'm sure of it."
  walk 21 up 2 left 4 fast
  hide 21
  setflag FLAG_UNOVA_HIDE_HUGH_ROUTE_20
  setvar VAR_UNOVA_ROUTE_20 6
  setvar VAR_UNOVA_VIRBANK 7
  clearflag FLAG_UNOVA_HIDE_ROXIE_AND_CAPTAIN
  endscene
  end


# The way east without a Badge (trigger 1, on 0)
script HikerBlocks
  face 0 player
  speaker "Hiker"
  if badge basic goto Ready
  say "Hold it! A kid without a single Gym Badge, heading on past here? Not on my watch!"
  say "Battle the Trainers and Pokémon round here, then battle some more. Come back with a Badge!"
  walk player left 1
  end
label Ready
  setvar VAR_UNOVA_ROUTE_20 1
  call HikerChallenge
  end


# The way east with the Basic Badge (trigger 2, on 1): the hiker tests the new Badge
script HikerChallenge
  face 0 player
  speaker "Hiker"
  say "Hey! That shiny thing is the Basic Badge! Don't let it go to your head. It's a rough world out there!"
  say "Let me show you just how rough!"
  battle 0 as "unova_hiker_jerome"
  say "All right, I'm beaten! With Pokémon you can count on like that, even a kid can hold their own."
  setvar VAR_UNOVA_ROUTE_20 2
  end


# Past the hiker (trigger 0, on 2): Cheren catches up with the player and Hugh at the dark grass, gives Pecha Berries
# for Virbank's Gym, and goes back; Hugh decides to grow stronger than him, and goes on ahead.
script CherensAdvice
  scene "Cheren on Route 20"
  speaker "Cheren"
  say "{player}! Wait!"
  clearflag FLAG_UNOVA_HIDE_HUGH_ROUTE_20
  clearflag FLAG_UNOVA_HIDE_CHEREN_ROUTE_20
  place 21 160 650 down
  place 23 161 650 down
  show 21
  show 23
  move 23 down 8
  walk 21 down 8
  waitmoves
  face player up
  say "Come with me, you two."
  face 23 player
  say "See the darker grass here? Now and then two Pokémon jump out of it at once."
  say "And the Pokémon that live in it can be quite rare. Keep your eyes open."
  say "If you're going to challenge the next Gym, that's Virbank's. These might help."
  give "Pecha Berry" 3
  face 23 21
  say "And some for you too, {rival}. Both of you, do your best! If you need anything, come and find me."
  walk 23 up 8
  hide 23
  setflag FLAG_UNOVA_HIDE_CHEREN_ROUTE_20
  speaker "{rival}"
  face 21 player
  say "Cheren really knows his stuff. And he's fought Team Plasma, too..."
  say "That's it, I've decided! I'm going to get stronger than him!"
  say "And you... do your best! Fill up that Pokédex, and watch my back. Got it?"
  walk 21 right 2 down 8 fast
  hide 21
  setflag FLAG_UNOVA_HIDE_HUGH_ROUTE_20
  setvar VAR_UNOVA_ROUTE_20 3
  endscene
  end


script Hiker
  speaker "Hiker"
  if var VAR_UNOVA_ROUTE_20 >= 2 goto Beaten
  say "The grass round here hides all sorts of Pokémon. Some of them even drop out of the trees!"
  end
label Beaten
  say "You've got a good team there. Mind you look after them on the road to Virbank!"
  end


script Hugh
  speaker "{rival}"
  if var VAR_UNOVA_ROUTE_20 == 5 goto Searching
  say "I have to get stronger than Team Plasma!"
  end
label Searching
  face 21 player
  say "{player}! Help me search Route 20! One of those grunts ran this way, I know it."
  end

  end
