# Saffron City's Magnet Train station (plan 20 · J15, part c). docs/scripts.md describes the language; the lines are
# our own, on HeartGold's beats (scr_seq_T11R0601).
#
# The officer at the gate to the platform makes the ride back to Goldenrod City in Johto, for anyone who has come this
# far: whoever stands in Saffron came by the S.S. Aqua or the train, both a Champion's.


script Officer
  face t11r0601_policeman player
  speaker "Officer"
  ask "The Magnet Train to Goldenrod City in Johto is about to leave. Would you like to board?"
  if no goto Later
  say "All aboard! Please stand clear of the doors."
  fade out 0.6
  text "The Magnet Train glided out of Saffron, floating silently west toward Johto..."
  warp "Goldenrod_MagnetTrainStation" 6 9 down
  fade in
  end
label Later
  say "The trains run all day. Come back whenever you're ready."
  end


# Lt. Surge's street meeting (plan 20, part q): the station's arrival shows him on Wednesdays once his number is in the
# Pokégear (HeartGold's FLAG_HIDE_SAFFRON_TRAIN_STATION_LT_SURGE; the day is ours)
script OnEnter
  setflag FLAG_HIDE_SAFFRON_TRAIN_STATION_LT_SURGE
  if not weekday wednesday end
  if flag FLAG_PHONE_CONTACT_LT_SURGE clearflag FLAG_HIDE_SAFFRON_TRAIN_STATION_LT_SURGE
  end


# Lt. Surge, waiting for a train on Wednesday (ours)
script MeetLtSurge
  face t11r0601_gsleader9 player
  speaker "Lt. Surge"
  random 2
  if result == 1 goto Two
  say "This train's powered by electricity, kid. Every time I ride it, it feels like a little bit of home."
  end
label Two
  say "I'm off to Goldenrod for the day. They say the Magnet Train hits top speed in under ten seconds. Shocking!"
  end
