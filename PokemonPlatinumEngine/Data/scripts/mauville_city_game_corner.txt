script CoinsClerk
  speaker "Clerk"
  say "Welcome to the Mauville Game Corner!"
  if not item "Coin Case" goto NoCase
  choose "Coins are $1,000 for 50. How many would you like?" "50 Coins" "500 Coins" "No, thanks"
  if result == 0 goto Fifty
  if result == 1 goto FiveHundred
  say "Then do enjoy yourself!"
  end
label Fifty
  if not money >= 1000 goto Poor
  gamecorner room 50
  if result == 0 goto Full
  takemoney 1000
  gamecorner add 50
  sound "coin_land"
  say "Thank you! Here are your 50 Coins."
  end
label FiveHundred
  if not money >= 10000 goto Poor
  gamecorner room 500
  if result == 0 goto Full
  takemoney 10000
  gamecorner add 500
  sound "coin_land"
  say "Thank you! Here are your 500 Coins."
  end
label Poor
  say "I'm sorry, you don't have enough money."
  end
label Full
  say "Your Coin Case is full! You must be doing very well."
  end
label NoCase
  say "You'll need a Coin Case to hold your Coins. I believe someone in the house next door has one."
  end


script PrizeCornerTMs
  speaker "Clerk"
  say "We exchange your Coins for prizes. You have {coins} Coins."
  if not item "Coin Case" goto NoCase
  gamecorner prize mauville
  if result == 0 goto Leave
  ask "The {item}, for {price} Coins?"
  if no goto Leave
  gamecorner buy
  if result == 0 goto Few
  sound "select"
  say "Here you go!"
  text "{player} put the {item} away."
  end
label Few
  say "I'm sorry, you don't have enough Coins."
  end
label Leave
  say "Oh, is that so? Come back any time."
  end
label NoCase
  say "Without a Coin Case, you have no Coins to exchange, I'm afraid."
  end


script PokefanM
  speaker "Man"
  if not item "Coin Case" goto NoCase
  if flag FLAG_RECEIVED_20_COINS goto Normal
  gamecorner coins
  if result >= 1 goto Normal
  setflag FLAG_RECEIVED_20_COINS
  gamecorner add 20
  say "No Coins? Luck only lasts so long, you know. Here, take some of mine."
  sound "coin_land"
  text "{player} received 20 Coins!"
label Normal
  say "Mauville has something for everyone. Even me!"
  end
label NoCase
  say "You need a Coin Case to play. The lady next door might help you out."
  end

# The two roulette tables (plan 21 · H12, part o; Emerald's Roulette_EventScript_Table1 and Table2, roulette.c): the left
# takes a Coin a ball, the right three. Sitting down opens the table itself (gamecorner wheel, Models/RouletteWheel.cs):
# the wheel turns, the ball rolls round it and drops into a slot, and a hit pays the wager times its odds. Emerald asks
# first whether to play at the table's wager, and turns away whoever has too few Coins for it.
script Roulette1
  if not item "Coin Case" goto NoCase
  gamecorner coins
  if result < 1 goto Short
  ask "A roulette table. A ball here takes 1 Coin. Will you play?"
  if no end
  fade out 0.3
  gamecorner wheel 1
  fade in 0.3
  call RouletteRecord
  end
label Short
  text "A roulette table. A ball here takes 1 Coin, and you haven't got one."
  end
label NoCase
  text "A roulette wheel. Without a Coin Case there's nothing to wager."
  end


script Roulette2
  if not item "Coin Case" goto NoCase
  gamecorner coins
  if weekday saturday goto Service
  if result < 3 goto Short
  ask "A roulette table. A ball here takes 3 Coins. Will you play?"
  if no end
  fade out 0.3
  gamecorner wheel 3
  fade in 0.3
  call RouletteRecord
  end
label Short
  text "A roulette table. A ball here takes 3 Coins, and you haven't got enough."
  end
label NoCase
  text "A roulette wheel. Without a Coin Case there's nothing to wager."
  end
# The right table's service day (Emerald's ROULETTE_SPECIAL_RATE, sTableMinBets' six; the day is every Saturday, ours,
# where Emerald's PokéNews of the Game Corner calls it): a ball takes six Coins and every hit pays twice as much
label Service
  if result < 6 goto ShortService
  ask "It's service day! A ball here takes 6 Coins today, and every win pays double. Will you play?"
  if no end
  fade out 0.3
  gamecorner wheel 6
  fade in 0.3
  call RouletteRecord
  end
label ShortService
  text "It's service day: a ball here takes 6 Coins today, and you haven't got enough."
  end


# After a sitting: a new best of wins in a row is told (Emerald's GAME_STAT_CONSECUTIVE_ROULETTE_WINS, roulette.c's
# Task_TryIncrementWins; a variable of ours keeps it)
script RouletteRecord
  gamecorner streak
  if result == 0 return
  text "A new record! {player} won {number} times in a row at the roulette."
  return
