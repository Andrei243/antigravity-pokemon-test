# for $10,000; the TM counter exchanges Emerald's five (gamecorner prize mauville). The man by the machines gives 20

script CoinsClerk
  speaker "Clerk"
  say "Welcome to the Mauville Game Corner!"
  if not item "Coin Case" goto NoCase
  choose "Coins are $1,000 for 50. How many would you like?" "50 Coins" "500 Coins" "No, thanks"
  if result == 0 goto Fifty
  if result == 1 goto FiveHundred
  say "Then do enjoy yourself!"
  end
label Fifty
  if not money >= 1000 goto Poor
  gamecorner room 50
  if result == 0 goto Full
  takemoney 1000
  gamecorner add 50
  sound "coin_land"
  say "Thank you! Here are your 50 Coins."
  end
label FiveHundred
  if not money >= 10000 goto Poor
  gamecorner room 500
  if result == 0 goto Full
  takemoney 10000
  gamecorner add 500
  sound "coin_land"
  say "Thank you! Here are your 500 Coins."
  end
label Poor
  say "I'm sorry, you don't have enough money."
  end
label Full
  say "Your Coin Case is full! You must be doing very well."
  end
label NoCase
  say "You'll need a Coin Case to hold your Coins. I believe someone in the house next door has one."
  end


script PrizeCornerTMs
  speaker "Clerk"
  say "We exchange your Coins for prizes. You have {coins} Coins."
  if not item "Coin Case" goto NoCase
  gamecorner prize mauville
  if result == 0 goto Leave
  ask "The {item}, for {price} Coins?"
  if no goto Leave
  gamecorner buy
  if result == 0 goto Few
  sound "select"
  say "Here you go!"
  text "{player} put the {item} away."
  end
label Few
  say "I'm sorry, you don't have enough Coins."
  end
label Leave
  say "Oh, is that so? Come back any time."
  end
label NoCase
  say "Without a Coin Case, you have no Coins to exchange, I'm afraid."
  end


script PokefanM
  speaker "Man"
  if not item "Coin Case" goto NoCase
  if flag FLAG_RECEIVED_20_COINS goto Normal
  gamecorner coins
  if result >= 1 goto Normal
  setflag FLAG_RECEIVED_20_COINS
  gamecorner add 20
  say "No Coins? Luck only lasts so long, you know. Here, take some of mine."
  sound "coin_land"
  text "{player} received 20 Coins!"
label Normal
  say "Mauville has something for everyone. Even me!"
  end
label NoCase
  say "You need a Coin Case to play. The lady next door might help you out."
  end

# The two roulette tables (plan 21 · H12, part k; Emerald's Roulette_EventScript_Table1 and Table2, roulette.c): the left
# takes a Coin a ball, the right three, and a hit pays that times the wager's odds (Models/Roulette.cs). A round is six
# balls; each lands on a square not yet hit, so the odds of what is left grow. The wager is a colour (VAR_TEMP_2 0, its
# row in VAR_TEMP_3), a Pokémon (1, its column in VAR_TEMP_4) or one square (2, both). Ours: the board and the throw
# are menus and the script's die, where Emerald's are a wheel the player throws the ball onto.
script Roulette1
  if not item "Coin Case" goto NoCase
  gamecorner roulette 1
  call Roulette
  end
label NoCase
  text "A roulette wheel. Without a Coin Case there's nothing to wager."
  end


script Roulette2
  if not item "Coin Case" goto NoCase
  gamecorner roulette 3
  call Roulette
  end
label NoCase
  text "A roulette wheel. Without a Coin Case there's nothing to wager."
  end


script Roulette
  speaker none
  gamecorner coins
  text "Place your wager! You have {coins} Coins."
label Ball
  choose "What will you wager on?" "A colour" "A Pokémon" "One square" "Quit"
  if result == 3 end
  setvar VAR_TEMP_2 RESULT
  if result == 1 goto Pokemon
  choose "Which colour?" "Orange" "Green" "Purple"
  setvar VAR_TEMP_3 RESULT
  if var VAR_TEMP_2 == 0 goto Throw
label Pokemon
  choose "Which Pokémon?" "Wynaut" "Azurill" "Skitty" "Makuhita"
  setvar VAR_TEMP_4 RESULT
label Throw
  gamecorner wager
  if result == 0 goto Gone
  gamecorner spin
  if result == 0 goto Broke
  setvar VAR_TEMP_5 RESULT
  if result == 2 goto Hit
  if result == 4 goto Hit
  text "The ball rolled into {roulette:SQUARE}. No luck this time."
  goto Next
label Hit
  text "The ball rolled into {roulette:SQUARE}. It's a hit! {roulette:WON} Coins won!"
label Next
  if var VAR_TEMP_5 >= 3 text "That was the round's last ball. The board is cleared for a new round."
  gamecorner coins
  if result == 0 end
  ask "Wager again? You have {coins} Coins."
  if no end
  goto Ball
label Gone
  text "Every square of that wager has been hit this round. Choose another!"
  goto Ball
label Broke
  text "You don't have enough Coins."
  end
