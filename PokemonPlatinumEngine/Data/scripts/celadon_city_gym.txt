# The Celadon City Gym (plan 19 · K5, part b). docs/scripts.md describes the language; the lines are our own, on
# FireRed's beats (data/maps/CeladonCity_Gym/scripts.inc).
#
# Erika battles for the Rainbow Badge (FLAG_BADGE04_GET is the badge itself, `givebadge rainbow`, which opens
# Strength in the field), counts her seven trainers beaten (FireRed's set_gym_trainers 4) and gives TM19, Giga Drain.


script Erika
  face erika player
  speaker "Erika"
  if badge rainbow goto Won
  scene "Erika"
  say "Hello... Lovely weather, isn't it? I must have dozed off in the sunshine."
  say "I'm Erika, the Leader of this Gym. I study flower arranging, and my Pokémon are all of the Grass type."
  say "Oh, you want to battle me? Very well. I shan't lose, though."
  battle self as "kanto_leader_erika"
  say "Oh... I concede. You are remarkably strong. Please, take the Rainbow Badge."
  givebadge rainbow
  text "{player} received the Rainbow Badge from Erika!"
  say "With the Rainbow Badge, your Pokémon can use Strength outside of battle, to push heavy boulders."
  setflag FLAG_DEFEATED_ERIKA
  defeat "kanto_lass_kay"
  defeat "kanto_lass_lisa"
  defeat "kanto_picnicker_tina"
  defeat "kanto_beauty_bridget"
  defeat "kanto_beauty_tamia"
  defeat "kanto_beauty_lori"
  defeat "kanto_cooltrainer_mary"
label Tm
  say "Please accept this as well. It's a move I'm very fond of."
  give "TM19"
  setflag FLAG_GOT_TM19_FROM_ERIKA
  say "TM19 is Giga Drain. It takes the foe's strength and gives half of it back to your Pokémon."
  endscene
  end
label Won
  if not flag FLAG_GOT_TM19_FROM_ERIKA goto Tm
  say "Losing to you was no disgrace. I must devote myself to my studies of flowers and of battle."
  end


# The statues by the door: who has beaten Erika.
script Statue
  speaker none
  if badge rainbow goto Won
  text "Celadon City Pokémon Gym. Leader: Erika. Winning trainers: {rival}."
  end
label Won
  text "Celadon City Pokémon Gym. Leader: Erika. Winning trainers: {rival}, {player}."
  end
