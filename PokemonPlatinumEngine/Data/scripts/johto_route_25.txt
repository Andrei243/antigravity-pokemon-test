# Route 25 as HeartGold has it three years on (plan 20 · J15, part e). docs/scripts.md describes the language; the
# lines are our own, on HeartGold's beats (scr_seq_R25).
#
# Misty is out on the cape with a young man while her Gym stands without its Leader. Whoever walks onto the cape's way
# in, or speaks to her, spoils the date: the young man takes his leave, and Misty goes home to Cerulean City, where she
# waits in her Gym from then on (FLAG_JOHTO_MISTY_BACK_IN_GYM, a flag of ours, as VAR_JOHTO_MISTY_DATE is: the rulings'
# "HeartGold's Kanto, part e").


script MistysDate
  if flag FLAG_JOHTO_MISTY_BACK_IN_GYM goto Done
  scene "MistysDate"
  setvar VAR_JOHTO_MISTY_DATE 1
  face r25_gsleader11 player
  speaker "Misty"
  say "Huh? Who's there? Can't a Gym Leader have one quiet afternoon on the cape?"
  speaker "Young man"
  say "Er... I think I'd better be going. See you, Misty!"
  hide r25_gsman1
  speaker "Misty"
  say "He ran off! Honestly. And you're a trainer, aren't you? I can tell. You came to Cerulean for my Badge."
  say "Fine. If my afternoon's spoiled anyway, I might as well battle. I'll be waiting at the Gym!"
  setflag FLAG_JOHTO_MISTY_BACK_IN_GYM
  hide r25_gsleader11
  endscene
  end
label Done
  setvar VAR_JOHTO_MISTY_DATE 1
  end


# Suicune's last stop, and its battle (plan 20 · J15, part o; plan 29's owed row). Once it has bounded off from Route 14
# (VAR_UNK_4088 1, johto_route_14.txt), Suicune waits at the head of the cape and Eusine has followed it here (HeartGold's
# FLAG_HIDE_ROUTE_25_SUICUNE and FLAG_HIDE_ROUTE_25_EUSINE cleared by the arrival). HeartGold's trigger on the cape's way in
# brings Eusine over to the player (VAR_UNK_4088 2); Suicune, spoken to, is battled at level 40, as HeartGold's. Beaten or
# caught it is gone for good (FLAG_JOHTO_FOUGHT_SUICUNE, ours) and Eusine with it; run from, it is back on the next visit.
script OnEnter
  setflag FLAG_HIDE_ROUTE_25_SUICUNE
  setflag FLAG_HIDE_ROUTE_25_EUSINE
  # Eusine's second place (plan 20, part q): once Suicune has been fought he stands by the cape's head until spoken to
  # (HeartGold's FLAG_HIDE_ROUTE_25_EUSINE_2; FLAG_JOHTO_EUSINE_PARTED is ours)
  setflag FLAG_HIDE_ROUTE_25_EUSINE_2
  if flag FLAG_JOHTO_FOUGHT_SUICUNE call ShowEusineAfter
  if flag FLAG_JOHTO_FOUGHT_SUICUNE end
  if var VAR_UNK_4088 == 0 end
  clearflag FLAG_HIDE_ROUTE_25_SUICUNE
  clearflag FLAG_HIDE_ROUTE_25_EUSINE
  end

# R25_000: Eusine comes over as the player steps onto the cape
script EusineOnTheCape
  if var VAR_UNK_4088 != 1 end
  if flag FLAG_JOHTO_FOUGHT_SUICUNE end
  scene "Eusine on the cape"
  setvar VAR_UNK_4088 2
  emote r25_minaki exclaim
  face r25_minaki player
  face player r25_minaki
  speaker "Eusine"
  say "{player}! So you followed it here too. Suicune went all the way to the end of the cape, and it won't move from there."
  say "I think it's waiting for someone. And I'm beginning to believe that someone is you."
  say "Go on. It's what you came for. I'll watch from here."
  endscene
  end

script ShowEusineAfter
  if not flag FLAG_JOHTO_EUSINE_PARTED clearflag FLAG_HIDE_ROUTE_25_EUSINE_2
  return


# R25_006: Eusine at his second place, after Suicune (plan 20, part q, our words on HeartGold's beat): he has seen the
# battle and takes his leave
script EusineAfterSuicune
  face r25_minaki_2 player
  speaker "Eusine"
  say "I saw it all from here, {player}. Suicune chose you. I chased it for ten years, and it chose you."
  say "I'm not bitter. Well... a little. But I'm glad it was someone who followed it as far as I did."
  say "I'll keep studying the legendary Pokémon. Perhaps one of them is waiting for me somewhere. Farewell!"
  setflag FLAG_JOHTO_EUSINE_PARTED
  setflag FLAG_HIDE_ROUTE_25_EUSINE_2
  hide self
  end


# R25_005: Eusine, after
script Eusine
  face self player
  speaker "Eusine"
  say "Suicune is waiting at the end of the cape. Go to it, {player}."
  end

# R25_007: Suicune at the head of the cape
script Suicune
  scene "Suicune on the cape"
  music stop
  face player self
  letterbox on 0.4
  # Met from below, it is behind the player's own head from the field's camera: a step back, facing it, shows it
  if facing up walk player back down 1
  camera pan 1421 37 1.2
  camera zoom 1.4 0.8
  effect sparkles 1421 37
  effect ripple 1421 37
  wait 1
  cry "Suicune"
  camera shake 0.5
  emote player exclaim
  speaker none
  text "The wind off the sea fell still as Suicune turned to face {player}!"
  if facing up walk player up 1
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Suicune" 40
  if result == 3 setflag FLAG_JOHTO_CAUGHT_SUICUNE
  if result == 2 goto Fled
  setflag FLAG_JOHTO_FOUGHT_SUICUNE
  setflag FLAG_HIDE_ROUTE_25_SUICUNE
  hide self
  setflag FLAG_HIDE_ROUTE_25_EUSINE
  hide r25_minaki
  end
label Fled
  setflag FLAG_HIDE_ROUTE_25_SUICUNE
  hide self
  text "Suicune bounded away over the sea!"
  end
