# The Celadon Prize Corner as HeartGold has it three years on (plan 20 · J15, part o). docs/scripts.md describes the
# language; the lines are our own, on HeartGold's beats (scr_seq_T07R0501).
#
# The counter exchanges the Game Corner's coins for prizes: plan 06 · R16b's list (gamecorner prize, gamecorner buy),
# the same as Veilstone's and FireRed's Celadon's, and since Johto's leftovers three Pokémon as HeartGold's counter
# offers them (Mr. Mime, Eevee and Porygon at 3,333, 6,666 and 9,999 coins, at level 15: our recollection of its list,
# which isn't in the pinned folders).


# T07R0501_002: the clerk
script Prizes
  speaker "Clerk"
  if not item "Coin Case" goto NoCase
  gamecorner coins
  choose "Welcome to the Prize Corner! You have {coins} coins. What would you like?" "An item" "A Pokémon" "Nothing"
  if result == 1 goto Pokemon
  if result == 2 goto Bye
  gamecorner prize
  if result == 0 goto Bye
  ask "The {item} for {price} coins?"
  if no goto Bye
  gamecorner buy
  if result == 0 goto Poor
  text "{player} received the {item}!"
label Bye
  say "Please come again!"
  end
label Pokemon
  choose "Which Pokémon? You have {coins} coins." "Mr. Mime (3,333)" "Eevee (6,666)" "Porygon (9,999)" "Nothing"
  if result == 0 goto MrMime
  if result == 1 goto Eevee
  if result == 2 goto Porygon
  goto Bye
label MrMime
  gamecorner take 3333
  if result == 0 goto Poor
  givepokemon "Mr. Mime" 15
  text "{player} received Mr. Mime!"
  goto Bye
label Eevee
  gamecorner take 6666
  if result == 0 goto Poor
  givepokemon "Eevee" 15
  text "{player} received Eevee!"
  goto Bye
label Porygon
  gamecorner take 9999
  if result == 0 goto Poor
  givepokemon "Porygon" 15
  text "{player} received Porygon!"
  goto Bye
label NoCase
  say "Prizes are exchanged for coins. The Game Corner next door will give you a Coin Case."
  end
label Poor
  say "I'm sorry, you don't have enough coins."
  end
