# Route 13 (plan 22 · U7, part b). docs/scripts.md describes the language; the lines are our own, on Black 2's beats (the
# decompilation's res/field/scripts/route_13.s, read for the order alone). Black 2's Cobalion at the north end (21, which
# roars as the player comes up and is battled) is plan 22 · U10's and stays hidden.


script OnEnter
  call UnovaCatchUpB
  end


# The treasure hunter (13): a Heart Scale a day
script TreasureHunter
  face 13 player
  speaker "Treasure Hunter"
  if flag FLAG_DAILY_UNOVA_TREASURE_HUNTER goto Done
  say "What am I doing? I'm hunting for treasure! Sometimes things lie buried in the sand and the soil."
  say "Actually, I just found something! But it's the same as one I found before, so you can have this."
  give "Heart Scale"
  setflag FLAG_DAILY_UNOVA_TREASURE_HUNTER
  say "I'll be hunting for treasure here again tomorrow. I might find something else, so stop by if you have time."
  say "I don't find as much as I used to, though..."
  end
label Done
  say "I'll be hunting for treasure here again tomorrow. Stop by if you have time."
  end


# Cobalion at the road's north end (21; plan 22 · U10, part a): once Unova's Hall of Fame is entered it stands there (shown by
# UnovaHallOfFame), roars and is battled at level 65, Black 2's first meeting (route_13.s); caught or beaten it is gone for good
# (FLAG_UNOVA_FOUGHT_COBALION), run from it leaps away and is back the next time the road is come to.
script Cobalion
  if not flag UnovaHallOfFame end
  scene "Cobalion on Route 13"
  music stop
  face player self
  face self player
  letterbox on 0.4
  camera path 684 186 684 190 3
  camera zoom 1.4 0.8
  effect beam 684 190 0.8
  shake self 8
  camera shake 0.8
  cry "Cobalion"
  speaker none
  text "Cobalion let out a ringing cry!\nIts steel-hard gaze holds {player} where they stand!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Cobalion" 65
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_COBALION
  hide self
  text "The road is quiet again. Only the sea breeze is left..."
  end
label Fled
  hide self
  text "Cobalion leapt away beyond the road..."
  end
