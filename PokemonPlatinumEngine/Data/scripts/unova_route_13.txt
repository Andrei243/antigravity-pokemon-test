# The treasure hunter (13): a Heart Scale a day
script TreasureHunter
  face 13 player
  speaker "Treasure Hunter"
  if flag FLAG_DAILY_UNOVA_TREASURE_HUNTER goto Done
  say "What am I doing? I'm hunting for treasure! Sometimes things lie buried in the sand and the soil."
  say "Actually, I just found something! But it's the same as one I found before, so you can have this."
  give "Heart Scale"
  setflag FLAG_DAILY_UNOVA_TREASURE_HUNTER
  say "I'll be hunting for treasure here again tomorrow. I might find something else, so stop by if you have time."
  say "I don't find as much as I used to, though..."
  end
label Done
  say "I'll be hunting for treasure here again tomorrow. Stop by if you have time."
  end
