# The Embedded Tower's Kyogre room (Johto's leftovers). docs/scripts.md describes the language; the lines are our own,
# on HeartGold's beat (scr_seq_D52R0102).
#
# Kyogre waits here once the Blue Orb has been given (route_30_mr_pokemon_house.txt) and until it has been fought
# (FLAG_JOHTO_FOUGHT_KYOGRE, ours): beaten or caught it is gone for good, run from it vanishes (HeartGold's
# FLAG_HIDE_EMBEDDED_TOWER_KYOGRE) and is back on the next visit. Its meeting is plan 29's beat for a Pokémon of
# legend; the scene ends before the battle, so the hold never skips it.


script OnEnter
  setflag FLAG_HIDE_EMBEDDED_TOWER_KYOGRE
  if flag FLAG_JOHTO_FOUGHT_KYOGRE end
  if flag FLAG_JOHTO_FOUGHT_GROUDON clearflag FLAG_HIDE_EMBEDDED_TOWER_KYOGRE
  end

# D52R0102_000: Kyogre
script Kyogre
  scene "Kyogre in the Embedded Tower"
  music stop
  face player self
  letterbox on 0.4
  # Met from below, it is behind the player's own head from the field's camera: a step back, facing it, shows it
  if facing up walk player back down 1
  camera pan 16 11 1.2
  camera zoom 1.4 0.8
  effect ripple 16 11
  effect sparkles 16 11
  wait 1
  cry "Kyogre"
  camera shake 0.5
  emote player exclaim
  speaker none
  text "The Blue Orb in the bag shone deep blue, and a roar like the sea filled the room as Kyogre rose before it!"
  if facing up walk player up 1
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Kyogre" 50
  if result == 2 goto Fled
  setflag FLAG_JOHTO_FOUGHT_KYOGRE
  setflag FLAG_HIDE_EMBEDDED_TOWER_KYOGRE
  hide self
  end
label Fled
  setflag FLAG_HIDE_EMBEDDED_TOWER_KYOGRE
  hide self
  text "Kyogre vanished into the tower's depths!"
  end
