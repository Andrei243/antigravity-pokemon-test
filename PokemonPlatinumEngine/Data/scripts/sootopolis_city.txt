# Sootopolis City (plan 21 · H8, part b). docs/scripts.md describes the language; the lines are our own, on Emerald's
# beats (data/maps/SootopolisCity/scripts.inc).
#
# VAR_SOOTOPOLIS_CITY_STATE is Emerald's: 0 the city as it is before Kyogre wakes, 1 the titans come into the harbour
# (seafloor_cavern_room_9.txt), 2 their clash seen, 3 Wallace gone to Sky Pillar (cave_of_origin_b1f.txt), 4 Wallace
# at Sky Pillar's door, 5 Rayquaza come (part c) and 6 the city calm again. From 1 to 5 four of the townsfolk stand by
# the harbour watching (Emerald's SetBattleSpectators), and the scene of the clash plays the first time the player
# comes (state 1): from the water or from the Pokémon Center, the camera goes to the harbour, the weather breaks into
# Emerald's downpour and drought by turns (FLAG_HOENN_SOOTOPOLIS_CLASH_SEEN, a flag of ours: overlays/sootopolis_city.json's
# zones; Rayquaza's coming clears it, part c), and Groudon and Kyogre strike at each other three times. Steven, spoken
# to, leads the player to the Cave of Origin, where the old man lets them by, and on into it
# (FLAG_STEVEN_GUIDES_TO_CAVE_OF_ORIGIN); afterwards the old man steps aside for a player who comes to him from the
# south, and Steven waits beside the cave's mouth. A save from before this stretch is given its hiding on its first
# arrival (common.HoennChapterTen).
#
# Part c: once Wallace is back from Sky Pillar (state 4) he waits at the cave's mouth between the old man and Steven.
# Once Rayquaza has woken (VAR_SKY_PILLAR_STATE 1, state 5) the first arrival plays its coming (Emerald's
# StartRayquazaScene): the camera to the harbour, the titans gone, Rayquaza over the water to its own song, its roars,
# and the weather clearing (FLAG_SYS_WEATHER_CTRL, FLAG_HOENN_SOOTOPOLIS_CLASH_SEEN and FLAG_LEGENDARIES_IN_SOOTOPOLIS
# cleared, VAR_SKY_PILLAR_STATE 2); Emerald lays rough water under it and warps the player out of the white, which is a
# fade here. From then on (state 5) Steven, Wallace, Archie and Maxie stand by the Gym (Emerald's
# SetOutsideGymObjPos); spoken to, Archie and Maxie each have their say, and once both have they leave
# (FLAG_SOOTOPOLIS_ARCHIE_MAXIE_LEAVE), which opens the Gym and the houses again. Wallace then gives HM07, Waterfall,
# and steps aside from the Gym's door (VAR_SOOTOPOLIS_WALLACE_STATE). Juan's Badge sets state 6
# (sootopolis_city_gym_1f.txt).


script OnEnter
  if not flag FLAG_HOENN_CHAPTER_TEN call HoennChapterTen
  if not flag FLAG_HOENN_CHAPTER_ELEVEN call HoennChapterEleven
  setflag FLAG_VISITED_SOOTOPOLIS_CITY
  if var VAR_SOOTOPOLIS_CITY_STATE == 0 end
  if var VAR_SOOTOPOLIS_CITY_STATE >= 6 end
  # Emerald's SetBattleSpectators: four of the townsfolk by the harbour
  place kiri 13 48 up
  place sootopolis_boy_1 46 32 left
  place sootopolis_ninja_boy 48 41 left
  place sootopolis_woman_1 45 43 left
  if flag FLAG_STEVEN_GUIDES_TO_CAVE_OF_ORIGIN call StevenByTheCave
  if var VAR_SOOTOPOLIS_CITY_STATE == 5 call OutsideGym
  if var VAR_SOOTOPOLIS_CITY_STATE == 1 goto Clash
  if var VAR_SKY_PILLAR_STATE == 1 goto Rayquaza
  end

# Rayquaza comes down over the harbour, the first time the city is come to after its waking on Sky Pillar
label Rayquaza
  scene "Rayquaza comes down over Sootopolis"
  wait 1
  camera pan 31 43 2.5
  wait 0.5
  fade out
  hide sootopolis_groudon
  setflag FLAG_HIDE_SOOTOPOLIS_CITY_GROUDON
  hide sootopolis_kyogre
  setflag FLAG_HIDE_SOOTOPOLIS_CITY_KYOGRE
  clearflag FLAG_HIDE_SOOTOPOLIS_CITY_RAYQUAZA
  show sootopolis_rayquaza
  music "hoenn/rayquaza_appears"
  fade in
  sound "thunder"
  wait 0.6
  cry "Rayquaza"
  camera shake 0.8
  wait 0.8
  cry "Rayquaza"
  camera shake 1
  wait 1.2
  clearflag FLAG_SYS_WEATHER_CTRL
  clearflag FLAG_HOENN_SOOTOPOLIS_CLASH_SEEN
  clearflag FLAG_LEGENDARIES_IN_SOOTOPOLIS
  wait 1.5
  speaker none
  text "Rayquaza's roar rang round the crater. The storm broke apart, and the titans were gone."
  cry "Rayquaza"
  sound "warp"
  hide sootopolis_rayquaza
  setflag FLAG_HIDE_SOOTOPOLIS_CITY_RAYQUAZA
  text "Rayquaza flew off into the clear sky..."
  setvar VAR_SKY_PILLAR_STATE 2
  # Sky Pillar is cracked all over from now on, and Rayquaza waits back on its top (plan 21 · H12, part c)
  setflag FLAG_HOENN_SKY_PILLAR_CRACKED
  camera release 2
  fade out
  call OutsideGym
  music "hoenn/sootopolis"
  fade in
  endscene
  end

# The titans' clash, the first time the city is come to after Kyogre has woken
label Clash
  scene "Groudon and Kyogre clash over Sootopolis"
  wait 1
  camera pan 31 44 2.5
  wait 1
  setflag FLAG_HOENN_SOOTOPOLIS_CLASH_SEEN
  wait 1
  face sootopolis_kyogre left
  face sootopolis_groudon right
  wait 1
  cry "Kyogre"
  walk sootopolis_kyogre onspot left 2 fast
  camera shake 0.8
  wait 0.8
  cry "Groudon"
  walk sootopolis_groudon onspot right 2 fast
  camera shake 0.8
  wait 0.8
  cry "Kyogre"
  walk sootopolis_kyogre onspot left 2 fast
  camera shake 0.8
  wait 1.2
  camera release 2
  setvar VAR_SOOTOPOLIS_CITY_STATE 2
  endscene
  end


# Steven waits beside the cave's mouth once he has brought the player there, until Wallace has gone (Emerald's
# SetEnterCaveOfOriginObjPos)
script StevenByTheCave
  if var VAR_SOOTOPOLIS_CITY_STATE > 4 return
  place sootopolis_steven 32 18 left
  # Wallace back from Sky Pillar, between the two (Emerald's SetExitCaveOfOriginObjPos)
  if var VAR_SOOTOPOLIS_CITY_STATE != 4 return
  place sootopolis_expert 30 18 down
  place sootopolis_wallace 31 18 down
  return


# Once Rayquaza has come: the old man back before the cave, Steven, Archie and Maxie by the Gym, and Wallace before its
# door, or where he stepped aside to (Emerald's SetOutsideGymObjPos, VAR_SOOTOPOLIS_WALLACE_STATE)
script OutsideGym
  place sootopolis_expert 31 18 down
  place sootopolis_steven 29 33 down
  place sootopolis_maxie 33 35 left
  place sootopolis_archie 34 35 left
  if var VAR_SOOTOPOLIS_WALLACE_STATE == 0 place sootopolis_wallace 31 33 down
  if var VAR_SOOTOPOLIS_WALLACE_STATE == 1 place sootopolis_wallace 32 33 down
  if var VAR_SOOTOPOLIS_WALLACE_STATE == 2 place sootopolis_wallace 30 33 down
  return


# Archie and Maxie, both spoken to once Rayquaza has come, go their ways (Emerald's MaxieArchieLeave)
script LeadersLeave
  fade out
  hide sootopolis_maxie
  setflag FLAG_HIDE_SOOTOPOLIS_CITY_MAXIE
  hide sootopolis_archie
  setflag FLAG_HIDE_SOOTOPOLIS_CITY_ARCHIE
  setflag FLAG_SOOTOPOLIS_ARCHIE_MAXIE_LEAVE
  fade in
  return


# The old man before the Cave of Origin's mouth
script Expert
  face self player
  speaker "Old Man"
  if var VAR_SOOTOPOLIS_CITY_STATE >= 2 goto Guided
  say "Beyond here lies the Cave of Origin, where the spirits of Pokémon are said to rest."
  say "Nobody may go in. Please, turn back."
  end
label Guided
  if not flag FLAG_STEVEN_GUIDES_TO_CAVE_OF_ORIGIN goto Waiting
  if var VAR_SOOTOPOLIS_CITY_STATE >= 5 goto Waiting
  say "Steven told me of you. Go on: Wallace is down there, and he has need of you."
  # A player who comes up from the south finds him in the way, and he steps aside (Emerald puts him aside as the city
  # is come to)
  if var PLAYER_X != 31 end
  if var PLAYER_Y != 19 end
  walk self left 1
  face self right
  end
label Waiting
  if var VAR_SOOTOPOLIS_CITY_STATE >= 5 goto Calm
  say "Two Pokémon of legend, raging in our harbour... Only someone strong enough to stand before them may enter the cave now."
  end
label Calm
  say "Rayquaza came down from the sky and the two of them went quiet. The spirits of the cave may rest again."
  say "Still, nobody may go in. I will keep watch here, as I always have."
  end


# Steven, on the harbour's west side once the titans have come
script Steven
  face self player
  speaker "Steven"
  if flag FLAG_SOOTOPOLIS_ARCHIE_MAXIE_LEAVE goto Left
  if var VAR_SOOTOPOLIS_CITY_STATE == 5 goto Calmed
  if not flag FLAG_STEVEN_GUIDES_TO_CAVE_OF_ORIGIN goto Guide
  if var VAR_SOOTOPOLIS_CITY_STATE == 2 goto HelpWallace
  say "Sky Pillar... I've never been there. Where could it be, I wonder?"
  end
label Calmed
  say "So that was Rayquaza... It calmed the two of them just by appearing. I can hardly believe what I saw."
  say "{player}, you did it. You woke it. Hoenn owes you a great deal."
  end
label Left
  say "The leaders of the two teams have gone. I hope they've learned something from all this."
  say "Wallace's mentor Juan leads the Gym here. Go and show him what you can do!"
  end
label HelpWallace
  say "I think you have just what it takes to help him..."
  end
label Guide
  scene "Steven leads to the Cave of Origin"
  say "Those Pokémon... Groudon and Kyogre, the two ancient Pokémon, woken from their long sleep..."
  say "And now they're hurling all their power at each other, and nothing can stop it."
  say "...{player}. If you're here now, I'll take it you're ready to be part of this."
  say "Then there's someone I'd like you to meet. Come with me, please."
  say "Tell me, {player}. Seeing Groudon and Kyogre, do you think Pokémon are things to be afraid of?"
  say "They're not. Pokémon are so much more than that... But why am I asking you? You know that already."
  fade out
  place sootopolis_expert 30 18 right
  place sootopolis_steven 32 18 left
  place player 31 18 up
  fade in
  speaker "Steven"
  say "Here we are. Inside, you'll find a man named Wallace."
  say "I think you have just what it takes to help him..."
  setflag FLAG_STEVEN_GUIDES_TO_CAVE_OF_ORIGIN
  walk player up 1
  warp "CaveOfOrigin_Entrance" 9 20 up
  endscene
  end

script Archie

  say "......"

script Wallace
  face self player
  speaker "Wallace"
  if var VAR_SOOTOPOLIS_CITY_STATE == 4 goto SkyPillar
  if flag FLAG_RECEIVED_HM_WATERFALL goto Gym
  if flag FLAG_SOOTOPOLIS_ARCHIE_MAXIE_LEAVE goto Waterfall
  say "{player}... The leaders of Team Magma and Team Aqua didn't mean any harm, I think."
  say "It wouldn't hurt to hear them out."
  end
label SkyPillar
  say "{player}, haven't you climbed Sky Pillar yet? Rayquaza is our only hope. Please, hurry!"
  end
label Waterfall
  scene "Wallace gives Waterfall"
  say "{player}, thank you. Thanks to you and Rayquaza, Sootopolis and all of Hoenn are safe."
  say "Please, take this as a token of my gratitude."
  give "HM07"
  setflag FLAG_RECEIVED_HM_WATERFALL
  say "That holds Waterfall. With the Rain Badge from this city's Gym, your Pokémon can climb waterfalls with it."
  say "The Gym is open again. My mentor Juan is waiting inside. Show him what you showed me!"
  # Out of the way of the Gym's door: aside to the west for a player who spoke to him from the east (he stands before
  # it while VAR_SOOTOPOLIS_WALLACE_STATE is 0, as the city's arrival puts him)
  place self 31 33 down
  if var PLAYER_X == 32 goto West
  walk self right 1
  face self down
  setvar VAR_SOOTOPOLIS_WALLACE_STATE 1
  endscene
  end
label West
  walk self left 1
  face self down
  setvar VAR_SOOTOPOLIS_WALLACE_STATE 2
  endscene
  end
label Gym
  say "Juan is the one who taught me everything. Even now, I find myself dazzled by his battles."
  end


# Kiri, who gives two berries a day (Emerald's daily FLAG_DAILY_SOOTOPOLIS_RECEIVED_BERRY: one of six, then a Figy or
# an Iapapa Berry); while the titans clash she watches the harbour
script Kiri
  speaker "Kiri"
  if var VAR_SOOTOPOLIS_CITY_STATE < 2 goto Berry
  if var VAR_SOOTOPOLIS_CITY_STATE >= 6 goto Berry
  face self player
  if var VAR_SOOTOPOLIS_CITY_STATE == 5 goto Calm
  say "Big, big Pokémon are fighting! I'm scared..."
  face self up
  end
label Calm
  say "A green Pokémon came down from the sky and the big ones went away! Was it a dream?"
  face self up
  end
label Berry
  face self player
  if flag FLAG_DAILY_SOOTOPOLIS_RECEIVED_BERRY goto Given
  say "Hello! I'm Kiri. Here, you can have one of these!"
  random 6
  if result == 0 give "Pomeg Berry"
  if result == 1 give "Kelpsy Berry"
  if result == 2 give "Qualot Berry"
  if result == 3 give "Hondew Berry"
  if result == 4 give "Grepa Berry"
  if result == 5 give "Tamato Berry"
  setflag FLAG_DAILY_SOOTOPOLIS_RECEIVED_BERRY
  say "And this one too!"
  random 2
  if result == 0 give "Figy Berry"
  if result == 1 give "Iapapa Berry"
  say "When you make a wish, what do you wish for?"
  end
label Given
  ask "Do you like the season you were born in?"
  if no goto Doesnt
  say "Me too! But I love autumn most of all."
  end
label Doesnt
  say "Oh. Well, it doesn't really matter."
  end
