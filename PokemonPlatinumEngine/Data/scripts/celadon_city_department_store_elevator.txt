# The Celadon Department Store's lift (plan 19 · K5, part b). docs/scripts.md describes the language; the lines are
# our own, on FireRed's beats (data/maps/CeladonCity_DepartmentStore_Elevator/scripts.inc).
#
# FireRed's car remembers the floor it stopped at (VAR_ELEVATOR_FLOOR) and its door leads there (MAP_DYNAMIC). Here
# the door leads to the ground floor, and the panel rides the car and puts the player out on the floor chosen, before the lift's
# door there: the ride is the original's set piece (`anim elevator`), up or down from the floor it last stopped at
# (VAR_ELEVATOR_FLOOR, the floor's number, 1 to 5; nought, never ridden, is the ground floor).


script Panel
  speaker none
  if var VAR_ELEVATOR_FLOOR == 0 setvar VAR_ELEVATOR_FLOOR 1
  choose "Which floor?" "5F" "4F" "3F" "2F" "1F" "Stay here"
  if result == 5 end
  setvar VAR_TEMP_1 5
  if result == 1 setvar VAR_TEMP_1 4
  if result == 2 setvar VAR_TEMP_1 3
  if result == 3 setvar VAR_TEMP_1 2
  if result == 4 setvar VAR_TEMP_1 1
  if var VAR_TEMP_1 == VAR_ELEVATOR_FLOOR goto Out
  if var VAR_TEMP_1 < VAR_ELEVATOR_FLOOR goto Down
  anim elevator up
  goto Ride
label Down
  anim elevator down
label Ride
  setvar VAR_ELEVATOR_FLOOR VAR_TEMP_1
label Out
  if var VAR_ELEVATOR_FLOOR == 5 warp "CeladonCity_DepartmentStore_5F" 6 2 down
  if var VAR_ELEVATOR_FLOOR == 4 warp "CeladonCity_DepartmentStore_4F" 6 2 down
  if var VAR_ELEVATOR_FLOOR == 3 warp "CeladonCity_DepartmentStore_3F" 6 2 down
  if var VAR_ELEVATOR_FLOOR == 2 warp "CeladonCity_DepartmentStore_2F" 6 2 down
  if var VAR_ELEVATOR_FLOOR == 1 warp "CeladonCity_DepartmentStore_1F" 6 2 down
  end
