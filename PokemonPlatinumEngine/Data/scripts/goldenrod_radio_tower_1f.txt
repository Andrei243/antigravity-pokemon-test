# The Radio Tower's ground floor in Goldenrod City (plan 20 · J7). docs/scripts.md describes the language; the lines
# and the questions are our own, on HeartGold's beats (scr_seq_D23R0101_002): five questions answered yes or no, all
# right in a row, win the Pokégear's Radio Card (FLAG_GOT_RADIO_CARD); one wrong ends the quiz until it is tried again.
#
# Team Rocket's takeover (plan 20 · J11, part b; VAR_SCENE_ROCKET_TAKEOVER, which goldenrod.txt describes): the floor's
# arrival keeps the staff off it from 2 to 4 (FLAG_HIDE_ROCKET_TAKEOVER_2) and puts a grunt on the stairs at 2 and 3
# (FLAG_HIDE_ROCKET_TAKEOVER_6), as HeartGold's D23R0101_008 does. He lets nobody up who isn't in uniform; in the
# uniform borrowed under the city (3) he steps aside, and as he does the red-haired boy walks in at the door, sees
# through the disguise and says so, and the grunt battles the player and runs upstairs to warn the others (4). HeartGold
# plays the boy's part from a trigger on the stairs once the grunt has stepped aside; here it follows the grunt's own
# words, so the one scene is played whichever way the player stands.


# D23R0101_008
script OnEnter
  setflag FLAG_HIDE_ROCKET_TAKEOVER_6
  setflag FLAG_HIDE_ROCKET_TAKEOVER_2
  setflag FLAG_HIDE_RADIO_TOWER_RIVAL
  if var VAR_SCENE_ROCKET_TAKEOVER == 2 clearflag FLAG_HIDE_ROCKET_TAKEOVER_6
  if var VAR_SCENE_ROCKET_TAKEOVER == 3 clearflag FLAG_HIDE_ROCKET_TAKEOVER_6
  if var VAR_SCENE_ROCKET_TAKEOVER < 2 clearflag FLAG_HIDE_ROCKET_TAKEOVER_2
  if var VAR_SCENE_ROCKET_TAKEOVER == 5 clearflag FLAG_HIDE_ROCKET_TAKEOVER_2
  end


# D23R0101_000: the receptionist at the desk
script Receptionist
  face d23r0101_gswoman6 player
  speaker "Receptionist"
  if var VAR_SCENE_ROCKET_TAKEOVER == 3 goto Uniform
  if var VAR_SCENE_ROCKET_TAKEOVER == 2 goto Closed
  if var VAR_SCENE_ROCKET_TAKEOVER == 4 goto Closed
  say "Welcome to the Radio Tower! Do look around the ground floor. The staff upstairs are hard at work on today's shows."
  end
label Closed
  say "I'm sorry... There are no visitors today. None at all. Please, just go home."
  end
label Uniform
  say "Oh! One of you... I've done as I was told. I'm sending every visitor home."
  end


# D23R0101_005 and _009: the grunt on the stairs, and the red-haired boy who sees through the uniform
script Grunt
  face d23r0101_rocketm player
  speaker "Grunt"
  if var VAR_SCENE_ROCKET_TAKEOVER == 3 goto Newcomer
  say "Stop right there. Only Team Rocket in uniform go up these stairs. Beat it, kid!"
  end
label Newcomer
  scene "The red-haired boy at the Radio Tower"
  say "Oh, you're one of us? Must be new. Go on up, they're waiting for you."
  if var PLAYER_Y == 5 walk d23r0101_rocketm down 1
  if var PLAYER_Y == 6 walk d23r0101_rocketm left 1
  clearflag FLAG_HIDE_RADIO_TOWER_RIVAL
  show radio_tower_rival
  place radio_tower_rival 2 14 up
  music "johto/rival"
  speaker "{rival}"
  say "Team Rocket. Always in a gang. They're cowards, every one of them."
  walk radio_tower_rival up 8 fast
  if var PLAYER_Y == 5 walk radio_tower_rival right 19 up 1
  if var PLAYER_Y == 6 walk radio_tower_rival right 20
  face player radio_tower_rival
  emote radio_tower_rival exclaim
  say "...What? It's you. In that uniform? Are you serious?"
  say "You look ridiculous. Nobody in their right mind would take you for one of them."
  text "{player} pulled off the borrowed uniform."
  emote d23r0101_rocketm exclaim
  face d23r0101_rocketm player
  speaker "Grunt"
  say "Hey! You're no newcomer at all! Nobody fools Team Rocket!"
  battle d23r0101_rocketm as "johto_team_rocket_grunt_3"
  speaker "Grunt"
  say "Too strong... I've got to warn the others!"
  if var PLAYER_Y == 5 walk d23r0101_rocketm up 1 fast
  if var PLAYER_Y == 6 walk d23r0101_rocketm right 1 fast
  hide d23r0101_rocketm
  setflag FLAG_HIDE_ROCKET_TAKEOVER_6
  face radio_tower_rival player
  speaker "{rival}"
  say "I didn't come here for Team Rocket. I came for Lance, the one who took me apart in their hideout."
  say "I'll beat him. And after him, you. Stay out of my way until then."
  if var PLAYER_Y == 5 walk radio_tower_rival down 1 left 19 down 8 fast
  if var PLAYER_Y == 6 walk radio_tower_rival left 20 down 8 fast
  hide radio_tower_rival
  setflag FLAG_HIDE_RADIO_TOWER_RIVAL
  setvar VAR_SCENE_ROCKET_TAKEOVER 4
  music area
  endscene
  end


script Quiz
  face d23r0101_gswoman6_3 player
  speaker "Receptionist"
  if flag FLAG_GOT_RADIO_CARD goto Won
  ask "It's quiz time at the Radio Tower! Answer five questions right and win a Radio Card for your Pokégear. Ready?"
  if no goto NotNow
  ask "Question one! Can a Pokémon learn new moves as it levels up?"
  if no goto Wrong
  sound "select"
  ask "Question two! Is Chikorita a Grass-type Pokémon?"
  if no goto Wrong
  sound "select"
  ask "Question three! Will an Antidote wake a Pokémon that is fast asleep?"
  if yes goto Wrong
  sound "select"
  ask "Question four! Do Fire-type moves hit Grass-type Pokémon especially hard?"
  if no goto Wrong
  sound "select"
  ask "Last question! Can you throw a Poké Ball to catch another trainer's Pokémon?"
  if yes goto Wrong
  sound "select"
  say "Five out of five! Congratulations! Here's your prize: a Radio Card for your Pokégear!"
  setflag FLAG_GOT_RADIO_CARD
  fanfare keyitem
  text "{player}'s Pokégear can now tune in to the radio!"
  say "Turn the dial to find our stations. We broadcast all over Johto, day and night!"
  end
label Wrong
  sound "error"
  say "Oh, I'm sorry! That's not right. Have another go any time you like."
  end
label NotNow
  say "No? Come back when you're feeling clever!"
  end
label Won
  say "Enjoying the radio? Try every station on the dial. There's always something on!"
  end


# The lucky number draw's counter (plan 42 · PS3; HeartGold's D23R0101 counter, our words): once a week the man of the
# Lucky Number Show matches the week's number (the one the Pokégear's radio reads out, Lottery.WeeklyTicket) against the
# ID numbers of the player's Pokémon, on the team and in the boxes, digit by digit from the right. The prizes are
# HeartGold's, which are Jubilife's five. FLAG_WEEKLY_CLAIMED_LUCKY_NUMBER goes when a new week begins (DailyEvents).
script LuckyNumber
  face self player
  speaker "Man"
  if flag FLAG_WEEKLY_CLAIMED_LUCKY_NUMBER goto NextWeek
  say "Hi there! I work on the Lucky Number Show. Every week we draw a lucky number and read it out on the radio."
  ask "Shall I check this week's number against your Pokémon's ID numbers?"
  if no goto NotToday
  setflag FLAG_WEEKLY_CLAIMED_LUCKY_NUMBER
  say "This week's lucky number is {luckynumber}! Let me look through your Pokémon's ID numbers..."
  lottery weekly
  if result == 5 goto LuckyFive
  if result == 4 goto LuckyFour
  if result == 3 goto LuckyThree
  if result == 2 goto LuckyTwo
  if result == 1 goto LuckyOne
  say "Aw, no luck. Not a single digit matched this week."
  say "There's a new number every week, though. Keep your radio tuned in!"
  end
label LuckyOne
  call LuckyWhose
  say "The last digit matches! That's our fifth prize."
  give "Ultra Ball"
  goto LuckyThanks
label LuckyTwo
  call LuckyWhose
  say "The last two digits match! That's our fourth prize."
  give "PP Up"
  goto LuckyThanks
label LuckyThree
  call LuckyWhose
  say "The last three digits match! That's our third prize."
  give "Exp. Share"
  goto LuckyThanks
label LuckyFour
  call LuckyWhose
  say "Four digits match! That's our second prize!"
  give "Max Revive"
  goto LuckyThanks
label LuckyFive
  call LuckyWhose
  say "All five digits match! I can hardly believe it! That's our first prize!"
  give "Master Ball"
  goto LuckyThanks
label LuckyThanks
  speaker "Man"
  say "Congratulations! Tune in again next week for a new number."
  end
label NextWeek
  say "You've had your go this week. A new lucky number goes out on the radio next week!"
  end
label NotToday
  say "No? Well, the number's good all week. Come back any time!"
  end


# Whose ID number won: on the team, or in the boxes.
script LuckyWhose
  lottery boxed
  if result == 1 goto Boxed
  say "Oh! The ID number of {winner}, on your team, matches!"
  return
label Boxed
  say "Oh! The ID number of {winner}, in your PC Boxes, matches!"
  return
