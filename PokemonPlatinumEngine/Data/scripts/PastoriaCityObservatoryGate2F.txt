# The Great Marsh's observatory in Pastoria City, upper floor (plan 01 · M11, part b), on the original's plan.
# docs/scripts.md describes the language; the lines are our own, on the original's beats
# (scripts_pastoria_city_observatory_gate_2f.s).
#
# Three pairs of binoculars look out over the marsh, $100 a look. The original swings the picture over the marsh and
# shows a Pokémon that lives where it comes to rest (GreatMarshBinoculars_GetMonSpecies); here a look names the
# Pokémon the day has brought to one of the marsh's six areas, drawn by the script's die ({marsh:1} to {marsh:6}:
# plan 06 · R13's daily Pokémon, which is what the binoculars are for). The view itself waits for plan 09's kit.


script Binoculars
  ask "A pair of binoculars, pointed out over the Great Marsh. Put $100 in?"
  if no end
  takemoney 100
  if result == 0 goto Short
  sound "select"
  fade out 0.3
  fade in 0.3
  random 6
  if result == 1 goto Area2
  if result == 2 goto Area3
  if result == 3 goto Area4
  if result == 4 goto Area5
  if result == 5 goto Area6
  text "The glass swings out over Area 1..."
  text "A {marsh:1} lifts its head out of the reeds and looks about."
  end
label Area2
  text "The glass swings out over Area 2..."
  text "A {marsh:2} is wading through the shallows."
  end
label Area3
  text "The glass swings out over Area 3..."
  text "A {marsh:3} peers out from the tall grass."
  end
label Area4
  text "The glass swings out over Area 4..."
  text "A {marsh:4} is splashing about in the mud."
  end
label Area5
  text "The glass swings out over Area 5..."
  text "A {marsh:5} darts across a patch of open ground."
  end
label Area6
  text "The glass swings out over Area 6..."
  text "A {marsh:6} is dozing in the sun by the tram's rails."
  end
label Short
  text "There isn't enough money for that..."
  end

# The photo tour (plan 44 · SD5, SD6; the user's answers 47a and 64a): its guide, Maribel Orr, stands by the binoculars
# once the National Pokédex is had (decision 7). She takes the player out along one of her six courses (each opened by
# the report's total), shows the report, and hands over her rewards as the report fills. A ride's photographs are judged
# into the report as it ends, the script hearing the ride's total.

script TourGuide
  if nationaldex goto Open
  say "I take trainers out along a river in a little cart, camera in hand, to photograph Pokémon going about their day."
  say "The tour is only for those with a National Pokédex, though. Come back once yours has grown."
  end
label Open
  say "Welcome to the photo tour! I'm Maribel Orr, and I judge every photograph that comes back from the rail."
  say "Aim as the cart rolls along, toss fruit to draw them close, and blow the whistle to turn their heads."
  choose "What would you like to do?" "RIDE A COURSE" "SEE THE REPORT" "NOTHING"
  if result == 1 goto Report
  if result == 2 goto Later
  choose "Which course shall we ride?" "REED RIVER" "DEEP WOOD" "REEF SHORE" "FURTHER AFIELD" "NOT TODAY"
  if result == 0 goto Reed
  if result == 1 goto Wood
  if result == 2 goto Reef
  if result == 4 goto Later
  choose "Further afield, then. Which one?" "CAVERN" "HIGH PASS" "NIGHT GARDEN" "NOT TODAY"
  if result == 0 goto Cavern
  if result == 1 goto Pass
  if result == 2 goto Garden
  goto Later
label Reed
  tour ride "reed_river"
  goto Rode
label Wood
  tour open "deep_wood"
  if result == 0 goto Shut
  tour ride "deep_wood"
  goto Rode
label Reef
  tour open "reef_shore"
  if result == 0 goto Shut
  tour ride "reef_shore"
  goto Rode
label Cavern
  tour open "cavern"
  if result == 0 goto Shut
  tour ride "cavern"
  goto Rode
label Pass
  tour open "high_pass"
  if result == 0 goto Shut
  tour ride "high_pass"
  goto Rode
label Garden
  tour open "night_garden"
  if result == 0 goto Shut
  say "The garden is at its best after dark, so we'll go out once the lamps are lit."
  tour ride "night_garden"
  goto Rode
label Rode
  if result == 0 goto Nothing
  say "Back already! I've judged your film: {result} points all told."
  say "Every best photograph goes into my report. Rain and the hour change what the residents get up to, so ride again!"
  goto Rewards
label Nothing
  say "Not a single photograph I could score... Never mind. Get them big, in the middle and facing you next time."
  end
label Rewards
  tour reward
  if result == 1 goto Rewards
  tour points
  say "My report stands at {result} points. Bring me more and there'll be more thanks in it for you."
  end
label Report
  tour report
  say "Every page is yours to fill. A rare moment, caught just right, is worth four stars!"
  end
label Shut
  say "That course opens once my report has more points in it. Bring me good photographs from the courses that are open!"
  end
label Later
  say "The cart waits whenever you do."
  end
