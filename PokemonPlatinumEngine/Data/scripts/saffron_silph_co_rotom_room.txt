# Silph Co.'s research floor, Rotom's room, in HeartGold's Kanto (plan 20 · J15, part n). docs/scripts.md describes
# the language; the lines are our own, on HeartGold's beats (scr_seq_T11R0702).
#
# Five appliances stand in a row. Each sends the first Rotom of the team into itself, with the appliance's move, as the
# Rotom Catalog does (`team rotom`, plan 06 · R29's rule: the move of the appliance it leaves is given up for the new
# one; with no such move and four moves known, the new one isn't learned and a line says so). An appliance with the
# Rotom in it already lets it out again, back into its own form. The room tells the lift that the player is upstairs
# (VAR_JOHTO_SILPH_LIFT_FLOOR, a variable of ours).


script OnEnter
  setvar VAR_JOHTO_SILPH_LIFT_FLOOR 1
  end

script Microwave
  speaker none
  text "It's a microwave oven, plugged in and humming faintly."
  if not has "Rotom" end
  ask "Will you let your Rotom go into the microwave oven?"
  if no end
  team rotom heat
  if result == 0 goto MicrowaveInside
  if result == 2 goto MicrowaveFull
  cry "Rotom-Heat"
  text "Rotom went into the microwave oven! Rotom learned Overheat!"
  end
label MicrowaveFull
  cry "Rotom-Heat"
  text "Rotom went into the microwave oven! It wanted to learn Overheat, but it knows four moves already."
  end
label MicrowaveInside
  ask "Rotom is in the microwave oven already. Will you let it out?"
  if no end
  team rotom plain
  cry "Rotom"
  text "Rotom came out of the microwave oven and went back to its own form!"
  end
