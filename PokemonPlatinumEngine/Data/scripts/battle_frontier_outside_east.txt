# The Battle Frontier island's east half (plan 21 · H12, part b). docs/scripts.md describes the language; the lines are
# our own, on Emerald's beats (data/maps/BattleFrontier_OutsideEast/scripts.inc).
#
# The strange tree by the water is Emerald's Sudowoodo (FLAG_HIDE_BATTLE_FRONTIER_SUDOWOODO). Watered with the Wailmer
# Pail it jumps up and battles at level 40; beaten or caught it is gone for good, run from it stays.


script Sudowoodo
  speaker none
  if not item "Wailmer Pail" goto Wiggles
  scene "The strange tree of the Battle Frontier"
  sound "bump"
  text "It's an odd tree. It is wiggling about!"
  ask "Water the tree with the Wailmer Pail?"
  if no goto NotNow
  text "{player} watered the tree with the Wailmer Pail!"
  sound "use_water_sport"
  text "The tree hated the water! It leapt up and attacked!"
  wildbattle "Sudowoodo" 40
  if result == 2 goto Fled
  hide self
  setflag FLAG_HIDE_BATTLE_FRONTIER_SUDOWOODO
  endscene
  end
label NotNow
  endscene
  end
label Fled
  endscene
  end
label Wiggles
  text "It's an odd tree. It is wiggling about, though there is no wind."
  end


# The halls' doors (plan 21 · H12, part m; docs/mechanics/rulings.md, "The Battle Frontier island's halls"): as the west
# half's (battle_frontier_outside_west.txt), each hall's attendant crosses a holder of the Frontier Pass to the hall's
# lobby at Sinnoh's Battle Frontier, and VAR_HOENN_FRONTIER_HALL brings the Hoenn guide's ferry back to this door.

script TowerDoor
  if not flag FLAG_RECEIVED_FRONTIER_PASS goto NoPass
  speaker "Attendant"
  say "Welcome to the Battle Tower! Its challenges are held at the Battle Frontier in Sinnoh."
  ask "Shall we take you there? Your team comes along, and your records stay as they are."
  if no goto Later
  setvar VAR_HOENN_FRONTIER_HALL 1
  fade out 0.8
  text "{player} crossed with the attendant to the Battle Frontier in Sinnoh..."
  travel to "BattleTower" 15 19 up
  fade in 0.8
  end
label Later
  say "Whenever you're ready. I'll be at this door."
  end
label NoPass
  text "The door is shut. A sign says that only holders of a Frontier Pass may enter."
  end
