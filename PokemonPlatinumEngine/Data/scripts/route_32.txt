# Route 32 (plan 20 · J4). docs/scripts.md describes the language; the lines are our own, on HeartGold's beats
# (scr_seq_R32).
#
# The man at the route's head (HeartGold's R32_002, the area's trigger 0, on VAR_UNK_408D 0): with no Badge he sends
# the player back to Violet City's Gym; with the Zephyr Badge he gives a Miracle Seed and lets them by for good.


script BadgeCheck
  face r32_gsmiddleman1 player
  call SeedMan
  if badge zephyr end
  walk player up 1
  end


script SeedMan
  speaker "Man"
  if flag FLAG_GOT_MIRACLE_SEED_ON_ROUTE_32 goto After
  if badge zephyr goto Seed
  say "Hold on there! Have you been to the Gym in Violet City yet?"
  say "The road south gets rough. Trainers without a Badge to their name tend to come back in tears. Go and earn one first!"
  return
label Seed
  say "Oh! That's the Zephyr Badge, isn't it? Falkner's? Then you're ready for the road south."
  say "Here, take this for your trouble. A little gift from me."
  give "Miracle Seed"
  setflag FLAG_GOT_MIRACLE_SEED_ON_ROUTE_32
  setvar VAR_UNK_408D 1
  say "A Grass-type Pokémon that holds it hits harder with its Grass moves. Good luck on the road!"
  return
label After
  say "The road goes on south to Union Cave, and Azalea Town beyond. Mind the fishermen, they don't like being disturbed."
  return

# Picnicker Liz and Fisherman Ralph battle as every trainer does; beaten, each asks for the player's number for the
# Pokégear's phone, and once it is registered calls back for a rematch on their day (plan 20 · J3; phone.txt).
script Liz
  if not defeated self goto Fight
  if rematch self goto Rematch
  if flag FLAG_PHONE_CONTACT_LIZ goto Known
  speaker "Liz"
  say "That was fun! Hey, let's swap numbers. I love a good chat, and I know all the gossip!"
  ask "Register Liz's number in your Pokégear?"
  if no goto Refused
  addcontact liz
  say "Yay! I'll call you all the time. Well, maybe not all the time."
  end
label Refused
  say "Aww. Ask me again whenever you like!"
  end
label Known
  trainerline after
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end

script Ralph
  if not defeated self goto Fight
  if rematch self goto Rematch
  if flag FLAG_PHONE_CONTACT_RALPH goto Known
  speaker "Ralph"
  say "You've got a strong team. I'd like a rematch one of these days. What's your number?"
  ask "Register Ralph's number in your Pokégear?"
  if no goto Refused
  addcontact ralph
  say "Good. I'll ring you when the fish are biting and my Pokémon are ready."
  end
label Refused
  say "Fair enough. I'll be here by the water either way."
  end
label Known
  trainerline after
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end

# Coming to the route (HeartGold's R32_004, the map's own script as it loads): Frieda of Friday is here on Fridays
# alone. The photographer HeartGold has here on some days after the Plain Badge waits for the photo album (plan 20).
script OnEnter
  setflag FLAG_HIDE_CAMERON
  setflag FLAG_UNK_208
  if weekday friday clearflag FLAG_UNK_208
  end

# Frieda of Friday (HeartGold's R32_005): on her day she gives a Poison Barb, once, and counts herself among the siblings
# met (VAR_NUM_MET_WEEKDAY_SIBLINGS); the ribbon she gives once all seven are met is the siblings' last chapter's.
script Frieda
  speaker "Frieda"
  if flag FLAG_GOT_POISON_BARB_FROM_FRIEDA goto Again
  if not weekday friday goto NotToday
  say "Yay! It's Friday! I'm Frieda of Friday. Nice to meet you! Here, I want you to have this."
  give "Poison Barb"
  setflag FLAG_GOT_POISON_BARB_FROM_FRIEDA
  addvar VAR_NUM_MET_WEEKDAY_SIBLINGS 1
  say "A Pokémon that holds it hits harder with Poison-type moves. Isn't Friday just the best day?"
  end
label Again
  say "Fridays are the best! The weekend's nearly here, and everyone's smiling."
  end
label NotToday
  say "Huh? Today isn't Friday. I only come out on Fridays!"
  end
