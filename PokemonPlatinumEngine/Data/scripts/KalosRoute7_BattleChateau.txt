# The Battle Château on Route 7 (plan 23, Kalos's last places; plan 37's house of titles, its first part). The lines are
# our own, and so are the house's people and titles: none of X and Y's noble ranks is here. The player's own team battles
# at its own levels and is paid as after any trainer's battle; a battle lost costs nothing and the team is made whole.
#
# Who is in today is drawn as the hall is first come into each day (FLAG_DAILY_KALOS_CHATEAU_DRAWN): one of three
# guests in the west place (VAR_KALOS_CHATEAU_GUEST_1: Mirabelle, Tristan, Celeste), one of three stronger ones in the
# east place once the house has made the player a friend (VAR_KALOS_CHATEAU_GUEST_2, title 3 and on: Octave, Isaure,
# Corentin), and once Kalos's Hall of Fame is entered the Leader whose day it is in the middle (Viola on Mondays to
# Olympia on Sundays, Wulfric beside her). Each guest is shown by a flag of ours (FLAG_KALOS_CHATEAU_HAS_<WHO>) and
# battles once a day (FLAG_DAILY_KALOS_CHATEAU_BEAT_<WHO>). A win counts toward the house's titles
# (VAR_KALOS_CHATEAU_POINTS: a guest of the west place 1, of the east 2, a Leader 3), which the butler hands out at
# thresholds of ours (VAR_KALOS_CHATEAU_TITLE 0 to 7: Visitor, Guest, Regular, Friend of the House, Honoured Guest,
# Patron, Steward, Master of the House). Plan 37 · FC4 builds the rest: the guests' levels by the title over the player's
# highest, other regions' Leaders and Champions, the title on the Trainer Card. The Leaders' teams here are Kalos's
# Leaders' rematch teams (kalos_<who>_rematch, plan 41 · EG5's parity): met as every rematch is, a region gone past lifts them.
# Plan 37 · FC4 (Models/Facilities/TitlesHouse.cs, `facility titles ...`): the members bring three of the later sets at a
# level their title sets over the highest of the player's team, drawn by the day's seed (VAR_KALOS_CHATEAU_SEED), and a
# visitor of another finished region stands by the north wall each day (VAR_KALOS_CHATEAU_VISITOR, shown by
# FLAG_KALOS_CHATEAU_HAS_VISITOR, once a day by FLAG_DAILY_KALOS_CHATEAU_BEAT_VISITOR): a Leader, an Elite Four member
# from the Steward's title, a Champion at the Master's.


script OnEnter
  if not flag FLAG_DAILY_KALOS_CHATEAU_DRAWN call Draw
  call Show
  end


# The day's draw: a guest for each of the two places
script Draw
  setflag FLAG_DAILY_KALOS_CHATEAU_DRAWN
  random 3
  setvar VAR_KALOS_CHATEAU_GUEST_1 RESULT
  random 3
  setvar VAR_KALOS_CHATEAU_GUEST_2 RESULT
  # Plan 37 · FC4: the day's seed, which the members' teams and the day's visitor are drawn by, the same all day
  random 65536
  setvar VAR_KALOS_CHATEAU_SEED RESULT
  facility titles visitor
  return


# Who stands in the hall today, by the draw, the title and the day of the week
script Show
  clearflag FLAG_KALOS_CHATEAU_HAS_MIRABELLE
  clearflag FLAG_KALOS_CHATEAU_HAS_TRISTAN
  clearflag FLAG_KALOS_CHATEAU_HAS_CELESTE
  clearflag FLAG_KALOS_CHATEAU_HAS_OCTAVE
  clearflag FLAG_KALOS_CHATEAU_HAS_ISAURE
  clearflag FLAG_KALOS_CHATEAU_HAS_CORENTIN
  clearflag FLAG_KALOS_CHATEAU_HAS_VIOLA
  clearflag FLAG_KALOS_CHATEAU_HAS_GRANT
  clearflag FLAG_KALOS_CHATEAU_HAS_KORRINA
  clearflag FLAG_KALOS_CHATEAU_HAS_RAMOS
  clearflag FLAG_KALOS_CHATEAU_HAS_CLEMONT
  clearflag FLAG_KALOS_CHATEAU_HAS_VALERIE
  clearflag FLAG_KALOS_CHATEAU_HAS_OLYMPIA
  clearflag FLAG_KALOS_CHATEAU_HAS_WULFRIC
  clearflag FLAG_KALOS_CHATEAU_HAS_VISITOR
  # Plan 37 · FC4: a visitor of another finished region, dressed as who the day drew
  facility titles dress
  if result == 1 setflag FLAG_KALOS_CHATEAU_HAS_VISITOR
  if var VAR_KALOS_CHATEAU_GUEST_1 == 0 setflag FLAG_KALOS_CHATEAU_HAS_MIRABELLE
  if var VAR_KALOS_CHATEAU_GUEST_1 == 1 setflag FLAG_KALOS_CHATEAU_HAS_TRISTAN
  if var VAR_KALOS_CHATEAU_GUEST_1 == 2 setflag FLAG_KALOS_CHATEAU_HAS_CELESTE
  if var VAR_KALOS_CHATEAU_TITLE < 3 goto Leaders
  if var VAR_KALOS_CHATEAU_GUEST_2 == 0 setflag FLAG_KALOS_CHATEAU_HAS_OCTAVE
  if var VAR_KALOS_CHATEAU_GUEST_2 == 1 setflag FLAG_KALOS_CHATEAU_HAS_ISAURE
  if var VAR_KALOS_CHATEAU_GUEST_2 == 2 setflag FLAG_KALOS_CHATEAU_HAS_CORENTIN
label Leaders
  if not flag KalosHallOfFame return
  if weekday monday setflag FLAG_KALOS_CHATEAU_HAS_VIOLA
  if weekday tuesday setflag FLAG_KALOS_CHATEAU_HAS_GRANT
  if weekday wednesday setflag FLAG_KALOS_CHATEAU_HAS_KORRINA
  if weekday thursday setflag FLAG_KALOS_CHATEAU_HAS_RAMOS
  if weekday friday setflag FLAG_KALOS_CHATEAU_HAS_CLEMONT
  if weekday saturday setflag FLAG_KALOS_CHATEAU_HAS_VALERIE
  if weekday sunday setflag FLAG_KALOS_CHATEAU_HAS_OLYMPIA
  if weekday sunday setflag FLAG_KALOS_CHATEAU_HAS_WULFRIC
  return

label Mirabelle
  speaker "Mirabelle"
  if flag FLAG_DAILY_KALOS_CHATEAU_BEAT_MIRABELLE goto MirabelleBeaten
  say "The tea is cold, the company is warm, and I have been waiting all afternoon for a battle. Will you oblige?"
  ask "Will you battle Mirabelle?"
  if no goto MirabelleRefused
  facility titles member
  if result == 0 goto Lost
  setflag FLAG_DAILY_KALOS_CHATEAU_BEAT_MIRABELLE
  speaker "Mirabelle"
  say "How lovely. I shall tell the others you play beautifully."
  goto Won
label MirabelleBeaten

# The house's titles, at thresholds of ours: the butler says when a new one is earned
script Title
  if var VAR_KALOS_CHATEAU_POINTS >= 90 goto Reach7
  if var VAR_KALOS_CHATEAU_POINTS >= 60 goto Reach6
  if var VAR_KALOS_CHATEAU_POINTS >= 40 goto Reach5
  if var VAR_KALOS_CHATEAU_POINTS >= 25 goto Reach4
  if var VAR_KALOS_CHATEAU_POINTS >= 15 goto Reach3
  if var VAR_KALOS_CHATEAU_POINTS >= 8 goto Reach2
  if var VAR_KALOS_CHATEAU_POINTS >= 3 goto Reach1
  return
label Reach1
  if var VAR_KALOS_CHATEAU_TITLE >= 1 return
  setvar VAR_KALOS_CHATEAU_TITLE 1
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Guest."
  fanfare item
  return
label Reach2
  if var VAR_KALOS_CHATEAU_TITLE >= 2 return
  setvar VAR_KALOS_CHATEAU_TITLE 2
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Regular."
  fanfare item
  return
label Reach3
  if var VAR_KALOS_CHATEAU_TITLE >= 3 return
  setvar VAR_KALOS_CHATEAU_TITLE 3
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Friend of the House."
  fanfare item
  return
label Reach4
  if var VAR_KALOS_CHATEAU_TITLE >= 4 return
  setvar VAR_KALOS_CHATEAU_TITLE 4
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Honoured Guest."
  fanfare item
  return
label Reach5
  if var VAR_KALOS_CHATEAU_TITLE >= 5 return
  setvar VAR_KALOS_CHATEAU_TITLE 5
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Patron."
  fanfare item
  return
label Reach6
  if var VAR_KALOS_CHATEAU_TITLE >= 6 return
  setvar VAR_KALOS_CHATEAU_TITLE 6
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Steward."
  fanfare item
  return
label Reach7
  if var VAR_KALOS_CHATEAU_TITLE >= 7 return
  setvar VAR_KALOS_CHATEAU_TITLE 7
  speaker "Butler"
  say "The house is pleased with you, {player}. From today you are its Master of the House."
  fanfare item
  return


# The butler by the clock: the house's rules and the player's title
script Butler
  face self player
  speaker "Butler"
  say "Welcome to the Château. Its guests battle whoever they please, once a day, and the house keeps count."
  if var VAR_KALOS_CHATEAU_TITLE == 7 goto Title7
  if var VAR_KALOS_CHATEAU_TITLE == 6 goto Title6
  if var VAR_KALOS_CHATEAU_TITLE == 5 goto Title5
  if var VAR_KALOS_CHATEAU_TITLE == 4 goto Title4
  if var VAR_KALOS_CHATEAU_TITLE == 3 goto Title3
  if var VAR_KALOS_CHATEAU_TITLE == 2 goto Title2
  if var VAR_KALOS_CHATEAU_TITLE == 1 goto Title1
  say "To the house you are a Visitor, for now. Win a few battles in this hall, and that will change."
  end
label Title1
  say "To the house you are its Guest. It wears well on you."
  end
label Title2
  say "To the house you are its Regular. It wears well on you."
  end
label Title3
  say "To the house you are its Friend of the House. It wears well on you."
  end
label Title4
  say "To the house you are its Honoured Guest. It wears well on you."
  end
label Title5
  say "To the house you are its Patron. It wears well on you."
  end
label Title6
  say "To the house you are its Steward. It wears well on you."
  end
label Title7
  say "To the house you are its Master of the House. It wears well on you."
  end
