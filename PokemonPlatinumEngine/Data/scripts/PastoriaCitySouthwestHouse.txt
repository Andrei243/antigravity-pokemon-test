# The woman of the house gives a berry a day (FLAG_DAILY_RECEIVED_PASTORIA_CITY_SOUTHWEST_HOUSE_RANDOM_BERRY, cleared
# with the day): one of the seventeen that weaken a hit of one type, Occa to Chilan in the order of the original's
# item numbers, by its die of seventeen.

  if result == 12 goto B12
  if result == 13 goto B13
  if result == 14 goto B14
  if result == 15 goto B15
  if result == 16 goto B16
  give "Occa Berry"
  end
