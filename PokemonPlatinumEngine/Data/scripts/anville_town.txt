# Anville Town (plan 22 · U6, part c). docs/scripts.md describes the language; the lines are our own, on Black 2's
# beats (the decompilation's res/field/scripts/anville_town.s, read for the order alone).
#
# Nobody walks to Anville: the Gear Station's train brings the player (GearStation.txt), and the station agent (2) takes
# them back. On a Saturday or a Sunday the train fans and the collectors come (FLAG_UNOVA_HIDE_ANVILLE_WEEKEND, which
# the arrival clears at the weekend and sets the rest of the week, as Black 2's 0x02a8 and 0x02a9 are).


script OnEnter
  call UnovaCatchUp
  setflag FLAG_UNOVA_HIDE_ANVILLE_WEEKEND
  if weekday saturday goto Weekend
  if weekday sunday goto Weekend
  end
label Weekend
  clearflag FLAG_UNOVA_HIDE_ANVILLE_WEEKEND
  end


# The station agent by the train (2)
script TrainHome
  face self player
  speaker "Station Agent"
  say "The train back to Nimbasa City leaves shortly."
  ask "Will you board?"
  if no goto Stay
  say "Then please take a seat and wait for the doors to close."
  scene "The train back to Nimbasa City"
  fade out 0.8
  music stop
  sound "crossing_engine"
  text "The train pulls away from the turntable and runs west through the woods, back toward the lights of Nimbasa..."
  endscene
  warp "GearStation" 17 4 down
  fade in
  end
label Stay
  say "It's a quiet town. Stay as long as you like, and take a good look at the cars."
  end


# The station agent by the turntable (7)
script RestingCars
  face self player
  speaker "Station Agent"
  if weekday saturday goto Busy
  if weekday sunday goto Busy
  say "The cars here are taking a rest. Come at the weekend, when the whole town turns out to see them."
  end
label Busy
  say "When the yard's full of people, I swear the trains look happier!"
  end


# The collectors' swaps (plan 22 · U6, part d). Black 2's weekend crowd trades with whoever has something to swap; the two
# here are ours, once each (NpcTrades "cogs" and "shelly"). The collector (14, hidden through the week) wants a Timburr
# for his Klink, the girl who tried swapping once (9) a Tympole for her Dwebble.
script Collector
  face self player
  speaker "Collector"
  if flag FLAG_UNOVA_ANVILLE_COGS_TRADED goto Done
  say "I collect anything to do with trains, and my best find is a Klink. All gears and no engine, if you ask me."
  ask "A Timburr is what I'm after, though. It'd look splendid with a girder on its shoulder. Will you swap one for my Klink?"
  if no goto Maybe
  say "Which Pokémon will you swap?"
  choosepokemon
  if result == 255 goto Maybe
  trade cogs
  if result == 0 goto Wrong
  setflag FLAG_UNOVA_ANVILLE_COGS_TRADED
  say "A Timburr! Oh, look at that grip! Take good care of Cogs, now. It likes a bit of oil."
  end
label Wrong
  say "Hmm? That isn't a Timburr. I'm only after a Timburr."
  end
label Maybe
  say "No? Then come back when you've a Timburr in your party. I'll be here at the weekends."
  end
label Done
  say "Cogs is doing well? Good! I've polished my Timburr's girder three times this week."
  end


script Swap
  face self player
  speaker "Backer"
  if flag FLAG_UNOVA_ANVILLE_SHELLY_TRADED goto Done
  say "I tried a swap here once, and it was more fun than I expected. Now I'm after another!"
  ask "I've a Dwebble that needs a wetter home. Will you swap me a Tympole for it?"
  if no goto Maybe
  say "Which Pokémon will you swap?"
  choosepokemon
  if result == 255 goto Maybe
  trade shelly
  if result == 0 goto Wrong
  setflag FLAG_UNOVA_ANVILLE_SHELLY_TRADED
  say "A Tympole! It's sweet! Be gentle with Shelly. She carries her rock very proudly."
  end
label Wrong
  say "Oh, that's not a Tympole. I'm after a Tympole, nothing else."
  end
label Maybe
  say "Never mind. Find me a Tympole, and I'll be waiting."
  end
label Done
  say "I hope Shelly's keeping her rock polished. Tympole's lovely, though. It sings to the rain!"
  end


# The girl's lost Pansage (16, and 15 in the plaza, Black 2's 0x0362). She tells of it, which puts it in the plaza by the
# cars (FLAG_UNOVA_ANVILLE_PANSAGE_LOST, which shows it); the Pansage, spoken to, hops into the player's arms (_FOUND, which
# hides it); the girl, spoken to once more, takes it back with a gift (_THANKED, which shows her Pansage by her).
script LostPansage
  face self player
  speaker "Girl"
  if flag FLAG_UNOVA_ANVILLE_PANSAGE_THANKED goto Thanked
  if flag FLAG_UNOVA_ANVILLE_PANSAGE_FOUND goto Return
  if flag FLAG_UNOVA_ANVILLE_PANSAGE_LOST goto Waiting
  say "My Pansage hopped onto a train on the way home from the amusement park, and it's been gone ever since."
  say "It must be somewhere about the cars. Would you look for it? It's a shy little thing, and it hides when it hears me."
  setflag FLAG_UNOVA_ANVILLE_PANSAGE_LOST
  end
label Waiting
  say "Any sign of my Pansage? It likes to sit where the sun's warm, and it can't resist a train."
  end
label Return
  say "That's my Pansage! Oh, you found it! Come here, you silly thing, you had me so worried!"
  say "You must take something for your trouble. Please, it's a Miracle Seed, from my grandfather's garden."
  give "Miracle Seed"
  setflag FLAG_UNOVA_ANVILLE_PANSAGE_THANKED
  say "I'll never let go of it on a train again. Thank you!"
  end
label Thanked
  say "Pansage hasn't left my side since. It likes the trains, so we go for a ride every weekend, but I hold on tight."
  end


script PansageLost
  cry own
  say "Pansage: Sage! Sage!"
  say "The Pansage hops into your arms, looking pleased with itself, and sits quite still. You carry it back to the girl by the fence."
  setflag FLAG_UNOVA_ANVILLE_PANSAGE_FOUND
  end


script PansageHome
  cry own
  say "Pansage: Pan! Pan-sage!"
  end
