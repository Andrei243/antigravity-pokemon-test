# NimbasaCity_BasketballCourt: Nimbasa City's basketball field (plan 42 · PS5; rulings, "Sports days"). Black 2's place, our words.
# Its three athletes battle once a day each (FLAG_DAILY_UNOVA_*), with the team the story calls for: the first
# under five of Unova's Badges, the second from five, the third with all eight, the fourth after the Hall of Fame
# (SportsDays.TierOf; SportsDayTests hold these scripts to SportsDays).

script Athlete1
  face self player
  if flag FLAG_DAILY_UNOVA_BASKETBALL_1 goto Done1
  say "Nothing but net! And nothing but wins!"
  if flag UnovaHallOfFame goto Fourth1
  if badges >= 8 goto Third1
  if badges >= 5 goto Second1
  battle self as "unova_hoopster_bobby"
  goto Won1
label Second1
  battle self as "unova_hoopster_bobby_2"
  goto Won1
label Third1
  battle self as "unova_hoopster_bobby_3"
  goto Won1
label Fourth1
  battle self as "unova_hoopster_bobby_4"
label Won1
  setflag FLAG_DAILY_UNOVA_BASKETBALL_1
label Done1
  say "You blocked every shot! Good game."
  end

script Athlete2

  if badges >= 5 goto Second2

  goto Won2
label Second2
  battle self as "unova_hoopster_john_4"
  goto Won2
label Third2
  battle self as "unova_hoopster_john"
  goto Won2
label Fourth2
  battle self as "unova_hoopster_john_2"
label Won2
  setflag FLAG_DAILY_UNOVA_BASKETBALL_2
label Done2
  say "Time-out for me. You're good!"
  end
