# The cable car's station at Mt. Chimney's summit (plan 21 · H4, part b). docs/scripts.md describes the language; the
# lines are our own, on Emerald's beats (data/maps/MtChimney_CableCarStation/scripts.inc, data/scripts/cable_car.inc).
#
# The other end of Route 112's ride (route_112_cable_car_station.txt has how it goes): the player comes up with
# VAR_CABLE_CAR_STATION_STATE 1, and the attendant here sends them down with 2.


script OnEnter
  if var VAR_CABLE_CAR_STATION_STATE != 1 end
  place mt_chimney_cable_car_attendant 6 3 down
  fade in
  walk player down 3
  move mt_chimney_cable_car_attendant down 3
  waitmoves
  face player up
  setvar VAR_CABLE_CAR_STATION_STATE 0
  end


script Attendant
  speaker "Attendant"
  ask "Thank you for riding the Mt. Chimney Cable Car! Would you like to go back down to Route 112?"
  if no goto Later
  say "Of course! Step this way, please."
  move mt_chimney_cable_car_attendant up 3
  waitmoves
  face mt_chimney_cable_car_attendant down
  walk player up 3
  setvar VAR_CABLE_CAR_STATION_STATE 2
  fade out
  music "hoenn/cable_car"
  text "The car glides down the mountainside, the summit shrinking behind it..."
  wait 1.5
  warp "Route112_CableCarStation" 6 4 down
  end
label Later
  say "Take your time up here. Do watch out for the ash, though!"
  end
