# The peak of Navel Rock (plan 21 · H12, part f). docs/scripts.md describes the language; the lines are our own, on
# Emerald's beats (data/maps/NavelRock_Top/scripts.inc).
#
# Ho-Oh waits over the peak until it is beaten (FLAG_DEFEATED_HO_OH) or caught (FLAG_CAUGHT_HO_OH). The peak's arrival
# shows it and readies the step before the summit (the area's trigger 0, VAR_TEMP_1 0), as Emerald's OnTransition does;
# that step meets it at level 70. As Emerald's, it perches high over the rock (the area's own tile, 12, 4) and flies
# down; its meeting is plan 29's beat (style guide, "Legendary scenes", flight and fire): the bars, the camera up over
# the rock, Ho-Oh taking off (the sky over the rock is solid ground to a walk) and its shadow sweeping down with the
# camera after it to the summit, the path's last tile, where it lands in a burst
# and the heat's ripple, the cry with the picture shaken, and then the battle's own way in. Run from, it flies off and is
# back the next time the peak is come to.


script OnEnter
  setvar VAR_TEMP_1 1
  if flag FLAG_CAUGHT_HO_OH goto Gone
  if flag FLAG_DEFEATED_HO_OH goto Gone
  clearflag FLAG_HIDE_HO_OH
  setvar VAR_TEMP_1 0
  end
label Gone
  setflag FLAG_HIDE_HO_OH
  end


script HoOh
  if flag FLAG_CAUGHT_HO_OH end
  if flag FLAG_DEFEATED_HO_OH end
  setvar VAR_TEMP_1 1
  scene "Ho-Oh at the peak of Navel Rock"
  music stop
  face player up
  walk player back down 1
  letterbox on 0.4
  camera pan 12 5 1.2
  camera zoom 1.4 0.8
  wait 0.4
  hide navel_rock_ho_oh
  effect shadow 12 9 1.2 down
  camera path 12 5 12 9 1.2
  place navel_rock_ho_oh 12 9 down
  show navel_rock_ho_oh
  effect burst 12 9
  wait 0.4
  effect ripple 12 9
  cry "Ho-Oh"
  camera shake 0.6
  emote player exclaim
  speaker none
  text "Ho-Oh swept down over the peak, and the air blazed with rainbow light!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Ho-Oh" 70
  hide navel_rock_ho_oh
  setflag FLAG_HIDE_HO_OH
  if result == 3 goto Caught
  if result == 1 setflag FLAG_DEFEATED_HO_OH
  speaker none
  text "Ho-Oh spread its wings and flew off into the sky..."
  end
label Caught
  setflag FLAG_CAUGHT_HO_OH
  end


# Ho-Oh looked at over the rock before the step that brings it down
script Waiting
  speaker none
  text "A Pokémon of rainbow colours is perched high on the rock above the peak."
  end
