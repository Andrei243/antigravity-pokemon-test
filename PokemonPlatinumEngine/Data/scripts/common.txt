# A patch of soft soil (plan 06 · R14a), on the beats of the original's scripts_berry_tree_interaction.s: what grows
# in it, a drink from the Sprayduck while it grows (standing south of it, facing it), its berries picked once it is
# in fruit, and on bare soil mulch laid and a berry planted, both chosen from the bag. The soil is the script's own
# (berry ..., Overworld/BerryPatches.cs).
script BerryPatch
  speaker none
  sound "select"
  berry status
  if result == 1 goto Planted
  if result == 2 goto Sprouted
  if result == 3 goto Growing
  if result == 4 goto Blooming
  if result == 5 goto Fruit
  berry mulched
  if result == 1 goto Mulched
  if not facing up goto Bare
  berry mulches
  if result == 1 goto Choose
  berry berries
  if result == 0 goto Bare
  say "It's soft, crumbly soil, just right for planting."
  goto OfferPlant
label Bare
  say "It's soft, crumbly soil."
  end
label Mulched
  if not facing up goto MulchOnly
  berry berries
  if result == 0 goto MulchOnly
  say "The soil has been covered with {mulch}."
  goto OfferPlant
label MulchOnly
  say "The soil has been covered with {mulch}."
  end
label Choose
  say "It's soft, crumbly soil, just right for planting."
  berry berries
  if result == 0 goto OfferMulch
  choose "What would you like to do?" "Spread mulch" "Plant a Berry" "Leave it"
  if result == 0 goto Mulch
  if result == 1 goto Plant
  end
label OfferMulch
  ask "Spread some mulch on the soil?"
  if yes goto Mulch
  end
label Mulch
  chooseitem mulch
  if result == 0 end
  berry mulch
  berry berries
  if result == 0 goto MulchSpread
  say "{player} spread the {mulch} over the soil."
  goto OfferPlant
label MulchSpread
  say "{player} spread the {mulch} over the soil."
  end
label OfferPlant
  ask "Plant a Berry here?"
  if no end
label Plant
  chooseitem berries
  if result == 0 end
  berry plant
  say "{player} pressed one {berry} into the soft soil."
  end
label Planted
  if not facing up goto PlantedOnly
  if not item "Sprayduck" goto PlantedOnly
  say "One {berry} was planted here."
  goto OfferWater
label PlantedOnly
  say "One {berry} was planted here."
  end
label Sprouted
  if not facing up goto SproutedOnly
  if not item "Sprayduck" goto SproutedOnly
  say "A little {berry} plant has pushed up through the soil."
  goto OfferWater
label SproutedOnly
  say "A little {berry} plant has pushed up through the soil."
  end
label Growing
  if not facing up goto GrowingOnly
  if not item "Sprayduck" goto GrowingOnly
  say "The {berry} plant is growing taller."
  goto OfferWater
label GrowingOnly
  say "The {berry} plant is growing taller."
  end
label Blooming
  if not facing up goto BloomingOnly
  if not item "Sprayduck" goto BloomingOnly
  say "The {berry} plant has burst into flower!"
  goto OfferWater
label BloomingOnly
  say "The {berry} plant has burst into flower!"
  end
label OfferWater
  ask "Give it some water?"
  if no end
  berry water
  say "There you go. It looks pleased."
  end
label Fruit
  ask "There are {yield} × {berry} here. Pick them?"
  if no goto LeaveFruit
  berry pick
  fanfare item
  say "{player} picked {yield} × {berry}!" "{player} put them away in the Berries pocket."
  end
label LeaveFruit
  say "{player} left the {berry} on the plant."
  end

# The nurse of a Pokémon Center.
script Nurse
  say "Good day! You've come to a Pokémon Center."
  ask "Your Pokémon look as if they could do with a rest. Shall I see to them?"
  if no goto Goodbye
  say "Leave them with me. It won't take a moment."
  heal
  fanfare heal
  say "There we are! Every one of them is fit and well again."
  # The first time one of the team has Pokérus, she says what it is (CommonScript_NurseCheckPokerus)
  if flag FLAG_POKECENTER_IDENTIFIED_POKERUS goto Goodbye
  if not pokerus goto Goodbye
  setflag FLAG_POKECENTER_IDENTIFIED_POKERUS
  say "Oh, one more thing. Your Pokémon seem to be carrying something very small... Pokérus, we call it."
  say "Don't worry, it does them no harm. Pokémon who have it grow a little stronger from every battle, and it fades away after a few days."
  end
label Goodbye
  say "Take care out there, and come back whenever you need us!"
  end

# A trainer: the challenge and the battle. Spoken to again once beaten, they say what they have left to say.
# A battle lost ends the script, as the player wakes up elsewhere.
script Trainer
  if defeated self goto Beaten
  trainerline before
  battle self
  end
label Beaten
  # The Vs. Seeker found them wanting a rematch (plan 06 · R12): the team of the level the story has reached
  if rematch self goto Rematch
  trainerline after
  if result == 0 sayown
  end
label Rematch
  trainerline before
  battle self rematch
  end

# ---- Field moves (plan 02 · S2): what the obstacles and the water ask, after the original's
# scripts_field_moves.s, in our own words. The obstacle scripts run when the player faces one and presses the
# button; the Use scripts run when a field move is chosen from the party menu, which has checked the move and its
# badge already. An obstacle's own flag is local to its area: what is cleared is back once the player leaves. An
# obstacle asks in the game's own voice, with no name on the box.

# A thin tree in the way. Cut clears it, with the Forest Badge.
script CutTree
  speaker none
  if not knows "Cut" goto Cannot
  if not badge forest goto Cannot
  ask "This thin tree could be cut down. Use Cut?"
  if no end
  usemove "Cut"
  setflag own
  end
label Cannot
  text "A thin tree stands in the way. A sharp enough move might clear it."
  end

script UseCut
  usemove "Cut"
  setflag own
  end


# A rock full of cracks. Rock Smash breaks it, with the Coal Badge. It stands in the Ravaged Path's way north, so a
# player who can't break it yet is told what is missing: the move, the HM that teaches it, or the Badge.
script Rock
  speaker none
  if not knows "Rock Smash" goto Cannot
  if not badge coal goto NoBadge
  ask "The rock is riddled with cracks. Use Rock Smash?"
  if no end
  usemove "Rock Smash"
  setflag own
  end
label NoBadge
  text "The rock is riddled with cracks. Rock Smash could break it, but not outside battle until Oreburgh's Gym Badge is won."
  end
label Cannot
  if item "HM06" goto Teach
  if not badge coal goto Early
  text "The rock is riddled with cracks. A Pokémon that knows Rock Smash could break it apart."
  end
label Teach
  text "The rock is riddled with cracks. A Pokémon taught Rock Smash with HM06 could break it apart."
  end
label Early
  text "The rock is riddled with cracks. A hard enough blow might break it apart, once Oreburgh's Gym Badge is won."
  end

script UseRockSmash
  usemove "Rock Smash"
  setflag own
  end


# A heavy boulder. Strength, with the Mine Badge, lets every boulder here be pushed until the player leaves.

  speaker none
  if flag FLAG_STRENGTH_ACTIVE goto Ready

  if not badge mine goto Cannot
  ask "A heavy boulder. Use Strength to shift it?"
  if no end
  call StrengthOn
  end
label Ready
  text "With Strength in use, the boulders here can be pushed."
  end
label Cannot
  text "A heavy boulder. Only a Pokémon of great power could shift it."
  end

script UseStrength
  if flag FLAG_STRENGTH_ACTIVE goto Ready
  call StrengthOn
  end
label Ready
  text "With Strength in use, the boulders here can be pushed."
  end

script StrengthOn
  usemove "Strength"
  setflag FLAG_STRENGTH_ACTIVE
  text "{user} braces itself. The boulders here can be pushed now."
  return

# Honey in the bag the player may spread some on the bark; six hours later whatever it drew is waiting there, and
# is battled at once, which takes the honey with it.
script HoneyTree
  sound "select"
  honeytree status
  if result == 3 goto Battle
  if result == 2 goto Slathered
  if not item "Honey" goto BareNoHoney
  text "The bark of this tree smells faintly sweet."
  goto Slather
label BareNoHoney
  text "The bark of this tree smells faintly sweet." "Pokémon might come to it, with a little Honey to tempt them."
  end
label Slathered
  if not item "Honey" goto SlatheredNoHoney
  text "Honey glistens on the bark. Nothing has come for it yet."
  goto Slather
label SlatheredNoHoney
  text "Honey glistens on the bark. Nothing has come for it yet."
  end
label Slather
  ask "Spread some Honey on the bark?"
  if no end
  take "Honey"
  honeytree slather
  wait 0.33
  text "{player} spread Honey all over the bark."
  end
label Battle
  honeytree battle
  if item "Honey" goto Slather
  end


# Honey used from the bag (plan 06 · R13; UseHoneyFromMenu): one is used up, and its scent draws out a wild Pokémon
# where any live, as Sweet Scent does.
script UseHoney
  take "Honey"
  text "{player} used the Honey."
  sweetscent
  if result == 0 text "The sweet smell hangs in the air, but nothing comes out."
  end


# Poison in the field (plan 06 · R13; the original's common script 3, CommonScript_FieldEffectPoisonFainted): each
# Pokémon of the team left with one hit point by the poison comes through it, cured, and the game says so.
script PoisonSurvived
  setvar VAR_POISON_SURVIVOR 0
label Next
  survivepoison VAR_POISON_SURVIVOR
  if result == 1 text "{member:VAR_POISON_SURVIVOR} held on through the poison!" "It shook the poison off."
  addvar VAR_POISON_SURVIVOR 1
  if var VAR_POISON_SURVIVOR < PARTY_COUNT goto Next
  setvar VAR_POISON_SURVIVOR 0
  end


# The day's swarm (plan 06 · R13; scripts_sandgem_town_counterpart_house_1f.s): the assistant's sister in Sandgem
# Town, once the Hall of Fame is behind the player. The first time she speaks of it, swarms begin; from then on
# she tells where today's is. Her house is built with plan 01 · M11's rooms, and she runs this then.
script SwarmNews
  if flag FLAG_TALKED_TO_COUNTERPART_SISTER_WITH_NATIONAL_DEX goto Today
  setflag FLAG_TALKED_TO_COUNTERPART_SISTER_WITH_NATIONAL_DEX
  swarms on
  say "Did you hear? Lately, whole crowds of one kind of Pokémon keep turning up in one place or another!" "Every day it's somewhere new. Come and ask me, and I'll tell you where it is today, {player}!"
  end
label Today
  say "Today there's a whole crowd of {swarm} on {swarmplace}!" "If you want to meet one, now's your chance, {player}!"
  end
