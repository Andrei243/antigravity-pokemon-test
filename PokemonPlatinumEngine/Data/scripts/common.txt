# What every game starts with. It is run with no screen, for a new game and for a save from before the story
# was kept, so it may set flags and variables and do nothing else. A chapter that hides someone until their
# scene comes sets the flag that hides them here.
#
# Plan 02 · S4, the original's InitNewGame: the rival bursting into the bedroom and his mother calling by later,
# the rival coming out of his door, the professor, the assistant and the briefcase on Route 201, the people of
# Lake Verity's later scenes and the legendary in its cavern, the rival and the professor in Sandgem Town; and the
# guitarist who keeps the player in town until the rival has left home.
script NewGame
  setflag FLAG_HIDE_TWINLEAF_TOWN_PLAYER_HOUSE_2F_RIVAL
  setflag FLAG_HIDE_TWINLEAF_TOWN_PLAYER_HOUSE_1F_RIVAL_MOM
  setflag FLAG_HIDE_TWINLEAF_TOWN_RIVAL
  setflag FLAG_HIDE_ROUTE_201_COUNTERPART
  setflag FLAG_HIDE_ROUTE_201_PROF_ROWAN
  setflag FLAG_HIDE_ROUTE_201_BRIEFCASE
  setflag FLAG_HIDE_LAKE_VERITY_LOW_WATER_STARLY
  setflag FLAG_HIDE_LAKE_VERITY_LOW_WATER_PROF_ROWAN
  setflag FLAG_HIDE_LAKE_VERITY_LOW_WATER_COUNTERPART
  setflag FLAG_HIDE_LAKE_VERITY_LOW_WATER_RIVAL
  setflag FLAG_HIDE_VERITY_CAVERN_MESPRIT
  setflag FLAG_HIDE_SANDGEM_TOWN_RIVAL
  setflag FLAG_HIDE_SANDGEM_TOWN_PROF_ROWAN
  setflag FLAG_HIDE_SANDGEM_TOWN_LAB_PROF_OAK

  return

# Furniture with nothing of its own to say (plan 08 · P8): what a television, a bookshelf, a bin and a map on a wall
# say when looked at (FieldScripts.ForFurniture). A television shows one of the programmes on (Story/Broadcasts.cs),
# made of what is true of the game and drawn on the script's die.
script Television
  speaker "TV"
  # An interview given to Jubilife City's reporters is shown first, the first time a television is looked at after it (plan 08 · P9)
  if flag FLAG_JUBILIFE_TV_INTERVIEW_ON_AIR goto OnAir
  # Gabby and Ty's segment on the interview the player gave them is shown first as well (plan 21 · H6, part c)
  if flag FLAG_GABBY_AND_TY_SHOW_NEW goto GabbyAndTyOnAir
  broadcast
  end
label OnAir
  clearflag FLAG_JUBILIFE_TV_INTERVIEW_ON_AIR
  broadcast interview
  end
label GabbyAndTyOnAir
  clearflag FLAG_GABBY_AND_TY_SHOW_NEW
  broadcast gabbyandty
  end

script Bookshelf
  speaker none
  random 3
  if result == 1 goto Raising
  if result == 2 goto Atlas
  say "The shelves are packed with books about Pokémon and the places they live."
  end
label Raising
  say "A row of books on raising Pokémon. Someone has marked a great many pages."
  end
label Atlas
  say "Picture books, travel guides and an old atlas, all lined up neatly."
  end

  end

# A patch of soft soil (plan 06 · R14a), on the beats of the original's scripts_berry_tree_interaction.s: what grows
# in it, a drink from the Sprayduck while it grows (standing south of it, facing it), its berries picked once it is
# in fruit, and on bare soil mulch laid and a berry planted, both chosen from the bag. The soil is the script's own
# (berry ..., Overworld/BerryPatches.cs).
script BerryPatch
  speaker none
  sound "select"

# A honey tree faced from the south (plan 06 · R13; the original's common script 8, CommonScript_HoneyTree). With

  honeytree status
  if result == 3 goto Battle
  if result == 2 goto Slathered
  if not item "Honey" goto BareNoHoney
  text "The bark of this tree smells faintly sweet."
  goto Slather

  text "The bark of this tree smells faintly sweet." "Pokémon might come to it, with a little Honey to tempt them."
  end

  if not item "Honey" goto SlatheredNoHoney
  text "Honey glistens on the bark. Nothing has come for it yet."
  goto Slather
label SlatheredNoHoney
  text "Honey glistens on the bark. Nothing has come for it yet."
  end
label Slather
  ask "Spread some Honey on the bark?"
  if no end
  take "Honey"
  honeytree slather
  wait 0.33
  text "{player} spread Honey all over the bark."
  end
label Battle
  honeytree battle
  if item "Honey" goto Slather
  end

# The Day-Care Lady, at her counter inside (plan 06 · R15; DayCareCommon_Lady): she takes up to two Pokémon to raise,
# which gain a point of EXP for every step the player takes, and gives them back for a fee of $100 and $100 more for
# each level they grew. An Egg, or the last Pokémon able to fight, she won't take.
script DayCareLady
  speaker "Day-Care Lady"
  daycare state
  if result == 1 goto EggWaiting
  if result == 2 goto One
  if result == 3 goto Two
  ask "Hello, dear. I'm the Day-Care Lady. We can raise Pokémon for you. Shall we look after one of yours?"
  if yes goto Raise
  say "Oh, never mind, then. Do come again."
  end
label Raise
  daycare check
  if result == 1 goto OnlyOne
  if result == 2 goto JustOne
  say "Which of your Pokémon shall we raise?"
  choosepokemon
  daycare leave
  if result == 1 goto ComeAgain
  if result == 2 goto MerelyAnEgg
  if result == 3 goto NothingToBattle
  setflag FLAG_STORED_POKEMON_AT_DAY_CARE
  # The couple's fan comes in (PokemonDayCare.txt: the original shows him as the room is next come to; plan 02 · S14)
  clearflag FLAG_HIDE_DAY_CARE_GYM_GUIDE
  say "Lovely. We'll raise your {left} for a while. Come back for it later."
  daycare state
  if result == 3 end
  ask "We can look after two Pokémon at once. Shall we raise another?"
  if yes goto Raise
label ComeAgain
  say "Very good. Do come again."
  end
label MerelyAnEgg
  say "Now, now. That's only an Egg, dear!"
  end
label NothingToBattle
  say "If you leave that one with me, what would you battle with? Come back another time."
  end
label OnlyOne
  say "Oh? But you've only the one Pokémon with you. You can't go off without a Pokémon! Come back another time."
  end
label JustOne
  say "Now, now. If you leave that one with me, you'll be left with just the one." "You'd do better to catch a few more first, if you ask me."
  end
label EggWaiting
  say "Ah, there you are! My husband has been looking for you."
  end
label One
  say "Ah, it's you! Good to see you. Now, about your Pokémon..."
  call Grown0
  ask "We can look after two Pokémon at once. Shall we raise another?"
  if yes goto Raise
  ask "Would you like your Pokémon back?"
  if yes goto TakeBack
  goto ComeAgain
label Two
  say "Ah, it's you! Good to see you. Now, about your Pokémon..."
  call Grown0
  call Grown1
  ask "Would you like your Pokémon back?"
  if yes goto TakeBack
  goto ComeAgain
label TakeBack
  if party >= 6 goto TeamFull
  daycare state
  if result == 2 goto TakeFirst
  choose "Which Pokémon will you take back?" "{daycare:0}  Lv. {level:0}" "{daycare:1}  Lv. {level:1}" "Cancel"
  if result == 1 goto TakeSecond
  if result == 2 goto ComeAgain
label TakeFirst
  ask "If you'd like your {daycare:0} back, that will be ${fee:0}."
  if no goto ComeAgain
  daycare take 0
  goto Taken
label TakeSecond
  ask "If you'd like your {daycare:1} back, that will be ${fee:1}."
  if no goto ComeAgain
  daycare take 1
label Taken
  if result == 1 goto Short
  say "Splendid! Here's your Pokémon."
  text "{player} took {taken} back from the Day-Care Lady."
  daycare state
  if result != 2 goto ComeAgain
  ask "Would you like the other one back as well?"
  if yes goto TakeBack
  goto ComeAgain
label Short
  say "Oh dear. You haven't got enough money..."
  end
label TeamFull
  say "Your team is full, dear. Make some room, then come and see me."
  end

# Route 111, then 118, then 120, and battle the player as a double battle with a party of their own each time (six
# of them; after the sixth they keep it and go on round the routes). VAR_GABBY_AND_TY_BATTLE_NUM is the count of their

# Gabby, spoken to: the battle and the interview, or, with the pair already fought on this visit, a word that they are
# watching
script GabbyAndTyGabby
  face self player
  speaker "Gabby"
  if flag FLAG_MAP_LOCAL_GABBY_AND_TY_BEATEN goto Watching
  if var PARTY_COUNT < 2 goto NotEnough
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 0 goto First
  say "There you are! Ty, get the camera rolling. We've been hoping to run into this trainer again!"
  goto Fight
label First
  say "Wait, wait! Ty, the camera! A trainer with Pokémon at their side, and we're after trainers!" "We're the reporters Gabby and Ty. Give us a battle for the cameras, two against two?"
label Fight
  call GabbyAndTyFight
  speaker "Gabby"
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 0 goto DefeatFirst
  say "Oh, we lost again! Still, that's the footage the viewers like."

label Interview
  call GabbyAndTyAfterTheBattle
  end
label Watching
  say "We'll be keeping an eye out for you. Keep up the good work out there!"
  end
label NotEnough
  say "A double battle's what we're after, and you've only one Pokémon to fight with. Come back with two!"
  end

# The battle won: what it is to be remembered for, the count moved on, and the interview (Emerald's
# GabbyAndTyBeforeInterview, then FirstInterview after the first battle and RequestInterview after the rest)
script GabbyAndTyAfterTheBattle
  speaker "Gabby"
  setflag FLAG_MAP_LOCAL_GABBY_AND_TY_BEATEN
  clearflag FLAG_GABBY_AND_TY_ON_AIR
  setvar VAR_GABBY_AND_TY_NEW_TRIVIA 0
  if var FAINTED > VAR_GABBY_AND_TY_FAINTED_BEFORE setvar VAR_GABBY_AND_TY_NEW_TRIVIA 4
  if var TURNS == 0 setvar VAR_GABBY_AND_TY_NEW_TRIVIA 1
  setvar VAR_GABBY_AND_TY_ROUTE 111
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 1 setvar VAR_GABBY_AND_TY_ROUTE 118
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 4 setvar VAR_GABBY_AND_TY_ROUTE 118
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 7 setvar VAR_GABBY_AND_TY_ROUTE 118
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 2 setvar VAR_GABBY_AND_TY_ROUTE 120
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 5 setvar VAR_GABBY_AND_TY_ROUTE 120
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 8 setvar VAR_GABBY_AND_TY_ROUTE 120
  addvar VAR_GABBY_AND_TY_BATTLE_NUM 1
  if var VAR_GABBY_AND_TY_BATTLE_NUM >= 9 setvar VAR_GABBY_AND_TY_BATTLE_NUM 6
  if var VAR_GABBY_AND_TY_BATTLE_NUM == 1 goto FirstTime
  if flag FLAG_GABBY_AND_TY_QUOTE_PENDING goto Quote
  ask "We never got an interview out of you last time. How about it now?"
  if no goto Refused
  goto Describe
label FirstTime
  ask "By the way, who ARE you? A trainer who battles like that deserves an interview. Would you mind?"
  if no goto Refused
  goto Describe
label Quote
  clearflag FLAG_GABBY_AND_TY_QUOTE_PENDING
  say "Last time you told us, \"{phrase:GABBY_AND_TY}\". The viewers loved it!"
  if var VAR_GABBY_AND_TY_TRIVIA == 1 goto Quick
  if var VAR_GABBY_AND_TY_TRIVIA == 4 goto Lost
  ask "Care to be interviewed again?"
  if no goto Refused
  goto Describe
label Quick
  ask "And you flattened us so fast that Ty had barely started filming! How about another interview?"
  if no goto Refused
  goto Describe
label Lost
  ask "And you lost a Pokémon and still won, which made us look rather good. How about another interview?"
  if no goto Refused
  goto Describe
label Describe
  say "Wonderful! Then tell us in two words: how did that battle feel?"
  phrase GABBY_AND_TY
  if result == 0 goto Refused
  say "Perfect! That's just what the viewers want. We'll be seeing you!"
  setvar VAR_GABBY_AND_TY_TRIVIA VAR_GABBY_AND_TY_NEW_TRIVIA
  setflag FLAG_GABBY_AND_TY_QUOTE_PENDING
  setflag FLAG_GABBY_AND_TY_ON_AIR
  setflag FLAG_GABBY_AND_TY_SHOW_NEW
  return
label Refused
  say "Oh, no interview? We won't give up. We'll find you again!"
  return

# The Bug-Catching Contest (plan 15 · E5), on HeartGold's beats in our own words. Both gates into the National Park
# keep its counter, and each says which it is (VAR_BUG_CONTEST_GATE: 0 Route 35's, 1 Route 36's) before it calls
# BugContestDesk: on a Tuesday, Thursday or Saturday the officer offers a place, sets the team aside but for its first
# Pokémon able to fight, hands over twenty Sport Balls and sends the player into the park, once a day.
script BugContestDesk
  speaker "Officer"
  bugcontest on
  if result == 1 goto Under
  if flag FLAG_DAILY_BUG_CONTEST_ENTERED goto Done
  bugcontest open
  if result == 0 goto Closed
  say "Today's the Bug-Catching Contest! Whoever catches the finest Bug Pokémon in the park wins a prize."
  ask "Would you like to take part?"
  if no goto Declined
  bugcontest start
  if result == 0 goto NoPokemon
  setflag FLAG_DAILY_BUG_CONTEST_ENTERED
  say "Only the first Pokémon of your team that can battle may come along. We'll look after the rest for you."
  say "Here are twenty Sport Balls. You have twenty minutes to catch the best Bug Pokémon you can. Good luck!"
  if var VAR_BUG_CONTEST_GATE == 1 goto FromRoute36
  warp "NationalPark" 40 85 up
  end
label FromRoute36
  warp "NationalPark" 74 39 left
  end
label Under
  call BugContestReturn
  end
label Declined
  say "No? Well, maybe next time. The contest is held every Tuesday, Thursday and Saturday."
  end
label NoPokemon
  say "Oh, none of your Pokémon can battle right now. Please heal them first."
  end
label Closed
  say "The Bug-Catching Contest is held in the park on Tuesdays, Thursdays and Saturdays."
  say "Bug Catchers come from all over Johto for it. Do take part!"
  end
label Done
  say "Today's contest is over. Thanks for taking part! The next one is on the next contest day."
  end


# Coming back into a gate from the park while a contest is under way (the gate's own arrival script): the officer asks
# whether the player is done. Yes goes to the judging; no sends the player back into the park by the gate they came in.
script BugContestReturn
  bugcontest on
  if result == 0 end
  speaker "Officer"
  ask "There's still time left in the contest. Are you finished catching?"
  if yes goto Judge
  say "Then back you go! Good luck!"
  if var VAR_BUG_CONTEST_GATE == 1 goto FromRoute36
  warp "NationalPark" 40 85 up
  end
label FromRoute36
  warp "NationalPark" 74 39 left
  end
label Judge
  call BugContestJudging
  end


# The contest's twenty minutes are up while the player is in the park (the game starts it): an announcement, and the
# player is brought to the gate they came in by for the judging.
script BugContestTimeUp
  speaker none
  text "Ding-dong! Time's up! The Bug-Catching Contest is over. All contestants, please make your way to the gate."
  call BugContestToTheGate
  end


# The last Sport Ball thrown (the game starts it, or BugContestSwap): the same, with its own announcement.
script BugContestOutOfBalls
  speaker none
  text "You've used your last Sport Ball. Your part in the contest is over. Please make your way to the gate."
  call BugContestToTheGate
  end


script BugContestToTheGate
  if var VAR_BUG_CONTEST_GATE == 1 goto FromRoute36
  warp "Route35_NationalParkGate" 23 4 left
  call BugContestJudging
  return
label FromRoute36
  warp "Route36_NationalParkGate" 5 6 up
  call BugContestJudging
  return


# A second Pokémon caught while one is kept (the game starts it after the battle): only one may be entered, so the
# player chooses which, and the other is set free. The last Sport Ball thrown ends the contest afterwards.
script BugContestSwap
  speaker none
  text "You're already holding on to {bugmon} for the contest."
  ask "Enter the {bugnew} you just caught instead?"
  if yes goto Switch
  text "{bugnew} was set free."
  bugcontest release
  goto Balls
label Switch
  text "{bugmon} was set free. {bugnew} is your entry now."
  bugcontest keep
label Balls
  bugcontest balls
  if result == 0 goto Out
  end
label Out
  call BugContestOutOfBalls
  end


# The judging, in the gate (BugContest_Judge): the player's catch is looked at, the six are judged and the board shows
# them third, second and first; the prize for the player's place, the team given back with the catch, and the others
# gone home.
script BugContestJudging
  speaker "Officer"
  bugcontest caught
  if result == 0 goto Nothing
  say "So you caught {bugmon}! Let's have the judges take a look."
  goto Judge
label Nothing
  say "You didn't catch anything? That's a shame. You're still in the running, though."
label Judge
  bugcontest judge
  setvar VAR_BUG_CONTEST_PLACE RESULT
  say "And now, the results of today's Bug-Catching Contest!"
  bugcontest results
  if var VAR_BUG_CONTEST_PLACE == 0 goto First
  if var VAR_BUG_CONTEST_PLACE == 1 goto Second
  if var VAR_BUG_CONTEST_PLACE == 2 goto Third
  say "You didn't make the top three this time, {player}. Here's a little something for taking part."
  goto Prize
label First
  say "First place goes to {player}, with a score of {score}! Your prize is a {prize}. Congratulations!"
  goto Prize
label Second
  say "Second place goes to {player}, with a score of {score}! Your prize is an {prize}. Well done!"
  goto Prize
label Third
  say "Third place goes to {player}, with a score of {score}! Your prize is a {prize}. Well done!"
label Prize
  bugcontest prize
  bugcontest end
  if result == 0 goto Bye
  if result == 2 goto Boxed
  text "{player}'s Pokémon came back, along with the one caught in the contest."
  goto Bye
label Boxed
  text "{player}'s Pokémon came back. The team was full, so the one caught in the contest was sent to the PC."
label Bye
  say "Thanks for taking part! We hold the contest every Tuesday, Thursday and Saturday."
  end
