# The Abundant Shrine (plan 22 · U10, part d, second piece). docs/scripts.md describes the language; the lines are our own.
# Black 2's Landorus waits before the shrine once the player has entered Unova's Hall of Fame (shown by UnovaHallOfFame), at
# level 70: caught or beaten it is gone for good (FLAG_UNOVA_FOUGHT_LANDORUS); run from, it flies off and is back the next
# time the wood is come to. Black 2 asks for Tornadus and Thundurus first (from the Dream Radar, which this game has no
# stand-in for), so ours asks only the Hall of Fame (the rulings' "The Abundant Shrine's Landorus").


script Landorus
  if not flag UnovaHallOfFame end
  scene "Landorus at the Abundant Shrine"
  music stop
  face player self
  face self player
  letterbox on 0.4
  camera path 22 14 22 15 3
  camera zoom 1.4 0.8
  effect beam 22 14 0.8
  shake self 8
  camera shake 0.8
  cry "Landorus"
  speaker none
  text "Landorus let out a thundering cry!\nThe soil of the whole wood seems to stir at its call!"
  camera release 0.4
  letterbox off 0.3
  endscene
  wildbattle "Landorus" 70
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_LANDORUS
  hide self
  text "The wood settles. Only the wind over the pond is left..."
  # Plan 40 · GE3: its two fellows answer (abundant_shrine.json's tornadus and thundurus, shown by the flag just set)
  camera shake 0.6
  text "Then thunder rolls overhead, and a gale tears through the trees!\nSomething else has come down to the shrine!"
  end
label Fled
  hide self
  text "Landorus soared away over the wood..."
  end


# Plan 40 · GE3, a way of ours for Black 2's Dream Radar, by which Black 2 gave the two: once Landorus has been met,
# Tornadus and Thundurus wait in the clearing before the shrine, each at level 70. Caught or beaten each is gone for good
# (FLAG_UNOVA_FOUGHT_TORNADUS, FLAG_UNOVA_FOUGHT_THUNDURUS); run from, it flies off and is back the next time the wood is
# come to.
script Tornadus
  if not flag FLAG_UNOVA_FOUGHT_LANDORUS end
  scene "Tornadus at the Abundant Shrine"
  music stop
  face player self
  letterbox on 0.4
  cry "Tornadus"
  speaker none
  text "Tornadus whips up a howling wind!\nThe leaves of the whole wood are torn into the air!"
  letterbox off 0.3
  endscene
  wildbattle "Tornadus" 70
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_TORNADUS
  hide self
  text "The wind drops. The leaves drift back down over the shrine..."
  end
label Fled
  hide self
  text "Tornadus rode the wind away over the wood..."
  end


script Thundurus
  if not flag FLAG_UNOVA_FOUGHT_LANDORUS end
  scene "Thundurus at the Abundant Shrine"
  music stop
  face player self
  letterbox on 0.4
  cry "Thundurus"
  speaker none
  text "Thundurus cracks the sky with lightning!\nThe air of the whole clearing hums!"
  letterbox off 0.3
  endscene
  wildbattle "Thundurus" 70
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_THUNDURUS
  hide self
  text "The thunder rolls away into the distance..."
  end
label Fled
  hide self
  text "Thundurus rode the storm away over the wood..."
  end
