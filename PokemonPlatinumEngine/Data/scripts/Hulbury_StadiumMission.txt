# lines, on Sword and Shield's beat of switches that turn the water; a switch, not the original's puzzle

  speaker "Gatekeeper"
  say "This floodgate holds the water back from the course. Want me to pull the switch? Here goes!"
  text "The water rushed off down a side channel, and the gateway drained dry!"
  setflag FLAG_GALAR_HULBURY_SWITCH_1
  end

  end


script MissionSwitch3

  end
