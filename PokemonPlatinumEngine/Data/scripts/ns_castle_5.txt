# N's Castle's throne room (plan 22 · U10, the last part). docs/scripts.md describes the language; the lines are our own,
# on Black 2's beats (N waiting at the top of his ruined castle with his dragon, after the Hall of Fame).
#
# N and Zekrom stand there once Unova's Hall of Fame is entered (shown by UnovaHallOfFame). Spoken to, N tells why he came
# back and steps aside; Zekrom is battled at Black 2's level 70. Caught or beaten it is gone for good (FLAG_UNOVA_FOUGHT_ZEKROM)
# and N leaves with a word (FLAG_UNOVA_HIDE_N_CASTLE); run from, it is back the next time the room is come to.


script N
  if not flag UnovaHallOfFame end
  face self player
  speaker "N"
  if flag FLAG_UNOVA_FOUGHT_ZEKROM goto Parting
  say "So you found your way up here. This castle was my whole world once. Now it's only stone and quiet."
  say "Zekrom came with me. It wants to know the Trainer who stood at the League's top."
  say "Go to it. Let it see what kind of truth and ideals you carry."
  end
label Parting
  say "Zekrom has seen you now. Whatever happens, it will remember."
  say "I'm going to keep travelling. There are so many Pokémon I still haven't heard. Farewell, {player}."
  setflag FLAG_UNOVA_HIDE_N_CASTLE
  fade out 0.5
  hide self
  fade in
  end


script Zekrom
  if not flag UnovaHallOfFame end
  scene "Zekrom in N's Castle"
  music stop
  face player self
  letterbox on 0.4
  effect burst 16 37
  cry "Zekrom"
  speaker none
  text "Zekrom's tail crackles with lightning!\nIt rears up to face {player}!"
  letterbox off 0.3
  endscene
  wildbattle "Zekrom" 70
  if result == 2 goto Fled
  setflag FLAG_UNOVA_FOUGHT_ZEKROM
  hide self
  text "The throne room is quiet once more..."
  end
label Fled
  text "Zekrom watches {player} from beside the throne..."
  end
