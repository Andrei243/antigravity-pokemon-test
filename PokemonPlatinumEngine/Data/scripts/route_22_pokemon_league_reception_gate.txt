# The Pokémon League's reception gate, where Routes 26, 22 and 28 meet (plan 20 · J10, part b). docs/scripts.md
# describes the language; the lines are our own, on HeartGold's beats.
#
# The guard in the middle of the hall lets a trainer north to Victory Road only with all eight of Johto's Badges: the
# row before him is the area's first trigger, which waits on VAR_UNK_4110 as HeartGold counts it (1 once he has seen
# them), and a trainer without them is turned back a step. The guard at the door to Route 22 keeps Kanto shut to
# anyone who hasn't stood before the League's Champion: his line of tiles is the third trigger, which turns everyone
# back but a Champion of Johto (plan 20 · J12). The guard at Route 28's door, toward Mt. Silver, lets through only a
# Champion of Johto who holds Kanto's eight Badges as well, sixteen in all, as HeartGold's does (plan 20 · J15): his
# line of tiles is the second trigger, which waits on VAR_UNK_4110 1 and is put to rest at 2 once he has let them by.


script BadgeCheck
  face r22r0101_policeman player
  speaker "Guard"
  if badges < 8 goto Short
  say "Hold on a moment. The road north is for trainers with all eight of Johto's Badges... and you have every one!"
  say "Victory Road waits beyond the north door. Good luck up there."
  setvar VAR_UNK_4110 1
  end
label Short
  say "Hold on a moment. The road north is for trainers with all eight of Johto's Badges, and you don't have them yet."
  say "Come back once you've won the rest."
  walk player down 1
  end


script Guard
  speaker "Guard"
  if var VAR_UNK_4110 >= 1 goto Passed
  say "Only a trainer with all eight of Johto's Badges may take the road north to Victory Road."
  end
label Passed
  say "You've shown me all eight Badges. The way north to Victory Road is open to you."
  end


# Since plan 20 · J12 a Champion of Johto (JohtoHallOfFame) is let through, and the trigger is put to rest
# (VAR_UNK_4121 1, as HeartGold's gate does once the player may pass)
script KantoShut
  face r22r0101_policeman_3 player
  speaker "Guard"
  if flag JohtoHallOfFame goto Champion
  say "Sorry, this door to Route 22 is closed. Kanto is open only to trainers who have stood before the League's Champion."
  walk player left 1
  end
label Champion
  say "You're the new Champion! Of course you may pass. Route 22 and Kanto are through this door."
  setvar VAR_UNK_4121 1
  end


# The guard at the door to Route 22
script KantoGuard
  speaker "Guard"
  if flag JohtoHallOfFame goto Champion
  say "Through this door is Route 22, and Kanto beyond it. It opens to those who have stood before the Champion."
  end
label Champion
  say "Through this door is Route 22, and Kanto beyond it. Safe travels, Champion."
  end


# Plan 20 · J15: the line before Route 28's door. Mt. Silver is for a Champion of Johto with Kanto's eight Badges too
# (HeartGold asks for all sixteen and Professor Oak's word; here Kanto's are the region's own, won in FireRed's Kanto),
# and once he has let the player by the trigger is put to rest (VAR_UNK_4110 2)
script MtSilverCheck
  face r22r0101_policeman_2 player
  speaker "Guard"
  if not flag JohtoHallOfFame goto Shut
  if badges of kanto < 8 goto Shut
  say "Sixteen Badges, Johto's and Kanto's... You're the trainer the League told me about. Mt. Silver is past this door."
  say "The mountain is wild and its Pokémon are fierce. Go carefully."
  setvar VAR_UNK_4110 2
  end
label Shut
  say "Hold on! Past this door is Route 28, and Mt. Silver. Only a Champion with all sixteen of Johto's and Kanto's Badges may go there."
  walk player right 1
  end


# The guard at Route 28's door
script MtSilverGuard
  speaker "Guard"
  if var VAR_UNK_4110 == 2 goto Passed
  say "The road west leads to Mt. Silver. Only a Champion with all sixteen of Johto's and Kanto's Badges may go that way."
  end
label Passed
  say "Mt. Silver is past this door. Not many trainers come back down it the same as they went up."
  end
