# The Cerulean Gym as HeartGold has it (plan 20 · J15, part e). docs/scripts.md describes the language; the lines are
# our own, on HeartGold's beats (scr_seq_T04GYM0101).
#
# Misty is out when a Champion of Johto first comes: she is on a date on Route 25's cape (johto_route_25.txt), and the
# Gym guide says so. Once she is home (FLAG_JOHTO_MISTY_BACK_IN_GYM, a flag of ours) she waits at the head of her pool
# and battles for the Cascade Badge, which is Kanto's (`givebadge cascade of kanto`, as the other Kanto Gyms of
# HeartGold's: the rulings' "HeartGold's Kanto, part b" to "part e"), counts her five trainers beaten and gives TM55,
# Brine. Whether she is beaten here is FLAG_JOHTO_DEFEATED_MISTY, a flag of ours, so a FireRed Champion battles her
# all the same.


script Misty
  face t04gym0101_gsleader11 player
  speaker "Misty"
  if flag FLAG_JOHTO_DEFEATED_MISTY goto Won
  scene "Misty"
  say "So you came after all. You did spoil my afternoon on the cape, you know."
  say "Still, a Champion from Johto doesn't turn up every day. My Water Pokémon have been itching for a real battle."
  say "Let's make a splash! Show me what you've got!"
  battle self as "johto_leader_misty_misty"
  say "Wow... you really are strong. All right, you've earned it. Here's the Cascade Badge."
  givebadge cascade of kanto
  text "{player} received the Cascade Badge from Misty!"
  setflag FLAG_JOHTO_DEFEATED_MISTY
  defeat "johto_swimmer_f_diana"
  defeat "johto_swimmer_f_joy"
  defeat "johto_swimmer_f_briana"
  defeat "johto_sailor_eddie"
  defeat "johto_sailor_parker"
  endscene
label Tm
  say "Take this too. It's a move my Pokémon love to use."
  give "TM55"
  setflag FLAG_JOHTO_GOT_TM55_FROM_MISTY
  say "TM55 is Brine. It hits twice as hard when the foe is down to half its HP or less."
  end
label Won
  if not flag FLAG_JOHTO_GOT_TM55_FROM_MISTY goto Tm
  say "Next time, I'll beat you. Then you can owe me an afternoon on the cape!"
  end


# The Gym guide by the door.
script Guide
  if flag FLAG_JOHTO_DEFEATED_MISTY goto Won
  if flag FLAG_JOHTO_MISTY_BACK_IN_GYM goto Home
  say "Hey, champ in the making! Bad news, I'm afraid. Misty isn't here."
  say "She slipped out to the cape north of town again. Try Route 25, past Nugget Bridge."
  end
label Home
  say "Misty's back! Her Pokémon are Water types, so Electric and Grass moves will serve you well."
  end
label Won
  say "You beat Misty! What a battle. The whole pool was splashing!"
  end


# The statues by the door.
script Statue
  if flag FLAG_JOHTO_DEFEATED_MISTY goto Won
  text "Cerulean City Pokémon Gym\nLeader: Misty"
  end
label Won
  text "Cerulean City Pokémon Gym\nLeader: Misty\nWinning trainers: {player}"
  end
