# The Celadon Department Store's lift in HeartGold's Kanto (plan 20 · J15, part k). docs/scripts.md describes the
# language; the lines are our own, on HeartGold's beats (scr_seq_T07R0107).
#
# As the Goldenrod store's lift (goldenrod_department_store_elevator.txt), the car's door leads to the ground floor and
# the attendant rides the car and lets the player out before the lift's door of the floor chosen, up or down from the
# floor the player came in from (VAR_JOHTO_CELADON_STORE_FLOOR, which each floor sets as it is come to: 1 to 5, and 6
# for the roof; nought, never set, is the ground floor).


script Attendant
  face t07r0107_gswoman6 player
  speaker "Attendant"
  if var VAR_JOHTO_CELADON_STORE_FLOOR == 0 setvar VAR_JOHTO_CELADON_STORE_FLOOR 1
  choose "Welcome. Which floor would you like?" "ROOF" "5F" "4F" "3F" "LOWER FLOORS" "Stay here"
  if result == 5 end
  setvar VAR_TEMP_1 6
  if result == 1 setvar VAR_TEMP_1 5
  if result == 2 setvar VAR_TEMP_1 4
  if result == 3 setvar VAR_TEMP_1 3
  if result != 4 goto Ride
  choose "The second floor or the ground floor?" "2F" "1F" "Stay here"
  if result == 2 end
  setvar VAR_TEMP_1 2
  if result == 1 setvar VAR_TEMP_1 1
label Ride
  if var VAR_TEMP_1 == VAR_JOHTO_CELADON_STORE_FLOOR goto Out
  if var VAR_TEMP_1 < VAR_JOHTO_CELADON_STORE_FLOOR goto Down
  say "Going up!"
  face t07r0107_gswoman6 down
  anim elevator up
  goto Out
label Down
  say "Going down!"
  face t07r0107_gswoman6 down
  anim elevator down
label Out
  face t07r0107_gswoman6 player
  say "Here we are. Please come again!"
  if var VAR_TEMP_1 == 6 warp "Celadon_DeptStore_Roof" 5 3 down
  if var VAR_TEMP_1 == 5 warp "Celadon_DeptStore_5F" 5 3 down
  if var VAR_TEMP_1 == 4 warp "Celadon_DeptStore_4F" 5 3 down
  if var VAR_TEMP_1 == 3 warp "Celadon_DeptStore_3F" 5 3 down
  if var VAR_TEMP_1 == 2 warp "Celadon_DeptStore_2F" 5 3 down
  if var VAR_TEMP_1 == 1 warp "Celadon_DeptStore_1F" 5 3 down
  end
