# The Black Tower's lobby in Black City (plan 22 · U10, the last part). docs/scripts.md describes the language; the lines
# are our own, on the beats of Black 2's Black Tower (a climb through the tower's areas, a trainer after another, Benga
# at the top).
#
# A climb (Models/Frontier/BlackTower.cs) lends the Frontier's rules as the Battle Subway does: three Pokémon of
# different species holding different items go in as copies at Level 50 or under, whole again before every battle, and
# nothing from the Bag. The tower's five areas are three battles each, told over the dark of the climb as the Subway's
# cars are told; the top's last battle is Benga's once Unova's Hall of Fame is entered. A climb is the story's own
# variables, so a game saved on the way up goes on at its battle when the desk is spoken to again.


script Desk
  face self player
  speaker "Receptionist"
  blacktower active
  if result == 1 goto GoOn
  say "Welcome to the Black Tower! Trainers climb through its five areas, battling whoever waits on the way, all the way to the top."
  choose "What would you like to do?" "CLIMB" "INFO" "RECORD" "EXIT"
  if result == 1 goto Info
  if result == 2 goto Record
  if result == 3 goto Bye
  blacktower enough
  if result == 0 goto NotEnough
  say "Please choose the three Pokémon who will climb with you."
  blacktower begin
  blacktower choose
  if result == 0 goto Changed
  blacktower check
  if result == 1 goto SameKind
  if result == 2 goto SameItem
  say "Very good! The way up is through the door behind me. Good luck!"
  goto Climb
label GoOn
  say "Welcome back! Your climb is waiting for you. Up you go!"
  goto Climb
label Info
  say "The tower has five areas, and three Trainers wait in each. Win them all and you reach the top."
  say "Three Pokémon go up, all of different species and holding different items. They battle at Level 50 or under, and nothing from the Bag may be used."
  say "Every area cleared earns a Battle Point, and the top earns ten more."
  end
label Record
  blacktower best
  say "The most areas you have cleared in one climb: {number}."
  blacktower cleared
  say "Times you have reached the top: {number}."
  end
label NotEnough
  say "You need three Pokémon of different species, holding different items, to climb. {banned} may not go up."
  end
label Changed
  blacktower finish
  say "Come back when you're ready to climb."
  end
label SameKind
  blacktower finish
  say "Two Pokémon of the same species can't climb together. Please choose again."
  end
label SameItem
  blacktower finish
  say "Two Pokémon holding the same item can't climb together. Please choose again."
  end
label Bye
  say "We hope to see you climb soon!"
  end

# The climb itself: each battle told over the dark, the area announced as it is come to
label Climb
  call Upward
  end


script Upward
  scene "A climb of the Black Tower"
  fade out 0.6
  music stop
label Next
  blacktower area
  text "Area {number} of the Black Tower."
  blacktower top
  if result == 1 goto Benga
  blacktower opponent
  text "{foe} steps out of the dark to battle!"
  goto Fight
label Benga
  blacktower opponent
  speaker "Benga"
  say "Hey, you made it all the way up! I'm Benga. My grandpa's a pretty good Trainer, and so am I!"
  say "Let's see what you've got!"
  speaker "Receptionist"
label Fight
  blacktower battle
  if result == 0 goto Lost
  blacktower won
  if result == 2 goto Top
  if result == 1 goto AreaDone
  text "The way goes on."
  goto Next
label AreaDone
  text "Area {number} is cleared!"
  ask "Will you keep climbing?"
  if no goto Down
  goto Next
label Top
  text "The top of the Black Tower! Nobody else waits here."
  goto Down
label Lost
  text "The climb ends here."
label Down
  blacktower finish
  endscene
  fade in
  speaker "Receptionist"
  if result == 0 goto NoPoints
  say "Welcome back! For your climb you've earned {number} Battle Points."
  return
label NoPoints
  say "Welcome back! Please climb again soon."
  return
