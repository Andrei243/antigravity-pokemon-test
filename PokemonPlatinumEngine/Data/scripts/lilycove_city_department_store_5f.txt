# Lilycove's Department Store, fifth floor (plan 21 · H10): the Pokédoll Floor, whose four counters sell what dresses
# up a secret base (LilycoveCity_DepartmentStore_5F_EventScript_ClerkFarLeft to _ClerkFarRight: Emerald's
# pokemartdecoration2 lists, at Emerald's prices, decorations.json). What is bought goes into the decoration PC
# (`openbase buy`: RESULT 1 bought, 0 short of money, 2 no room for its kind). A question takes six answers, so a long
# list is asked a page at a time; a purchase ends the talk.


# The far left counter: dolls
script Dolls
  speaker "Clerk"
  say "Welcome to the Pokédoll Floor! These dolls are made to sit about a secret base."
label Page1
  choose "Which would you like?" "PICHU DOLL  3,000" "PIKACHU DOLL  3,000" "MARILL DOLL  3,000" "JIGGLYPUFF DOLL  3,000" "MORE" "NO, THANKS"
  if result == 0 goto Pichu
  if result == 1 goto Pikachu
  if result == 2 goto Marill
  if result == 3 goto Jigglypuff
  if result == 4 goto Page2
  goto Bye
label Page2
  choose "Which would you like?" "DUSKULL DOLL  3,000" "WYNAUT DOLL  3,000" "BALTOY DOLL  3,000" "KECLEON DOLL  3,000" "MORE" "NO, THANKS"
  if result == 0 goto Duskull
  if result == 1 goto Wynaut
  if result == 2 goto Baltoy
  if result == 3 goto Kecleon
  if result == 4 goto Page3
  goto Bye
label Page3
  choose "Which would you like?" "AZURILL DOLL  3,000" "SKITTY DOLL  3,000" "SWABLU DOLL  3,000" "GULPIN DOLL  3,000" "BACK TO THE START" "NO, THANKS"
  if result == 0 goto Azurill
  if result == 1 goto Skitty
  if result == 2 goto Swablu
  if result == 3 goto Gulpin
  if result == 4 goto Page1
  goto Bye
label Pichu
  openbase buy "Pichu Doll"
  goto Sold
label Pikachu
  openbase buy "Pikachu Doll"
  goto Sold
label Marill
  openbase buy "Marill Doll"
  goto Sold
label Jigglypuff
  openbase buy "Jigglypuff Doll"
  goto Sold
label Duskull
  openbase buy "Duskull Doll"
  goto Sold
label Wynaut
  openbase buy "Wynaut Doll"
  goto Sold
label Baltoy
  openbase buy "Baltoy Doll"
  goto Sold
label Kecleon
  openbase buy "Kecleon Doll"
  goto Sold
label Azurill
  openbase buy "Azurill Doll"
  goto Sold
label Skitty
  openbase buy "Skitty Doll"
  goto Sold

  goto Sold

  say "Please do come again!"

# What a counter says once `openbase buy` has been answered
script Told
  if result == 0 goto Poor
  if result == 2 goto Full
  say "Thank you very much! The {thing} has been sent to your decoration PC."
  return
label Poor
  say "I'm terribly sorry, but you haven't enough money for that."
  return
label Full
  say "I'm sorry, but your decoration PC has no more room for things of that kind."
  return
