# Route 45 (Johto's leftovers: the phone's gifts). docs/scripts.md describes the language; the lines are our own, on
# HeartGold's beats.
#
# Black Belt Kenji and Hiker Parry, beaten, give their numbers; once either has rung about a present (phone.txt,
# FLAG_JOHTO_PHONE_GIFT_KENJI and _PARRY, ours) he hands it over here: Kenji a PP Up, Parry an Iron.


script Kenji
  if not defeated self goto Fight
  if rematch self goto Rematch
  speaker "Kenji"
  if flag FLAG_JOHTO_PHONE_GIFT_KENJI goto Gift
  if flag FLAG_PHONE_CONTACT_KENJI goto Known
  say "Your spirit is strong. Let us exchange numbers, so that we may train against each other again."
  ask "Register Kenji's number in your Pokégear?"
  if no goto Refused
  addcontact kenji
  say "Good. I will call when my training has borne fruit."
  end
label Refused
  say "I see. A lone path is also a path."
  end
label Known
  trainerline after
  end
label Gift
  say "You have come. Take this. Training is not only for the body: your Pokémon's moves can grow too."
  give "PP Up"
  clearflag FLAG_JOHTO_PHONE_GIFT_KENJI
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end


script Parry
  if not defeated self goto Fight
  if rematch self goto Rematch
  speaker "Parry"
  if flag FLAG_JOHTO_PHONE_GIFT_PARRY goto Gift
  if flag FLAG_PHONE_CONTACT_PARRY goto Known
  say "Ha! You're as tough as the mountain. Let's swap numbers, and I'll tell you what I find up here."
  ask "Register Parry's number in your Pokégear?"
  if no goto Refused
  addcontact parry
  say "Great! I'll ring you when I've dug up something good."
  end
label Refused
  say "No? Well, the mountain and I will be here."
  end
label Known
  trainerline after
  end
label Gift
  say "There you are! Here's what I found on the mountain. It'll make your Pokémon as hard as rock!"
  give "Iron"
  clearflag FLAG_JOHTO_PHONE_GIFT_PARRY
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end
