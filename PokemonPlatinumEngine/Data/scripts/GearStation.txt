# Nimbasa City's Gear Station (plan 22 · U6, part c). docs/scripts.md describes the language; the lines are our own, on
# Black 2's beats (the decompilation's res/field/scripts/gear_station.s and anville_town.s, read for the order alone).
#
# Black 2's hall stands on rails, which the reader leaves out, so it is a map of ours (Data/maps/GearStation.json): the
# desks of the Battle Subway's three lines and of the Anville Town Line along its north wall. The Anville Town Line's
# agent takes the player out to Anville Town, a scene of the fade and lines over the dark as the voyage to Castelia is,
# and Anville's station agent brings them back (anville_town.txt). Since plan 22 · U6, part d the Subway's three desks
# run the Battle Subway (Models/Frontier/BattleSubway.cs, the cars in BattleSubwayTrain.txt): the Single Train takes
# three Pokémon, the Double Train four and the Multi Train two beside a partner of the Subway's own, all copies at
# Level 50, and a run ends in the cars with the player brought back here.


script AnvilleLine
  face self player
  speaker "Station Agent"
  say "This is the Anville Town Line. Our train runs out to Anville Town, where the line ends, and back again all day."
  ask "Will you ride the train to Anville Town?"
  if no goto Later
  say "Then please board and take a seat. We'll be leaving shortly!"
  call ToAnville
  end
label Later
  say "Anville Town is a quiet little place. Do pay it a visit when you have the time."
  end


script ToAnville
  scene "The train to Anville Town"
  fade out 0.8
  music stop
  sound "crossing_engine"
  text "The train glides out of the Gear Station and rattles east past fields and woods..."
  text "With a long sigh of its brakes it rolls into Anville Town, the end of the line."
  endscene
  warp "AnvilleTown" 20 52 right
  fade in
  return


# The Single Train: one on one
script SingleTrain
  face self player
  speaker "Depot Agent"
  subway end
  say "Welcome to the Battle Subway! On the Single Train you battle one on one, car after car, seven to a set."
  choose "What would you like to do?" "RIDE" "INFO" "RECORD" "EXIT"
  if result == 1 goto Info
  if result == 2 goto Record
  if result == 3 goto Bye
  subway opened single
  if result == 0 goto Normal
  choose "Which train will you ride?" "SINGLE TRAIN" "SUPER SINGLE TRAIN"
  if result == 1 goto Super
label Normal
  subway begin single
  goto Ride
label Super
  say "The Super Single Train! Its trainers are the toughest on the line, and Ingo waits in the 49th car."
  subway begin supersingle
  goto Ride
label Info
  say "A trainer waits in every car. Clear seven and you may get off with Battle Points, or go on to the next set."
  say "The Single Train takes three Pokémon, all of different species and holding different items. They battle at Level 50 or under, and nothing from the Bag may be used."
  say "The Subway Bosses, Ingo and Emmet, wait for those who ride on to the 21st car and the 49th."
  say "Beat Ingo in the 21st car and the Super Single Train opens to you, with tougher trainers and more Battle Points."
  end
label Record
  subway begin single
  subway best
  subway end
  say "Your longest streak on the Single Train is {number} cars."
  subway opened single
  if result == 0 end
  subway begin supersingle
  subway best
  subway end
  say "And on the Super Single Train, {number} cars."
  end
label Ride
  call Board
  end
label Bye
  say "Please ride with us again!"
  end

# The Pokémon for the run are chosen and the player goes down to the train. A run begun is let go of if backed out of.
script Board
  subway enough
  if result == 1 goto Choose
  say "I'm sorry. You need {number} Pokémon that can take part, all of different species and holding different items."
  say "An Egg can't take part, and nor can {banned}."
  subway end
  return
label Choose
  say "Please choose the Pokémon you will take aboard."
  subway choose
  if result == 0 goto Withdraw
  subway check
  if result == 1 goto SameSpecies
  if result == 2 goto SameItem
  say "All aboard! Mind the doors."
  scene "Down to the train"
  fade out 0.6
  sound "crossing_engine"
  endscene
  warp "BattleSubwayTrain" 3 5 right
  fade in
  return
label SameSpecies
  say "I'm sorry. The Pokémon taken aboard must all be of different species."
  goto Withdraw
label SameItem
  say "I'm sorry. No two of the Pokémon taken aboard may hold the same item."
label Withdraw
  subway end
  say "Please ride with us again!"
  return
