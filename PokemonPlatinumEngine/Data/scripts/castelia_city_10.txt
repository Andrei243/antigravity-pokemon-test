# (FLAG_UNOVA_HIDE_HUGH_PIER).


script OnEnter
  call UnovaCatchUp
  if var VAR_UNOVA_CASTELIA == 1 goto Ashore
  end
label Ashore
  scene "Ashore at Castelia City"
  clearflag FLAG_UNOVA_HIDE_HUGH_PIER
  place player 12 10 left
  place 1 12 9 down
  show 1
  speaker "Captain"
  say "And here we are! Castelia City, the biggest city in all of Unova!"
  face player up
  face 1 player
  speaker "{rival}"
  say "Whoa... So this is Castelia. The towers go on forever! Team Plasma could be hiding anywhere in a place like this."
  face 1 left
  emote 1 exclaim
  say "Hey, look out there! That ship with the black sails, heading out of the bay... Something about it feels wrong."
  say "I'm going to have a look around. See you, {player}!"
  walk 1 up 8 fast
  hide 1
  setflag FLAG_UNOVA_HIDE_HUGH_PIER
  setvar VAR_UNOVA_CASTELIA 2
  endscene
  end


# The captain: back across the bay to Virbank City (Black 2's Script_1)
script Captain
  face 0 player
  speaker "Captain"
  ask "Shall we sail back to Virbank City?"
  if no goto Stay
  say "Right you are! Step this way!"
  fade out 0.8
  music stop
  sound "crossing_horn"
  text "The ship pulls away from the pier and heads back across the bay toward Virbank City..."
  warp "VirbankCity_Terminal" 6 6 up
  fade in
  end
label Stay
  say "Come and find me whenever you fancy a trip. I'm always on this pier!"
  end
