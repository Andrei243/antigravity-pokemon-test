# Bug Catcher Wade battles as every trainer does; beaten, he asks for the player's number for the Pokégear's phone,
# and once it is registered he calls back for a rematch on his day (Models/Pokegear.cs, phone.txt).


script Wade
  if not defeated self goto Fight
  if rematch self goto Rematch
  if flag FLAG_PHONE_CONTACT_WADE goto Known
  speaker "Wade"
  say "My bugs and I will be training hard from now on. Can I have your number? I'll call when we're ready."
  ask "Register Wade's number in your Pokégear?"
  if no goto Refused
  addcontact wade
  say "Thanks! I'll let you know if I find a really rare bug, too."
  end
label Refused
  say "Oh. Well, my bugs will still be here if you change your mind."
  end
label Known
  trainerline after
  end
label Fight
  trainerline before
  battle self
  end
label Rematch
  trainerline before
  battle self rematch
  end
